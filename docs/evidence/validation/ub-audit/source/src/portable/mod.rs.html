    1 //! Portable cube backend (aarch64 NEON, wasm SIMD128, scalar fallback): the
    2 //! cube as two 16-byte vectors, with the same byte layout, operations and
    3 //! results as the x86 backend (`simd.rs`), so that tables and searches are
    4 //! identical on every backend.
    5 //!
    6 //! `corners` holds bytes 0..16 (corners in 0..8: `piece | twist << 5`),
    7 //! `edges` bytes 16..32 (edges in 0..12: `piece | flip << 4`). The x86 code
    8 //! shuffles both halves with one 32-byte `vpshufb`, which works per 16-byte
    9 //! lane: here the halves are shuffled separately. Where an index byte
   10 //! carries orientation bits, it is masked to its piece first (`vpshufb`
   11 //! ignores bits 4..6 of an index; `tbl` returns 0 for indices from 16).
   12 //! Only the vector operations differ per target (`V16`); this file and
   13 //! `coord.rs` are shared.

   66 /// Bytes `16 * half .. 16 * half + 16` of `bytes`.
   67 const fn half(bytes: &[u8; 32], high: bool) -> Aligned16 {
   68     let mut result = [0u8; 16];
   69     let mut i = 0;
   70     while i < 16 {
   71         result[i] = bytes[i + if high { 16 } else { 0 }];
   72         i += 1;
   73     }
   74     Aligned16(result)
   75 }

   92 /// `halves` with every byte masked to its piece (`& 0x0F`): the index
   93 /// vectors of a lookup.

   95     let mut masked = bytes.0;

  104 /// The edge conjugation vectors of a symmetry (see
  105 /// `coord::edge_parts_sym_raw`): the piece classes of T and Tinv's eo by
  106 /// source position (edge halves of `SymEdge`), and the position permutation
  107 /// Tinv as lookup indices (0xFF: a byte outside the edges, 0 after the
  108 /// lookup).

  117 const fn sym_edge_line(sym: &SymEdge) -> SymEdgeLine {
  118     let mut classes = [0u8; 16];
  119     let mut positions = [0xFFu8; 16];
  120     let mut source_flips = [0u8; 16];
  121     let mut i = 0;
  122     while i < 16 {
  123         classes[i] = sym.classes.0[16 + i];
  124         source_flips[i] = sym.perm_eo.0[16 + i];
  125         let position = sym.perm.0[16 + i];
  126         // (0x80 marks a byte outside the edges in the x86 vector.)
  127         if position & 0x80 == 0 {
  128             positions[i] = position & 0x0F;
  129         }
  130         i += 1;
  131     }
  132     SymEdgeLine {
  133         classes: Aligned16(classes),
  134         positions: Aligned16(positions),
  135         source_flips: Aligned16(source_flips),
  136     }
  137 }

  186 /// Each of `N` constants as halves (`index`: as lookup indices).
  187 const fn map_halves<const N: usize>(table: &[Aligned32; N], index: bool) -> [Halves; N] {
  188     let empty = Halves {
  189         corners: Aligned16([0; 16]),
  190         edges: Aligned16([0; 16]),
  191     };
  192     let mut result = [empty; N];
  193     let mut i = 0;
  194     while i < N {
  195         result[i] = if index {
  196             index_halves(&table[i])
  197         } else {
  198             halves(&table[i])
  199         };
  200         i += 1;
  201     }
  202     result
  203 }

  205 /// Times 3, for `HotTables::co_low_x3` and `co_high_x3`.
  206 const fn times3<const N: usize>(table: [u32; N]) -> [u32; N] {
  207     let mut result = [0u32; N];
  208     let mut i = 0;
  209     while i < N {
  210         result[i] = 3 * table[i];
  211         i += 1;
  212     }
  213     result
  214 }

  216 pub static HOT_TABLES: HotTables = HotTables {
  217     moves: map_halves(&MOVES, true),
  218     move_orients: map_halves(&MOVE_ORIENTS, false),
  219     inverse_moves: map_halves(&INVERSE_MOVES, false),
  220     move_cubes: map_halves(&MOVES, false),
  221     sym_edges: {
  222         let empty = SymEdgeLine {
  223             classes: Aligned16([0; 16]),
  224             positions: Aligned16([0; 16]),
  225             source_flips: Aligned16([0; 16]),
  226         };
  227         let mut result = [empty; 48];
  228         let mut sym = 0;
  229         while sym < 48 {
  230             result[sym] = sym_edge_line(&SYM_EDGES[sym]);
  231             sym += 1;
  232         }
  233         result
  234     },
  235     sym_cubes: map_halves(&SYM_CUBES, false),
  236     sym_cube_inverses: map_halves(&SYM_CUBE_INVERSES, false),
  237     co_low: crate::coord::CO_LOW,
  238     co_high: crate::coord::CO_HIGH,
  239     co_low_x3: times3(crate::coord::CO_LOW),
  240     co_high_x3: times3(crate::coord::CO_HIGH),
  241     tetrads_x3: {
  242         // (Only masks of 4 corners occur; the others keep the 0xFF marker.)
  243         let mut result = [0xFFu8; 256];
  244         let mut i = 0;
  245         while i < 256 {
  246             if crate::coord::RANK8_CONST[i] != 0xFF {
  247                 result[i] = 3 * crate::coord::RANK8_CONST[i];
  248             }
  249             i += 1;
  250         }
  251         result
  252     },
  253     eslot_complement: {
  254         let mut result = [0u16; 4096];
  255         let mut mask = 0;
  256         while mask < 4096 {
  257             result[mask] = crate::coord::ESLOT_CONST[mask ^ 0xFFF];
  258             mask += 1;
  259         }
  260         result
  261     },
  262     rank8: crate::coord::RANK8_CONST,
  263     pext6: {
  264         let mut result = [[0u8; 64]; 64];
  265         let mut mask = 0;
  266         while mask < 64 {
  267             let mut x = 0;
  268             while x < 64 {
  269                 let (mut out, mut n, mut bit) = (0, 0, 0);
  270                 while bit < 6 {
  271                     if mask >> bit & 1 == 1 {
  272                         out |= (x >> bit & 1) << n;
  273                         n += 1;
  274                     }
  275                     bit += 1;
  276                 }
  277                 #[expect(clippy::cast_possible_truncation, reason = "6 bits")]
  278                 let packed = out as u8;
  279                 result[mask][x] = packed;
  280                 x += 1;
  281             }
  282             mask += 1;
  283         }
  284         result
  285     },
  286     popcnt6: {
  287         let mut result = [0u8; 64];
  288         let mut mask = 0u32;
  289         while mask < 64 {
  290             #[expect(clippy::cast_possible_truncation, reason = "at most 6")]
  291             let count = mask.count_ones() as u8;
  292             result[mask as usize] = count;
  293             mask += 1;
  294         }
  295         result
  296     },
  297     stat_pairs: crate::lanes::STAT_PAIRS,
  298     allowed_after: crate::solver::ALLOWED_AFTER_CONST,
  299     face_keep: {
  300         let mut result = [0u32; 6];
  301         let mut face = 0;
  302         while face < 6 {
  303             result[face] = !(7 << (3 * face));
  304             face += 1;
  305         }
  306         result
  307     },
  308     move_face: {
  309         let mut result = [0u8; 18];
  310         let mut mv = 0u8;
  311         while mv < 18 {
  312             result[mv as usize] = mv / 3;
  313             mv += 1;
  314         }
  315         result
  316     },
  317     move_kind: {
  318         let mut result = [0u8; 18];
  319         let mut mv = 0u8;
  320         while mv < 18 {
  321             result[mv as usize] = mv % 3;
  322             mv += 1;
  323         }
  324         result
  325     },
  326     move_keep: {
  327         let mut result = [0u32; 18];
  328         let mut mv = 0;
  329         while mv < 18 {
  330             result[mv] = !(7 << (3 * (mv / 3)));
  331             mv += 1;
  332         }
  333         result
  334     },
  335     x_move_mask: {
  336         let mut result = [0u32; 18];
  337         let mut mv = 0;
  338         while mv < 18 {
  339             result[mv] = if mv.is_multiple_of(3) { !0 } else { 0 };
  340             mv += 1;
  341         }
  342         result
  343     },

  365 /// The orientation bits of each piece (twists in bits 5..7, flips in bit 4).
  366 static ORIENT_BITS: Halves = Halves {
  367     corners: first_bytes(0x60, 8),
  368     edges: first_bytes(0x10, 12),

  370 /// The orientation that wraps to 0 (3 twists, 2 flips).

  414     #[must_use]
  415     pub fn from_cubie(cube: &CubieCube) -> Cube {
  416         let mut bytes = [0u8; 32];
  417         for (i, byte) in bytes[..8].iter_mut().enumerate() {
  418             *byte = cube.corner_perm[i] | (cube.corner_orient[i] << 5);
  419         }
  420         for (i, byte) in bytes[16..28].iter_mut().enumerate() {
  421             *byte = cube.edge_perm[i] | (cube.edge_orient[i] << 4);
  422         }
  423         Cube::from_bytes(bytes)
  424     }

  426     /// The 32 bytes, the unspecified ones zeroed.
  427     #[must_use]
  428     pub fn to_bytes(self) -> [u8; 32] {
  429         let used = USED_BYTES.load();
  430         let corners = self.corners.and(used.corners).to_array();
  431         let edges = self.edges.and(used.edges).to_array();
  432         let mut bytes = [0u8; 32];
  433         for (byte, x) in bytes.iter_mut().zip(corners.into_iter().chain(edges)) {
  434             *byte = x;
  435         }
  436         bytes
  437     }

  439     #[must_use]
  440     pub fn to_cubie(self) -> CubieCube {
  441         let bytes = self.to_bytes();
  442         let mut cube = CubieCube::SOLVED;
  443         for (i, &byte) in bytes[..8].iter().enumerate() {
  444             cube.corner_perm[i] = byte & 7;
  445             cube.corner_orient[i] = byte >> 5;
  446         }
  447         for (i, &byte) in bytes[16..28].iter().enumerate() {
  448             cube.edge_perm[i] = byte & 15;
  449             cube.edge_orient[i] = (byte >> 4) & 1;
  450         }
  451         cube
  452     }

  454     /// Orientation fix-up after adding two orientations (mod 3 for twists,
  455     /// mod 2 for flips): `min(x, x - ORIENT_WRAP)` per byte.
  456     #[inline(always)]
  457     fn wrap(corners: V16, edges: V16) -> Cube {
  458         let wrap = ORIENT_WRAP.load();
  459         Cube {
  460             corners: corners.min(corners.wrapping_sub(wrap.corners)),
  461             edges: edges.min(edges.wrapping_sub(wrap.edges)),
  462         }
  463     }

  465     /// "self then other".
  466     #[inline(always)]
  467     #[must_use]
  468     pub fn compose(self, other: Cube) -> Cube {
  469         let orient = ORIENT_BITS.load();
  470         let corners = self
  471             .corners
  472             .lookup(pieces(other.corners))
  473             .wrapping_add(other.corners.and(orient.corners));
  474         let edges = self
  475             .edges
  476             .lookup(pieces(other.edges))
  477             .wrapping_add(other.edges.and(orient.edges));
  478         Cube::wrap(corners, edges)
  479     }

  508             corners.wrapping_add(V16::load(&orient.corners.0)),

  534     /// Corner twists negated (mod 3): 1 <-> 2 (mirror symmetries).
  535     #[inline(always)]
  536     #[must_use]
  537     pub fn negate_corner_orient(self) -> Cube {
  538         let twist_bits = V16::load(&ORIENT_BITS.corners.0);
  539         let twists = self.corners.and(twist_bits);
  540         let negated = twists.shl1().or(twists.shr1()).and(twist_bits);
  541         Cube {
  542             corners: self.corners.xor(twists).or(negated),
  543             edges: self.edges,
  544         }

  547     /// Inverse cube (vcube's addition chain for c^27719, as in `simd.rs`).
  548     #[must_use]
  549     pub fn inverse(self) -> Cube {
  550         // The permutation of `a` after `b` (x86 `vpshufb(a, b)`).
  551         let shuffle = |a: Cube, b: Cube| Cube {
  552             corners: a.corners.lookup(pieces(b.corners)),
  553             edges: a.edges.lookup(pieces(b.edges)),
  554         };
  555         let c = self;
  556         let c3 = shuffle(shuffle(c, c), c);
  557         let mut x = shuffle(c3, c3);
  558         for step in [None, None, Some(c3), None, None, None, None, Some(c)] {
  559             x = shuffle(x, step.unwrap_or(x));
  560         }
  561         for step in [None, None, None, None, None, Some(c3), None, Some(c)] {
  562             x = shuffle(x, step.unwrap_or(x));
  563         }
  564         let (orient, used) = (ORIENT_BITS.load(), USED_BYTES.load());
  565         let orients = Cube {
  566             corners: c.corners.and(orient.corners),
  567             edges: c.edges.and(orient.edges),
  568         };
  569         let orients = shuffle(orients, x);
  570         Cube {
  571             corners: x
  572                 .corners
  573                 .and_not(orient.corners)
  574                 .or(orients.corners)
  575                 .and(used.corners),
  576             edges: x
  577                 .edges
  578                 .and_not(orient.edges)
  579                 .or(orients.edges)
  580                 .and(used.edges),
  581         }
  582         .negate_corner_orient()
  583     }
  584 
  585     #[inline(always)]
  586     #[must_use]
  587     pub fn equals(self, other: Cube) -> bool {
  588         let used = USED_BYTES.load();
  589         let corners = self.corners.xor(other.corners).and(used.corners);
  590         let edges = self.edges.xor(other.edges).and(used.edges);
  591         corners.or(edges).is_zero()
  592     }

  594     #[inline(always)]
  595     #[must_use]
  596     pub fn is_solved(self) -> bool {
  597         self.equals(Cube::solved())
  598     }

  600     /// Conjugate by symmetry `sym`.
  601     #[inline(always)]
  602     #[must_use]
  603     pub fn transform(self, sym: usize) -> Cube {
  604         let result = self.transform_edges(sym);
  605         if SYM_MIRROR[sym] {
  606             result.negate_corner_orient()
  607         } else {
  608             result
  609         }
  610     }

  612     /// Conjugate by symmetry `sym`; only the edges are guaranteed correct.
  613     #[inline(always)]
  614     #[must_use]
  615     pub fn transform_edges(self, sym: usize) -> Cube {
  616         HOT_TABLES.sym_cubes[sym]
  617             .load()
  618             .compose(self)
  619             .compose(HOT_TABLES.sym_cube_inverses[sym].load())
  620     }

  622     /// `if choose_a { a } else { b }`, as a byte blend (no branch).
  623     #[inline(always)]
  624     #[must_use]
  625     pub fn select(choose_a: bool, a: Cube, b: Cube) -> Cube {
  626         Cube {
  627             corners: V16::select(choose_a, a.corners, b.corners),
  628             edges: V16::select(choose_a, a.edges, b.edges),
  629         }
  630     }

  632     /// `select` by a mask that is all ones (`if_set`) or all zeros
  633     /// (`if_clear`), e.g. loaded from a table: its low byte is the blend
  634     /// mask of every byte (no compare).
  635     #[inline(always)]
  636     #[must_use]
  637     pub fn select_mask(mask: u32, if_set: Cube, if_clear: Cube) -> Cube {
  638         #[expect(clippy::cast_possible_truncation, reason = "all ones or all zeros")]
  639         let byte_mask = mask as u8;
  640         Cube {
  641             corners: V16::select_bytes(byte_mask, if_set.corners, if_clear.corners),
  642             edges: V16::select_bytes(byte_mask, if_set.edges, if_clear.edges),
  643         }
  644     }

  646     /// The cube that move `mv`'s child looks up first, for a node with cubes
  647     /// `moved` and `other`: `other.premove_inverse(mv)` for an X move
  648     /// (`x_move_mask` all ones), else `moved.apply_move(mv)` (`x_move_mask`
  649     /// 0). One composition of selected operands, as in `simd.rs`.
  650     ///
  651     /// # Safety
  652     /// `mv < 18`.
  653     #[inline(always)]
  654     #[must_use]
  655     pub unsafe fn child_lookup(moved: Cube, other: Cube, mv: usize, x_move_mask: u32) -> Cube {
  656         // SAFETY: `mv < 18` (the caller's contract).
  657         let (inverse_move, move_cube) = unsafe {
  658             (
  659                 HOT_TABLES.inverse_moves.get_unchecked(mv).load(),
  660                 HOT_TABLES.move_cubes.get_unchecked(mv).load(),
  661             )
  662         };
  663         let left = Cube::select_mask(x_move_mask, inverse_move, moved);
  664         let right = Cube::select_mask(x_move_mask, other, move_cube);
  665         left.compose(right)
  666     }

  693     /// Each operation of the SIMD `V16` gives the scalar reference's bytes,
  694     /// on random vectors and on lookup indices over the whole byte range.
  695     #[cfg(any(backend = "neon", backend = "wasm"))]
  696     #[test]
  697     fn v16_matches_scalar() {
  698         use v16_scalar::V16 as Scalar;
  699         let mut rng = Rng(9);
  700         let mut random = || -> [u8; 16] {
  701             let mut bytes = [0u8; 16];
  702             for chunk in bytes.chunks_mut(8) {
  703                 chunk.copy_from_slice(&rng.next_u64().to_le_bytes()[..chunk.len()]);
  704             }
  705             bytes
  706         };
  707         for round in 0..if cfg!(miri) { 50 } else { 20_000 } {
  708             let (a, b) = (random(), random());
  709             // Lookup indices: random bytes, or below 16 (the usual case).
  710             let mut indices = random();
  711             if round % 2 == 0 {
  712                 indices = indices.map(|x| x & 15);
  713             }
  714             let (va, vb, vi) = (V16::load(&a), V16::load(&b), V16::load(&indices));
  715             let (sa, sb, si) = (Scalar::load(&a), Scalar::load(&b), Scalar::load(&indices));
  716             let pairs = [
  717                 (va.lookup(vi), sa.lookup(si)),
  718                 (va.wrapping_add(vb), sa.wrapping_add(sb)),
  719                 (va.wrapping_sub(vb), sa.wrapping_sub(sb)),
  720                 (va.min(vb), sa.min(sb)),
  721                 (va.and(vb), sa.and(sb)),
  722                 (va.or(vb), sa.or(sb)),
  723                 (va.xor(vb), sa.xor(sb)),
  724                 (va.and_not(vb), sa.and_not(sb)),
  725                 (va.shl1(), sa.shl1()),
  726                 (va.shr1(), sa.shr1()),
  727                 (
  728                     V16::select(round % 3 == 0, va, vb),
  729                     Scalar::select(round % 3 == 0, sa, sb),
  730                 ),
  731                 (
  732                     V16::select_bytes(0xFF, va, vb),
  733                     Scalar::select_bytes(0xFF, sa, sb),
  734                 ),
  735                 (
  736                     V16::select_bytes(0, va, vb),
  737                     Scalar::select_bytes(0, sa, sb),
  738                 ),
  739                 (V16::splat(a[0]), Scalar::splat(a[0])),
  740             ];
  741             for (simd, scalar) in pairs {
  742                 assert_eq!(simd.to_array(), scalar.to_array());
  743             }
  744             assert_eq!(va.low_u64(), sa.low_u64());
  745             assert_eq!(va.bit_mask::<2>(), sa.bit_mask::<2>());
  746             assert_eq!(va.bit_mask::<3>(), sa.bit_mask::<3>());
  747             assert_eq!(va.bit_mask::<4>(), sa.bit_mask::<4>());
  748             assert_eq!(va.bit_masks_764(vb), sa.bit_masks_764(sb));
  749             let sparse = a.map(|x| if x < 32 { x } else { 0 });
  750             assert_eq!(
  751                 V16::load(&sparse).is_zero(),
  752                 Scalar::load(&sparse).is_zero()
  753             );
  754             assert!(V16::splat(0).is_zero());
  755         }
  756     }

  758     #[test]
  759     fn portable_cube_matches_cubie() {
  760         let mut rng = Rng(5);
  761         let syms = symmetries();
  762         for _ in 0..if cfg!(miri) { 20 } else { 200 } {
  763             let a = CubieCube::random(&mut rng);
  764             let b = CubieCube::random(&mut rng);
  765             let (fa, fb) = (Cube::from_cubie(&a), Cube::from_cubie(&b));
  766             assert_eq!(fa.to_cubie(), a);
  767             assert_eq!(fa.compose(fb).to_cubie(), a.compose(&b));
  768             assert_eq!(fa.inverse().to_cubie(), a.inverse());
  769             for m in Move::ALL {
  770                 assert_eq!(fa.apply_move(m).to_cubie(), a.apply_move(m));
  771                 assert_eq!(
  772                     fa.inverse().premove_inverse(m).to_cubie(),
  773                     a.apply_move(m).inverse()
  774                 );
  775             }
  776             for (sym, symmetry) in syms.iter().enumerate() {
  777                 assert_eq!(fa.transform(sym).to_cubie(), conjugate(&a, symmetry));
  778             }
  779             assert!(Cube::select(true, fa, fb) == fa && Cube::select(false, fa, fb) == fb);
  780             let (inverse_a, inverse_b) = (fa.inverse(), fb.inverse());
  781             for (mv, m) in (0..18).zip(Move::ALL) {
  782                 // SAFETY: `mv < 18`.
  783                 unsafe {
  784                     assert_eq!(Cube::child_lookup(fa, inverse_b, mv, 0), fa.apply_move(m));
  785                     assert_eq!(
  786                         Cube::child_lookup(inverse_a, fb, mv, !0),
  787                         fb.premove_inverse(m)
  788                     );
  789                     assert_eq!(fa.apply_move_unchecked(mv), fa.apply_move(m));
  790                     assert_eq!(fa.premove_inverse_unchecked(mv), fa.premove_inverse(m));
  791                 }
  792             }
  793         }
  794         assert!(Cube::from_cubie(&CubieCube::SOLVED).is_solved());
  795     }
