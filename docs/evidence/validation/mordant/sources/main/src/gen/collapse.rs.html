    1 //! hk tables from tables of a larger k with the same base, and table
    2 //! comparison.
    3 //!
    4 //! An hk entry is the minimum over the hK entries (K > k) that differ only in
    5 //! the eo bits hk drops, i.e. over the 2^(K - k) rows `r << (K - k) ..` of
    6 //! hK at the same slot; a group minimum is the minimum of those groups'
    7 //! minima. With the same base the 2-bit codes collapse the same way
    8 //! (`min(code)` is the code of the minimum), so for example h7..h10 of base
    9 //! 10 follow from h11, and this checks directly generated tables.
   10

   17 /// Per 2-bit field minimum of two words of codes.
   18 #[inline(always)]
   19 fn min_codes(x: u64, y: u64) -> u64 {
   20     const LO: u64 = 0x5555_5555_5555_5555;
   21     let (x0, x1) = (x & LO, (x >> 1) & LO);
   22     let (y0, y1) = (y & LO, (y >> 1) & LO);
   23     let hi = x1 & y1;
   24     // Equal high bits: the low bits' AND; else the low bit of the one with
   25     // high bit 0.
   26     let lo = ((!(x1 ^ y1)) & x0 & y0) | (x1 & !y1 & y0) | (!x1 & y1 & x0);
   27     (hi << 1) | (lo & LO)
   28 }

   30 /// The word `w` (of a row) of the collapse of `rows` (`16 * n` words, n
   31 /// rows of the larger table).
   32 #[inline(always)]
   33 fn collapsed_word(rows: &[AtomicU64], w: usize) -> u64 {
   34     let mut out = u64::MAX;
   35     let mut min = 15;
   36     for r in rows.as_chunks::<WORDS_PER_ROW>().0 {
   37         let x = r[w].load(Relaxed);
   38         out = min_codes(out, x);
   39         min = min.min(x & 15);
   40     }
   41     if w.is_multiple_of(2) {
   42         (out & !15) | min
   43     } else {
   44         out
   45     }
   46 }

   48 /// The rows of hk and the number of rows of h`larger_k` (`k <= larger_k`)
   49 /// that collapse into each, if `larger_words` holds them all.
   50 fn collapse_shape(
   51     k: u32,
   52     larger_k: u32,
   53     larger_words: usize,
   54 ) -> Result<(usize, usize), TableError> {
   55     let shift = larger_k
   56         .checked_sub(k)
   57         .ok_or_else(|| TableError::Generate(format!("h{k} is not a collapse of h{larger_k}")))?;
   58     let rows = usize::try_from(h48_rows(k)).map_err(|_| TableError::TooLarge(k))?;
   59     let factor = 1usize.checked_shl(shift).unwrap_or(0);
   60     // (Then every slice of rows below is within the larger table.)
   61     match factor
   62         .checked_mul(WORDS_PER_ROW)
   63         .and_then(|x| x.checked_mul(rows))
   64     {
   65         Some(words) if factor > 0 && words <= larger_words => Ok((rows, factor)),
   66         _ => Err(TableError::TooLarge(larger_k)),
   67     }
   68 }

   70 /// hk of the same base from `src` (k <= src's k), in anonymous memory.
   71 ///
   72 /// # Errors
   73 /// If `k` is above `src`'s, the memory cannot be allocated, a table is larger
   74 /// than its memory or a slice can be, or a thread panicked.
   75 pub fn derive(source: &H48Table, k: u32, threads: usize) -> Result<H48Table, TableError> {
   76     let source_words = source.memory.words()?;
   77     let (rows, factor) = collapse_shape(k, source.meta.k, source_words.len())?;
   78     let memory =
   79         Memory::anonymous(usize::try_from(h48_bytes(k)).map_err(|_| TableError::TooLarge(k))?)?;
   80     let words = memory.words()?;
   81     par_rows(rows, threads, |row| {
   82         let covered = &source_words[row * factor * 16..(row + 1) * factor * 16];
   83         for w in 0..16 {
   84             words[row * 16 + w].store(collapsed_word(covered, w), Relaxed);
   85         }
   86     })?;
   87     Ok(H48Table {
   88         memory,
   89         meta: H48Meta {
   90             k,
   91             base: source.meta.base,
   92         },
   93     })
   94 }

   96 /// Differences found by `compare`.
   97 #[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
   98 pub struct Diff {
   99     /// Real entries compared, and those whose codes differ.
  100     pub entries: u64,
  101     pub codes: u64,
  102     /// Group minima that differ, among those that matter (at most the base
  103     /// on either side; larger ones are never read: a group minimum is used
  104     /// only for a code-0 entry, whose depth is at most the base).
  105     pub group_minima: u64,
  106     /// Group minima above the base that differ (harmless).
  107     pub unread_group_minima: u64,
  108 }

  110 /// Compare `table` with the collapse of `larger` to `table`'s k (the same
  111 /// table if the ks are equal).
  112 ///
  113 /// # Errors
  114 /// If the bases differ, `table`'s k is above `larger`'s, a table is larger
  115 /// than its memory or a slice can be, or a thread panicked.
  116 pub fn compare(table: &H48Table, larger: &H48Table, threads: usize) -> Result<Diff, TableError> {
  117     if table.meta.base != larger.meta.base {
  118         return Err(TableError::Generate(format!(
  119             "tables of different bases ({} and {})",
  120             table.meta.base, larger.meta.base
  121         )));
  122     }
  123     let base = u64::from(table.meta.base);
  124     let (words, larger_words) = (table.memory.words()?, larger.memory.words()?);
  125     let (rows, factor) = collapse_shape(table.meta.k, larger.meta.k, larger_words.len())?;
  126     if words.len() < rows * WORDS_PER_ROW {
  127         return Err(TableError::TooLarge(table.meta.k));
  128     }
  129     let total = [(); 4].map(|()| AtomicU64::new(0));
  130     par_rows_chunked(rows, threads, |first, end| {
  131         let mut counts = [0u64; 4];
  132         for row in first..end {
  133             let collapsed = &larger_words[row * factor * 16..(row + 1) * factor * 16];
  134             for w in 0..16 {
  135                 let x = words[row * 16 + w].load(Relaxed);
  136                 let y = if factor == 1 {
  137                     collapsed[w].load(Relaxed)
  138                 } else {
  139                     collapsed_word(collapsed, w)
  140                 };
  141                 let mask = entry_low_bits(w as u64);
  142                 counts[0] += u64::from(mask.count_ones());
  143                 let differ = (x ^ y) | ((x ^ y) >> 1);
  144                 counts[1] += u64::from((differ & mask).count_ones());
  145                 if w.is_multiple_of(2) && (x & 15) != (y & 15) {
  146                     if (x & 15).min(y & 15) <= base {
  147                         counts[2] += 1;
  148                     } else {
  149                         counts[3] += 1;
  150                     }
  151                 }
  152             }
  153         }
  154         for (sum, count) in total.iter().zip(counts) {
  155             sum.fetch_add(count, Relaxed);
  156         }
  157     })?;
  158     let [entries, codes, group_minima, unread_group_minima] = total.map(AtomicU64::into_inner);
  159     Ok(Diff {
  160         entries,
  161         codes,
  162         group_minima,
  163         unread_group_minima,
  164     })
  165 }

  167 fn par_rows(rows: usize, threads: usize, f: impl Fn(usize) + Sync) -> Result<(), WorkerPanicked> {
  168     par_rows_chunked(rows, threads, |a, b| (a..b).for_each(&f))
  169 }
  170 
  171 fn par_rows_chunked(
  172     rows: usize,
  173     threads: usize,
  174     f: impl Fn(usize, usize) + Sync,
  175 ) -> Result<(), WorkerPanicked> {
  176     let chunk = 1 << 12;
  177     let next = std::sync::atomic::AtomicUsize::new(0);
  178     crate::parallel::on_threads(threads, |_| loop {
  179         let a = next.fetch_add(chunk, Relaxed);
  180         if a >= rows {
  181             break;
  182         }
  183         f(a, (a + chunk).min(rows));
  184     })?;
  185     Ok(())
  186 }
