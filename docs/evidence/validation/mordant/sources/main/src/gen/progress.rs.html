    1 //! Progress of the table generators. Each finished step is reported as a
    2 //! line, printed to stderr unless a `Reporter` is installed; the `gen`
    3 //! binary installs one on a terminal, which also draws a progress bar of
    4 //! the current step. Reports cost an atomic load each, per chunk of work.
    5 
    6 use std::fmt::Arguments;
    7 use std::sync::OnceLock;
    8 
    9 /// Receives the generators' progress (from any thread).
   10 pub trait Reporter: Send + Sync {
   11     /// A finished step, as the line printed without a reporter.
   12     fn line(&self, line: &str);
   13     /// A step (`name`) of `total` units starts.
   14     fn start(&self, name: &str, total: u64);
   15     /// `units` more units of the current step are done.
   16     fn advance(&self, units: u64);
   17 }
   18 
   19 static REPORTER: OnceLock<Box<dyn Reporter>> = OnceLock::new();
   20 
   21 /// Send all progress to `reporter` from now on. Returns false (and drops
   22 /// it) if one is installed already.
   23 pub fn install(reporter: Box<dyn Reporter>) -> bool {
   24     REPORTER.set(reporter).is_ok()
   25 }
   26 
   27 /// Whether a reporter is installed.
   28 #[must_use]
   29 pub fn installed() -> bool {
   30     REPORTER.get().is_some()
   31 }
   32 
   33 /// Report a finished step: to the reporter, else to stderr.
   34 pub fn line(line: Arguments<'_>) {
   35     match REPORTER.get() {
   36         Some(reporter) => reporter.line(&line.to_string()),
   37         None => eprintln!("{line}"),
   38     }
   39 }
   40 
   41 /// Start a step of `total` units (for the reporter only).
   42 pub fn start(name: Arguments<'_>, total: u64) {
   43     if let Some(reporter) = REPORTER.get() {
   44         reporter.start(&name.to_string(), total);
   45     }
   46 }
   47 
   48 /// `units` more units of the current step are done (for the reporter only).
   49 pub fn advance(units: u64) {
   50     if let Some(reporter) = REPORTER.get() {
   51         reporter.advance(units);
   52     }
   53 }
