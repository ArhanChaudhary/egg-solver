   27     /// Start logging to stderr: warnings only by default (nothing in normal
   28     /// runs, so that stderr is as before), more with each `-v`.
   29     pub fn init_logging(self) {
   30         let level = match self.verbose {
   31             0 => log::LevelFilter::Warn,
   32             1 => log::LevelFilter::Info,
   33             2 => log::LevelFilter::Debug,
   34             _ => log::LevelFilter::Trace,
   35         };
   36         let start = Instant::now();
   37         env_logger::Builder::new()
   38             .filter_level(level)
   39             .parse_default_env()
   40             .format(move |out, record| {
   41                 writeln!(
   42                     out,
   43                     "[{:8.3}s {} {}] {}",
   44                     start.elapsed().as_secs_f64(),
   45                     record.level(),
   46                     record.target(),
   47                     record.args()
   48                 )
   49             })
   50             .init();
   51     }

  116 /// `--load` (an empty `EGG_SOLVER_LOAD` is `auto`, as before clap).
  117 fn parse_load_mode(text: &str) -> Result<LoadMode, String> {
  118     match text {
  119         "auto" | "" => Ok(LoadMode::Auto),
  120         "map" => Ok(LoadMode::Map),
  121         "copy" => Ok(LoadMode::Copy),
  122         "shm" => Ok(LoadMode::Shm),
  123         _ => Err("expected auto, map, copy or shm".to_string()),
  124     }
  125 }

  168     /// `open`, and how the table was loaded.
  169     ///
  170     /// # Errors
  171     /// As `open`.
  172     pub fn open_reported(self, path: &str) -> anyhow::Result<(H48Table, LoadReport)> {
  173         self.preflight(path)?;
  174         self.open_logged(path).map_err(|error| {
  175             // (Most messages name the file already.)
  176             let what = if error.to_string().contains(path) {
  177                 "cannot open table".to_string()
  178             } else {
  179                 format!("cannot open table {path}")
  180             };
  181             anyhow::Error::new(error).context(BadInput(what))
  182         })
  183     }

  185     /// Before the table is loaded: that the memory for it is there
  186     /// (`preflight`), for a copy (`--load copy`, or `auto` off hugetlbfs and
  187     /// /dev/shm on Linux) and for a new shared memory segment that only 4
  188     /// KiB pages are left for. Nothing is needed when `auto` or `--shm`
  189     /// attaches the table's segment, or for a segment on huge pages (pool
  190     /// pages, which `tables::shm` checks itself).
  191     fn preflight(self, path: &str) -> anyhow::Result<()> {
  192         let options = self.options();
  193         let mode = options.mode_for(path);
  194         if !matches!(mode, LoadMode::Copy | LoadMode::Shm) {
  195             return Ok(());
  196         }
  197         // (A table that does not open fails in `open_logged`.)
  198         let Ok(meta) = crate::tables::read_meta(path) else {
  199             return Ok(());
  200         };
  201         if matches!(options.mode, LoadMode::Auto | LoadMode::Shm) && shm::loaded(path, &options) {
  202             return Ok(());
  203         }
  204         if mode == LoadMode::Copy {
  205             return preflight::memory(
  206                 &format!("copying table {path} (h{}) into memory", meta.k),
  207                 h48_bytes(meta.k),
  208                 "use a copy on hugetlbfs (mapped, not copied: --load map) or a smaller table",
  209             );
  210         }
  211         if shm::new_segment_page(path, &options) == Some(shm::NORMAL_PAGE) {
  212             preflight::memory(
  213                 &format!(
  214                     "a shared memory segment of table {path} (h{}) on 4 KiB pages",
  215                     meta.k
  216                 ),
  217                 shm::segment_bytes(h48_bytes(meta.k), shm::NORMAL_PAGE),
  218                 "allow huge pages (README.md \"Shared memory\") or use a smaller table",
  219             )?;
  220         }
  221         Ok(())
  222     }

  224     /// `open_reported`, with the table's own error.
  225     fn open_logged(self, path: &str) -> Result<(H48Table, LoadReport), TableError> {
  226         let options = self.options();
  227         let mode = options.mode_for(path);
  228         log::info!(
  229             "table {path}: {mode:?}{}{}",
  230             if options.mode == LoadMode::Auto && cfg!(target_os = "linux") {
  231                 " (or the table's shared memory segment, if there)"
  232             } else {
  233                 ""
  234             },
  235             if matches!(mode, LoadMode::Map | LoadMode::Shm) && !options.prefault {
  236                 ", pages read on first use"
  237             } else {
  238                 ""
  239             }
  240         );
  241         let start = Instant::now();
  242         let mut waiting = false;
  243         let (table, report) = loop {
  244             match H48Table::open_reporting(path, &options) {
  245                 // (Another process is filling the segment: wait for it.)
  246                 Err(TableError::ShmLoading { key, pid }) => {
  247                     if !waiting {
  248                         log::warn!(
  249                             "table {path}: shared memory segment {key:#010x} is being loaded by \
  250                              process {pid}: waiting for it"
  251                         );
  252                         waiting = true;
  253                     }
  254                     std::thread::sleep(std::time::Duration::from_millis(100));
  255                 }
  256                 result => break result?,
  257             }
  258         };
  259         for note in &report.notes {
  260             log::info!("table {path}: {note}");
  261         }
  262         let len = usize::try_from(h48_bytes(table.meta.k)).unwrap_or(usize::MAX);
  263         log::info!(
  264             "table {path}: h{} base {}, {:.2} GiB, {:?}, loaded in {:.2}s",
  265             table.meta.k,
  266             table.meta.base,
  267             gib(len),
  268             report.mode,
  269             start.elapsed().as_secs_f64()
  270         );
  271         if let Some(segment) = &report.segment {
  272             log_segment(path, segment, &report.notes, options.shm_pages);
  273         }
  274         if log::log_enabled!(log::Level::Info) {
  275             log::info!(
  276                 "table {path}: {}",
  277                 page_report(table.memory.ptr().addr(), len)
  278             );
  279         }
  280         Ok((table, report))
  281     }

  283     /// `solve --shm-load`: make the table's shared memory segment from the
  284     /// file (or check the one there) and print it, without solving.
  285     ///
  286     /// # Errors
  287     /// As `open` with `--load shm`.
  288     pub fn shm_load(self, path: &str) -> anyhow::Result<ExitCode> {
  289         let args = LoadArgs { shm: true, ..self };
  290         let start = Instant::now();
  291         let (table, report) = args.open_reported(path)?;
  292         let seconds = start.elapsed().as_secs_f64();
  293         let segment = report
  294             .segment
  295             .ok_or_else(|| anyhow::anyhow!("no segment"))?;
  296         let how = if segment.created {
  297             format!("made from {path} in {seconds:.2}s")
  298         } else {
  299             format!(
  300                 "already loaded (from {} by process {})",
  301                 segment.source, segment.creator
  302             )
  303         };
  304         println!(
  305             "segment {:#010x} shmid {}: h{} base {}, {:.2} GiB on {} pages, {how}",
  306             segment.key,
  307             segment.id,
  308             table.meta.k,
  309             table.meta.base,
  310             gib(usize::try_from(segment.bytes).unwrap_or(usize::MAX)),
  311             shm::page_name(segment.page_bytes),
  312         );
  313         Ok(ExitCode::SUCCESS)
  314     }

  316     /// `solve --shm-remove`: remove the table's shared memory segment (as
  317     /// `ipcrm -M KEY`; the key from the table's k and base, or `--shm-key`).
  318     ///
  319     /// # Errors
  320     /// If the table's k and base cannot be read, or the segment cannot be
  321     /// removed (another user's).
  322     pub fn shm_remove(self, path: &str) -> anyhow::Result<ExitCode> {
  323         let key = if let Some(key) = self.shm_key {
  324             key
  325         } else {
  326             let meta = crate::tables::read_meta(path).map_err(|error| {
  327                 anyhow::Error::new(error).context(BadInput(format!("cannot read table {path}")))
  328             })?;
  329             shm::key(meta.k, meta.base)
  330         };
  331         match shm::remove(key)? {
  332             Some(removed) => {
  333                 let still = match removed.attached {
  334                     0 => String::new(),
  335                     n => format!(" (attached by {n} processes: freed when they detach)"),
  336                 };
  337                 println!(
  338                     "removed segment {key:#010x} shmid {}: {}, {:.2} GiB{still}",
  339                     removed.id,
  340                     removed.held,
  341                     gib(usize::try_from(removed.bytes).unwrap_or(usize::MAX))
  342                 );
  343             }
  344             None => println!("no segment {key:#010x} for {path}"),
  345         }
  346         Ok(ExitCode::SUCCESS)
  347     }
  348 }

  350 /// Log the table's shared memory segment: its page size, and whether it was
  351 /// made or found; warn if it is on 4 KiB pages although huge pages were
  352 /// allowed (much slower).
  353 fn log_segment(path: &str, segment: &Segment, notes: &[String], allowed: ShmPages) {
  354     let key = segment.key;
  355     let how = if segment.created {
  356         "made from the file".to_string()
  357     } else {
  358         format!(
  359             "attached (made by process {} from {})",
  360             segment.creator, segment.source
  361         )
  362     };
  363     log::info!(
  364         "table {path}: shared memory segment {key:#010x} (shmid {}), {:.2} GiB on {} pages: {how}",
  365         segment.id,
  366         gib(usize::try_from(segment.bytes).unwrap_or(usize::MAX)),
  367         shm::page_name(segment.page_bytes)
  368     );
  369     if segment.page_bytes == shm::NORMAL_PAGE && allowed.sizes().any(|page| page > shm::NORMAL_PAGE)
  370     {
  371         let fix = if segment.created {
  372             notes.join("; ")
  373         } else {
  374             format!(
  375                 "to move it to huge pages, remove it (`solve --shm-remove {path}`) and load it \
  376                  again once they can be had"
  377             )
  378         };
  379         log::warn!(
  380             "table {path}: shared memory segment {key:#010x} is on 4 KiB pages, much slower \
  381              than huge pages (h7: 27% fewer solves/s at 1 thread, 32% at 48, than on 1 GiB \
  382              pages; README.md \"Shared memory\"); {fix}"
  383         );
  384     }
  385 }

  395 /// The page sizes of the memory at `start..start + len` (from
  396 /// `/proc/self/smaps`): 1 GiB or 2 MiB pages (hugetlbfs), else how much of
  397 /// it is in transparent huge pages.
  398 #[cfg(target_os = "linux")]
  399 fn page_report(start: usize, len: usize) -> String {
  400     let Ok(smaps) = std::fs::read_to_string("/proc/self/smaps") else {
  401         return "page sizes unknown (no /proc/self/smaps)".to_string();
  402     };
  403     let end = start.saturating_add(len);
  404     // (KiB, as smaps gives them.)
  405     let (mut page, mut resident, mut huge) = (0u64, 0u64, 0u64);
  406     let mut inside = false;
  407     for line in smaps.lines() {
  408         let mut words = line.split_whitespace();
  409         let first = words.next().unwrap_or("");
  410         // A mapping's header line starts with its address range.
  411         let range = first.split_once('-').and_then(|(a, b)| {
  412             Some((
  413                 usize::from_str_radix(a, 16).ok()?,
  414                 usize::from_str_radix(b, 16).ok()?,
  415             ))
  416         });
  417         if let Some((a, b)) = range {
  418             inside = a < end && b > start;
  419             continue;
  420         }
  421         let value = words.next().and_then(|v| v.parse::<u64>().ok());
  422         match (inside, first, value) {
  423             (true, "KernelPageSize:", Some(kib)) => page = page.max(kib),
  424             (true, "Rss:", Some(kib)) => resident += kib,
  425             (true, "AnonHugePages:" | "FilePmdMapped:" | "ShmemPmdMapped:", Some(kib)) => {
  426                 huge += kib;
  427             }
  428             _ => {}
  429         }
  430     }
  431     match page {
  432         0 => "page sizes unknown (not in /proc/self/smaps)".to_string(),
  433         1_048_576 => "1 GiB pages (hugetlbfs)".to_string(),
  434         2048 => "2 MiB pages (hugetlbfs)".to_string(),
  435         _ => {
  436             #[expect(clippy::cast_precision_loss, reason = "a report")]
  437             let share = 100.0 * huge as f64 / resident.max(1) as f64;
  438             format!(
  439                 "{page} KiB pages, {share:.0}% of {:.2} GiB resident in transparent huge pages",
  440                 gib(usize::try_from(resident << 10).unwrap_or(usize::MAX))
  441             )
  442         }
  443     }
  444 }

  452 /// Lanes per thread for `threads` threads on the table at `path` (of `k`):
  453 /// `asked`, else from `<path>.tune` (`bench tune`), else the untuned
  454 /// default for `k` (`tables::file::default_lanes`). Logs where it came from.
  455 #[must_use]
  456 pub fn lanes(asked: Option<usize>, path: &str, k: u32, threads: usize) -> usize {
  457     let lanes = asked.unwrap_or_else(|| crate::tables::file::default_lanes(path, k)(threads));
  458     let source = if asked.is_some() {
  459         "as asked"
  460     } else if std::path::Path::new(&format!("{path}.tune")).exists() {
  461         "from the table's .tune file"
  462     } else {
  463         "untuned default"
  464     };
  465     log::info!("{threads} threads, {lanes} lanes per thread ({source})");
  466     lanes
  467 }

  482 /// The exit status of a binary's run: its own if it ran, else (after
  483 /// printing `name: error: cause: ...` to stderr) 2 for a `BadInput`, 1 for
  484 /// other errors.
  485 #[must_use]
  486 pub fn exit_status(name: &str, result: anyhow::Result<ExitCode>) -> ExitCode {
  487     match result {
  488         Ok(status) => status,
  489         Err(error) => {
  490             eprintln!("{name}: {error:#}");
  491             if error.downcast_ref::<BadInput>().is_some() {
  492                 ExitCode::from(2)
  493             } else {
  494                 ExitCode::FAILURE
  495             }
  496         }
  497     }
