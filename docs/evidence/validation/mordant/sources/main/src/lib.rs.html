    1 //! A fast optimal solver for the 3x3x3 Rubik's cube.
    2 //!
    3 //! It finds a shortest move sequence (in the half-turn metric: any turn of
    4 //! a face is one move) with IDA*: depth-first searches with an increasing
    5 //! depth limit, pruned by lower bounds on the distance to solved read from
    6 //! precomputed tables. Each search runs on the cube and on its inverse at
    7 //! once, moving whichever side has fewer moves to try.
    8 //!
    9 //! Modules, from the bottom up:
   10 //! * `cubie`: the reference cube model (pieces and their orientations).
   11 //! * `simd`: the fast cube (one AVX2 register, or two 16-byte vectors on
   12 //!   other targets: `portable`) and its moves and symmetries.
   13 //! * `coord`: coordinates, the table indices of a cube.
   14 //! * `tables`: the corner table and the big H48 table (and its files,
   15 //!   `tables::file`); `gen` builds the latter, for any k.
   16 //! * `solver`: a plain recursive IDA*, the reference search.
   17 //! * `lanes`: the fast search: many cubes per thread, interleaved so that
   18 //!   table loads overlap; `lanes::share` lets all threads work on a few
   19 //!   cubes.
   20 //! * `symstart`: for cubes with a symmetry, one job of each class of
   21 //!   equivalent job prefixes.
   22 //! * `cli` (feature `cli`, on by default): what the binaries share.
   23

   68 /// Generate a table by BFS on h0 or h11, plus derived tables (each
   69 /// `(k, base, file)`). Returns the main table and the derived tables (each
   70 /// with its own memory); tables with a file are written to it.
   71 ///
   72 /// # Errors
   73 /// If `k` is not 0 or 11, a derived `k` is above `k`, memory cannot be
   74 /// allocated, or a table file cannot be created or written.
   75 pub fn generate(
   76     corners: &Cocsep,
   77     k: u32,
   78     base: u8,
   79     out: Option<&str>,
   80     derived: &[(u32, u8, Option<String>)],
   81     threads: usize,
   82 ) -> Result<(H48Table, Vec<H48Table>), TableError> {
   83     if k != 0 && k != 11 {
   84         return Err(TableError::Generate(format!(
   85             "BFS only on true coordinates (h0, h11), not h{k}"
   86         )));
   87     }
   88     let allocate = |k: u32, path: Option<&str>| -> Result<Memory, TableError> {
   89         let len = h48_bytes(k);
   90         let too_large = |_| TableError::TooLarge(k);
   91         match path {
   92             Some(path) => {
   93                 // (Room for the trailer after the data.)
   94                 let file_len = tables::file::file_bytes(len, tables::file::is_hugetlbfs(path));
   95                 let file_len = usize::try_from(file_len).map_err(too_large)?;
   96                 Memory::map_file(path, file_len, true).map_err(tables::file::io_error(path))
   97             }
   98             None => Memory::anonymous(usize::try_from(len).map_err(too_large)?),
   99         }
  100     };
  101     let memory = allocate(k, out)?;
  102     let derived_memory: Vec<Memory> = derived
  103         .iter()
  104         .map(|(dk, _, path)| allocate(*dk, path.as_deref()))
  105         .collect::<Result<_, _>>()?;
  106     let derived_words = derived_memory
  107         .iter()
  108         .map(Memory::words)
  109         .collect::<Result<Vec<_>, _>>()?;
  110     let generator = gen::Generator {
  111         corners,
  112         k,
  113         base,
  114         threads,
  115         derived: derived
  116             .iter()
  117             .zip(derived_words)
  118             .map(|((dk, dbase, _), words)| {
  119                 (
  120                     H48Meta {
  121                         k: *dk,
  122                         base: *dbase,
  123                     },
  124                     words,
  125                 )
  126             })
  127             .collect(),
  128     };
  129     let _counts = generator.run(&memory)?;
  130     let table = H48Table {
  131         memory,
  132         meta: H48Meta { k, base },
  133     };
  134     if let Some(path) = out {
  135         tables::file::write_trailer(path, &table.trailer(threads)?)?;
  136     }
  137     let derived_tables = derived
  138         .iter()
  139         .zip(derived_memory)
  140         .map(|((dk, dbase, path), memory)| {
  141             let table = H48Table {
  142                 memory,
  143                 meta: H48Meta {
  144                     k: *dk,
  145                     base: *dbase,
  146                 },
  147             };
  148             if let Some(path) = path {
  149                 tables::file::write_trailer(path, &table.trailer(threads)?)?;
  150             }
  151             Ok(table)
  152         })
  153         .collect::<Result<_, TableError>>()?;
  154     Ok((table, derived_tables))
  155 }

  157 /// Generate hk of any k (0..=11) in anonymous memory: by BFS for the true
  158 /// coordinates h0 and h11, else by `gen::dfs`.
  159 ///
  160 /// # Errors
  161 /// If `k` is above 11 or `base` above 12, memory cannot be allocated, or a
  162 /// thread panicked.
  163 pub fn generate_table(
  164     corners: &Cocsep,
  165     k: u32,
  166     base: u8,
  167     threads: usize,
  168     verbose: bool,
  169 ) -> Result<H48Table, TableError> {
  170     if k == 0 || k == 11 {
  171         return Ok(generate(corners, k, base, None, &[], threads)?.0);
  172     }
  173     let settings = gen::dfs::Settings {
  174         k,
  175         base,
  176         threads,
  177         prefix_depth: gen::dfs::PREFIX_DEPTH.min(base.saturating_add(2)),
  178         verbose,
  179     };
  180     let generator = gen::dfs::Generator::new(corners, settings)?;
  181     let len = usize::try_from(h48_bytes(k)).map_err(|_| TableError::TooLarge(k))?;
  182     let memory = Memory::anonymous(len)?;
  183     let rows = 0..h48_rows(k);
  184     let part = gen::dfs::Part {
  185         words: memory.words()?,
  186         first_slot: 0,
  187         end_slot: rows.end * coord::N_SLOTS as u64,
  188     };
  189     generator.generate(&rows, &part)?;
  190     Ok(H48Table {
  191         memory,
  192         meta: H48Meta { k, base },
  193     })
  194 }
