    1 //! Table files: where a table's `k` and `base` are stored, and how its data
    2 //! is brought into memory.
    3 //!
    4 //! A table file holds the table data at offset 0 (so that it can be mapped
    5 //! as is), followed by a 4 KiB text trailer in the last 4 KiB of the file:
    6 //! a magic line, then `key value` lines (`layout`, `k`, `base`, `bytes`, and
    7 //! the code `distribution` and `checksum` of the data when known), padded
    8 //! with NUL bytes. Unknown keys are ignored, except that a `layout` other
    9 //! than `h48` (the index layout of `coord`) is refused: a variant table (say
   10 //! with a parity bit in place of an eo bit) must not be read as an h48 one. Files on hugetlbfs are whole 1 GiB pages; their trailer sits in the
   11 //! slack after the data, so tables made before trailers existed can be
   12 //! stamped in place (`gen stamp`). Tables without a trailer are described by
   13 //! a `<path>.meta` sidecar (`k base`), or, for the legacy tables on
   14 //! `/mnt/huge`, by `tables/<name>.hugemeta` in the build folder. A table can
   15 //! also be loaded into a shared memory segment that outlives the process
   16 //! (`LoadMode::Shm`, see `tables::shm`).
   17

  183 /// What a table's trailer says about it.
  184 #[derive(Clone, Copy, Debug, PartialEq, Eq)]
  185 pub struct Trailer {
  186     pub k: u32,
  187     pub base: u8,
  188     /// Bytes of table data at the start of the file.
  189     pub data_bytes: u64,
  190     /// Number of real entries with each 2-bit code, if recorded.
  191     pub distribution: Option<[u64; 4]>,
  192     /// `checksum` of the data, if recorded.
  193     pub checksum: Option<u64>,
  194 }
  195 
  196 impl Trailer {
  197     /// The trailer as stored: text lines padded with NUL bytes (the text is
  198     /// far shorter than the trailer).
  199     #[must_use]
  200     pub fn to_bytes(&self) -> Vec<u8> {
  201         use std::fmt::Write as _;
  202         let mut text = String::with_capacity(TRAILER_LEN);
  203         text.push_str(MAGIC);
  204         let _ = writeln!(text, "layout {LAYOUT}");
  205         let _ = writeln!(text, "k {}", self.k);
  206         let _ = writeln!(text, "base {}", self.base);
  207         let _ = writeln!(text, "bytes {}", self.data_bytes);
  208         if let Some(d) = self.distribution {
  209             let _ = writeln!(text, "distribution {} {} {} {}", d[0], d[1], d[2], d[3]);
  210         }
  211         if let Some(c) = self.checksum {
  212             let _ = writeln!(text, "checksum {c:016x}");
  213         }
  214         let mut bytes = text.into_bytes();
  215         bytes.resize(TRAILER_LEN, 0);
  216         bytes
  217     }
  218 
  219     /// Parse a trailer. Returns `Ok(None)` if `bytes` does not start with the
  220     /// magic line (a file without a trailer).
  221     ///
  222     /// # Errors
  223     /// `MalformedTrailer` if the magic line is there but the rest is
  224     /// malformed.
  225     pub fn parse(bytes: &[u8]) -> Result<Option<Trailer>, TableError> {
  226         let Some(rest) = bytes
  227             .strip_prefix(MAGIC.as_bytes())
  228             .or_else(|| bytes.strip_prefix(LEGACY_MAGIC.as_bytes()))
  229         else {
  230             return Ok(None);
  231         };
  232         let end = rest.iter().position(|&b| b == 0).unwrap_or(rest.len());
  233         let invalid = |what: &str| TableError::MalformedTrailer(what.to_string());
  234         let text = std::str::from_utf8(&rest[..end]).map_err(|_| invalid("not text"))?;
  235         let (mut k, mut base, mut data_bytes) = (None, None, None);
  236         let (mut distribution, mut checksum) = (None, None);
  237         for line in text.lines() {
  238             let mut words = line.split_whitespace();
  239             let key = words.next().unwrap_or("");
  240             let values: Vec<&str> = words.collect();
  241             let number = |i: usize| -> Result<u64, TableError> {
  242                 values
  243                     .get(i)
  244                     .and_then(|v| v.parse().ok())
  245                     .ok_or_else(|| invalid(line))
  246             };
  247             match key {
  248                 "layout" if values != [LAYOUT] => {
  249                     return Err(invalid(&format!("unsupported layout {values:?}")));
  250                 }
  251                 "k" => k = Some(number(0)?),
  252                 "base" => base = Some(number(0)?),
  253                 "bytes" => data_bytes = Some(number(0)?),
  254                 "distribution" => {
  255                     distribution = Some([number(0)?, number(1)?, number(2)?, number(3)?]);
  256                 }
  257                 "checksum" => {
  258                     let v = values.first().ok_or_else(|| invalid(line))?;
  259                     checksum = Some(u64::from_str_radix(v, 16).map_err(|_| invalid(line))?);
  260                 }
  261                 // Unknown keys are allowed (later versions may add some).
  262                 _ => {}
  263             }
  264         }
  265         let (Some(k), Some(base), Some(data_bytes)) = (k, base, data_bytes) else {
  266             return Err(invalid("k, base or bytes missing"));
  267         };
  268         let k = u32::try_from(k).map_err(|_| invalid("k"))?;
  269         let base = u8::try_from(base).map_err(|_| invalid("base"))?;
  270         Ok(Some(Trailer {
  271             k,
  272             base,
  273             data_bytes,
  274             distribution,
  275             checksum,
  276         }))
  277     }
  278 }

  280 /// Whether `path` is on a hugetlbfs mount (whose files are whole huge pages
  281 /// and can only be written through a mapping).
  282 #[must_use]
  283 pub fn is_hugetlbfs(path: &str) -> bool {
  284     #[cfg(target_os = "linux")]
  285     {
  286         // The statfs magic of hugetlbfs.
  287         const HUGETLBFS_MAGIC: i64 = 0x9584_58f6;
  288         let Ok(c_path) = std::ffi::CString::new(path) else {
  289             return false;
  290         };
  291         // A missing file: judge by its folder (the file may be about to be
  292         // created there).
  293         let dir = std::path::Path::new(path)
  294             .parent()
  295             .and_then(|p| std::ffi::CString::new(p.as_os_str().as_encoded_bytes()).ok());
  296         for p in [Some(c_path), dir].into_iter().flatten() {
  297             // SAFETY: `p` is a NUL-terminated path, `st` a valid out-pointer.
  298             unsafe {
  299                 let mut st: libc::statfs = std::mem::zeroed();
  300                 if libc::statfs(p.as_ptr(), &raw mut st) == 0 {
  301                     #[allow(clippy::unnecessary_cast, reason = "f_type's type varies by target")]
  302                     return st.f_type as i64 == HUGETLBFS_MAGIC;
  303                 }
  304             }
  305         }
  306         false
  307     }
  308     #[cfg(not(target_os = "linux"))]
  309     {
  310         let _ = path;
  311         false
  312     }
  313 }

  326 /// Read the trailer of a table file, if it has one.
  327 ///
  328 /// # Errors
  329 /// If the file cannot be read, or has a malformed trailer.
  330 pub fn read_trailer(path: &str) -> Result<Option<Trailer>, TableError> {
  331     let read = || -> std::io::Result<Option<Vec<u8>>> {
  332         let f = std::fs::File::open(path)?;
  333         let len = f.metadata()?.len();
  334         if len < TRAILER_BYTES {
  335             return Ok(None);
  336         }
  337         let mut buf = vec![0u8; TRAILER_LEN];
  338         (&f).seek(SeekFrom::Start(len - TRAILER_BYTES))?;
  339         (&f).read_exact(&mut buf)?;
  340         Ok(Some(buf))
  341     };
  342     match read().map_err(io_error(path))? {
  343         Some(buf) => Trailer::parse(&buf),
  344         None => Ok(None),
  345     }
  346 }

  348 /// Write `trailer` into the last 4 KiB of an existing table file, then read
  349 /// it back.
  350 ///
  351 /// # Errors
  352 /// If the file is too small to hold the data and a trailer after it (the
  353 /// data region is never written), or on any I/O error, or if the trailer
  354 /// does not read back as written.
  355 pub fn write_trailer(path: &str, trailer: &Trailer) -> Result<(), TableError> {
  356     let f = std::fs::OpenOptions::new()
  357         .read(true)
  358         .write(true)
  359         .open(path)
  360         .map_err(io_error(path))?;
  361     let len = f.metadata().map_err(io_error(path))?.len();
  362     if len < trailer.data_bytes + TRAILER_BYTES {
  363         return Err(TableError::NoRoom {
  364             path: path.to_string(),
  365             bytes: len,
  366             data_bytes: trailer.data_bytes,
  367         });
  368     }
  369     let offset = len - TRAILER_BYTES;
  370     let bytes = trailer.to_bytes();
  371     #[cfg(target_os = "linux")]
  372     if is_hugetlbfs(path) {
  373         // hugetlbfs has no write(): map the last page and copy the trailer in.
  374         let page = HUGE_PAGE.min(len);
  375         let page_start = len - page;
  376         // (A page is at most 1 GiB, and the offset within it smaller: they
  377         // fit in `usize` on any target.)
  378         let as_usize = |x: u64| usize::try_from(x).map_err(std::io::Error::other);
  379         let as_offset = |x: u64| libc::off_t::try_from(x).map_err(std::io::Error::other);
  380         let (page_len, in_page, file_offset) = (|| {
  381             Ok::<_, std::io::Error>((
  382                 as_usize(page)?,
  383                 as_usize(offset - page_start)?,
  384                 as_offset(page_start)?,
  385             ))
  386         })()
  387         .map_err(io_error(path))?;
  388         // SAFETY: a fresh shared mapping of the file's last page; only the
  389         // trailer bytes (past the data, checked above) are written.
  390         unsafe {
  391             use std::os::unix::io::AsRawFd;
  392             let p = libc::mmap(
  393                 std::ptr::null_mut(),
  394                 page_len,
  395                 libc::PROT_READ | libc::PROT_WRITE,
  396                 libc::MAP_SHARED,
  397                 f.as_raw_fd(),
  398                 file_offset,
  399             );
  400             if p == libc::MAP_FAILED {
  401                 return Err(io_error(path)(std::io::Error::last_os_error()));
  402             }
  403             let dst = p.cast::<u8>().add(in_page);
  404             std::ptr::copy_nonoverlapping(bytes.as_ptr(), dst, bytes.len());
  405             libc::munmap(p, page_len);
  406         }
  407         return read_back(path, trailer);
  408     }
  409     let write = || -> std::io::Result<()> {
  410         (&f).seek(SeekFrom::Start(offset))?;
  411         (&f).write_all(&bytes)?;
  412         f.sync_data()
  413     };
  414     write().map_err(io_error(path))?;
  415     read_back(path, trailer)
  416 }

  418 /// Check that the trailer of `path` reads back as `trailer`.
  419 fn read_back(path: &str, trailer: &Trailer) -> Result<(), TableError> {

  421         Some(t) if t == *trailer => Ok(()),
  422         read => Err(TableError::ReadBack {

  430 /// Bytes per chunk of `checksum`.
  431 pub const CHECKSUM_CHUNK: usize = 1 << 20;

  433 fn mix(h: u64, x: u64) -> u64 {
  434     (h ^ x).wrapping_mul(0x9E37_79B9_7F4A_7C15).rotate_left(29)
  435 }

  437 /// Hash of each `CHECKSUM_CHUNK` bytes of `data` (the last chunk may be
  438 /// shorter), in parallel.
  439 ///
  440 /// # Errors
  441 /// If a worker thread panicked.
  442 pub fn chunk_hashes(
  443     data: &[u8],
  444     threads: usize,
  445 ) -> Result<Vec<u64>, crate::parallel::WorkerPanicked> {
  446     fn chunk_hash(chunk: &[u8]) -> u64 {
  447         // Four independent lanes (no single dependency chain), then folded.
  448         let mut lanes = [1u64, 2, 3, 4];
  449         let (words, rest) = chunk.as_chunks::<32>();
  450         for w in words {
  451             for (lane, x) in lanes.iter_mut().zip(w.as_chunks::<8>().0) {
  452                 *lane = mix(*lane, u64::from_le_bytes(*x));
  453             }
  454         }
  455         let mut h = lanes.iter().fold(chunk.len() as u64, |h, &l| mix(h, l));
  456         for &b in rest {
  457             h = mix(h, u64::from(b));
  458         }
  459         h
  460     }
  461     let n = data.len().div_ceil(CHECKSUM_CHUNK);
  462     let threads = threads.clamp(1, n.max(1));
  463     let per = n.div_ceil(threads);
  464     Ok(crate::parallel::on_threads(threads, |t| {
  465         data.chunks(CHECKSUM_CHUNK)
  466             .skip(t * per)
  467             .take(per)
  468             .map(chunk_hash)
  469             .collect::<Vec<u64>>()
  470     })?
  471     .concat())
  472 }

  474 /// The checksum of `len` bytes of data from its `chunk_hashes` (so that a
  475 /// table generated in parts aligned to chunks needs no second pass).
  476 #[must_use]
  477 pub fn combine_hashes(len: u64, hashes: &[u64]) -> u64 {
  478     hashes.iter().fold(len, |h, &x| mix(h, x))
  479 }

  481 /// Checksum of table data: a hash of each 1 MiB chunk (in parallel), then of
  482 /// the chunk hashes in order. Detects corruption (bit flips, truncation),
  483 /// not tampering.
  484 ///
  485 /// # Errors
  486 /// If a worker thread panicked.
  487 pub fn checksum(data: &[u8], threads: usize) -> Result<u64, crate::parallel::WorkerPanicked> {
  488     Ok(combine_hashes(
  489         data.len() as u64,
  490         &chunk_hashes(data, threads)?,
  491     ))
  492 }

  494 /// How `H48Table::open` brings a table file into memory.

  496 pub enum LoadMode {
  497     /// The table's shared memory segment if there is one that holds it
  498     /// (`Shm`, Linux); else `Map` for files on hugetlbfs or `/dev/shm`
  499     /// (already in memory) and on other unix systems; `Copy` for other files
  500     /// on Linux (anonymous memory gets transparent huge pages reliably, the
  501     /// page cache may not) and where there are no file mappings (WASI,

  505     Map,
  506     /// Copy the file into anonymous memory (transparent huge pages where

  508     Copy,
  509     /// Attach the table's shared memory segment, made from the file first if
  510     /// there is none (Linux; see `tables::shm`). `Auto` also uses a segment
  511     /// that is there, if it holds this file's table.
  512     Shm,
  513 }

  540 impl LoadOptions {
  541     /// Options from the environment: `EGG_SOLVER_LOAD` = `auto` | `map` |
  542     /// `copy` | `shm`, `EGG_SOLVER_PREFAULT` = `1` | `0`, `EGG_SOLVER_SHM_PAGES`
  543     /// (`ShmPages::parse`) and `EGG_SOLVER_SHM_KEY` (`shm::parse_key`).
  544     ///
  545     /// # Errors
  546     /// If a variable is set to another value.
  547     pub fn from_env() -> Result<LoadOptions, TableError> {
  548         let mut o = LoadOptions::default();
  549         let bad = |var, value: String, expected| TableError::BadEnv {
  550             var,
  551             value,
  552             expected,
  553         };
  554         if let Ok(v) = crate::env::var("EGG_SOLVER_LOAD") {
  555             o.mode = match v.as_str() {
  556                 "auto" | "" => LoadMode::Auto,
  557                 "map" => LoadMode::Map,
  558                 "copy" => LoadMode::Copy,
  559                 "shm" => LoadMode::Shm,
  560                 _ => return Err(bad("EGG_SOLVER_LOAD", v, "auto, map, copy or shm")),
  561             };
  562         }
  563         if let Ok(v) = crate::env::var("EGG_SOLVER_SHM_PAGES") {
  564             o.shm_pages = match ShmPages::parse(&v) {
  565                 Some(pages) => pages,
  566                 None => return Err(bad("EGG_SOLVER_SHM_PAGES", v, "e.g. 1g,2m,4k")),
  567             };
  568         }
  569         if let Ok(v) = crate::env::var("EGG_SOLVER_SHM_KEY") {
  570             o.shm_key = match super::shm::parse_key(&v) {
  571                 Some(key) => Some(key),
  572                 None => return Err(bad("EGG_SOLVER_SHM_KEY", v, "e.g. 0x636f0709")),
  573             };
  574         }
  575         if let Ok(v) = crate::env::var("EGG_SOLVER_PREFAULT") {
  576             o.prefault = match v.as_str() {
  577                 "1" | "" => true,
  578                 "0" => false,
  579                 _ => return Err(bad("EGG_SOLVER_PREFAULT", v, "0 or 1")),
  580             };
  581         }
  582         Ok(o)
  583     }

  585     /// The mode used for the file at `path` (resolves `Auto`, unless a
  586     /// segment of the table is there: see `H48Table::open_reporting`).
  587     #[must_use]
  588     pub fn mode_for(&self, path: &str) -> LoadMode {
  589         match self.mode {
  590             LoadMode::Auto => {
  591                 let in_memory = path.starts_with("/dev/shm") || is_hugetlbfs(path);
  592                 if cfg!(unix) && (!cfg!(target_os = "linux") || in_memory) {
  593                     LoadMode::Map
  594                 } else {
  595                     LoadMode::Copy
  596                 }
  597             }
  598             m => m,
  599         }
  600     }
  601 }

  603 /// Where `bench tune` saves a table's best lane counts.
  604 fn tune_path(table: &str) -> String {
  605     format!("{table}.tune")
  606 }
  607 
  608 /// Save the best lane count per thread count (`threads lanes` lines).
  609 ///
  610 /// # Errors
  611 /// If the file cannot be written (e.g. the table's folder is read-only).
  612 pub fn save_tuned_lanes(table: &str, best: &[(usize, usize)]) -> std::io::Result<String> {
  613     use std::fmt::Write as _;
  614     let mut text = String::new();
  615     for (threads, lanes) in best {
  616         let _ = writeln!(text, "{threads} {lanes}");
  617     }
  618     let path = tune_path(table);
  619     std::fs::write(&path, text)?;
  620     Ok(path)
  621 }

  623 /// Lanes per thread for a table of `k`, when not tuned (`bench tune`).
  624 /// Smaller tables gain from more lanes (measured, solves/s, 24 -> 40 lanes):
  625 /// h7 +10% at 1T (2.017 -> 2.217) and at 8 threads (15.43 -> 17.07), also
  626 /// on 1 GiB pages (2.017 -> 2.283 at 1T: not a TLB effect); h8 equal at 1T,
  627 /// +8.9% at 8 threads (26.73 -> 29.10); h6 equal (+4.5% at 1T, -1 to -5%
  628 /// at 8 threads). h11 keeps `lanes::DEFAULT_LANES`, 24 (tuned at 1 and 48
  629 /// threads; 40 is 3.4% slower at 8 threads, 197.9 -> 191.2). h9 and h10
  630 /// (no RAM for them here) get 32, in between.
  631 #[must_use]
  632 pub fn untuned_lanes(k: u32) -> usize {
  633     match k {
  634         0..=8 => 40,
  635         9 | 10 => 32,
  636         _ => crate::lanes::DEFAULT_LANES,
  637     }
  638 }
  639 
  640 /// The default lanes per thread for `threads` threads on the table at
  641 /// `path`: from `<path>.tune` (`bench tune`; the entry of the nearest
  642 /// thread count), else `untuned_lanes(k)`.
  643 pub fn default_lanes(path: &str, k: u32) -> impl Fn(usize) -> usize {
  644     let tuned: Vec<(usize, usize)> = std::fs::read_to_string(tune_path(path))
  645         .unwrap_or_default()
  646         .lines()
  647         .filter_map(|l| {
  648             let mut w = l.split_whitespace().map(|x| x.parse::<usize>().ok());
  649             Some((w.next()??, w.next()??))
  650         })
  651         .filter(|&(threads, lanes)| threads > 0 && lanes > 0)
  652         .collect();
  653     move |threads| {
  654         tuned
  655             .iter()
  656             .min_by_key(|&&(t, _)| t.abs_diff(threads))
  657             .map_or(untuned_lanes(k), |&(_, lanes)| lanes)
  658     }
  659 }

  687             data_bytes: 3_891_363_840,

  693         assert_eq!(Trailer::parse(&bytes).unwrap(), Some(t));
  694         let bare = Trailer {
  695             distribution: None,
  696             checksum: None,
  697             ..t
  698         };
  699         assert_eq!(Trailer::parse(&bare.to_bytes()).unwrap(), Some(bare));
  700         // No magic: no trailer. Magic but garbage: an error.
  701         assert_eq!(Trailer::parse(&[0u8; 64]).unwrap(), None);
  702         assert!(Trailer::parse(format!("{MAGIC}k 7\n").as_bytes()).is_err());
  703         assert!(Trailer::parse(format!("{MAGIC}k x\nbase 9\nbytes 1\n").as_bytes()).is_err());
  704         // Another index layout is refused, unknown keys are not.
  705         let other = format!("{MAGIC}layout h48-parity\nk 7\nbase 9\nbytes 1\n");
  706         assert!(Trailer::parse(other.as_bytes()).is_err());
  707         let extra = format!("{MAGIC}k 7\nbase 9\nbytes 1\nmade-by gen 2\n");
  708         assert_eq!(
  709             Trailer::parse(extra.as_bytes()).unwrap().map(|t| t.k),
  710             Some(7)
  711         );
  712     }
