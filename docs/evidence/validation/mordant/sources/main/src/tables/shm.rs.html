    1 //! System V shared memory segments for tables (`LoadMode::Shm`, `--shm`,
    2 //! as vcube's `--shm`): the first run copies a table file into a segment,
    3 //! which stays in memory after the process exits (until reboot or `ipcrm`);
    4 //! later runs attach it read-only and start at once.
    5 //!
    6 //! A segment holds the table file's bytes as a hugetlbfs file does: the data
    7 //! at offset 0 (so the search sees the same alignment), and the file's
    8 //! trailer in the segment's last 4 KiB, with `segment-*` lines added. The
    9 //! trailer is written last, after the data is in and its checksum checked:
   10 //! a segment without one is still loading, or its load was interrupted. The
   11 //! key is legacy ASCII "co" (retained for compatibility), then k, then base (`key`: 0x636f0b0a for h11
   12 //! base 10), so every run of a table finds the same segment; every attach
   13 //! compares the segment's trailer with the table file's (k, base, size,
   14 //! distribution, checksum), so a segment of another table under that key is
   15 //! never used.
   16 //!
   17 //! A new segment gets 1 GiB pages, else 2 MiB, else 4 KiB pages (as
   18 //! `ShmPages` allows). Huge pages come from the kernel's hugetlb pool and
   19 //! need free pages there and the permission (`vm.hugetlb_shm_group`); a size
   20 //! that cannot be had is skipped with a note that says how to get it.
   21 //! Linux only: elsewhere `LoadMode::Shm` fails and `Auto` skips segments.
   22 //!
   23 //! Measured on h7 (3.62 GiB, the 48-core server): a later run attaches the
   24 //! segment in 0.00 s on 1 GiB pages (0.6 s on 4 KiB pages, mapping 0.95
   25 //! million pages; 0.00 s with `--prefault 0`), where copying the file from
   26 //! a ~140 MB/s disk takes 28 s; making it from a file in memory takes 0.10 s
   27 //! on 1 GiB pages and 1.9 s on 4 KiB pages, its checksum check included. The
   28 //! search on a segment is as fast as on the same pages otherwise (1T, 6
   29 //! pairs each: equal to a hugetlbfs map on 1 GiB pages and to a `/dev/shm`
   30 //! map on 4 KiB pages); 4 KiB pages are 27% slower at 1T and 32% at 48T
   31 //! than 1 GiB pages.

   36 /// The first two bytes of every key: legacy ASCII "co" (retained for compatibility).
   37 pub const KEY_PREFIX: u32 = 0x636f_0000;

   39 /// The segment key of table hk with `base`: `KEY_PREFIX`, then k and base,
   40 /// a byte each (`ipcs -m` shows 0x636f0709 for h7 base 9).
   41 #[must_use]
   42 pub fn key(k: u32, base: u8) -> u32 {
   43     KEY_PREFIX | (k & 0xFF) << 8 | u32::from(base)
   44 }

   46 /// A key as `ipcs` shows it (`0x636f0709`) or in decimal; not 0
   47 /// (`IPC_PRIVATE`, which makes a new segment every time).
   48 #[must_use]
   49 pub fn parse_key(text: &str) -> Option<u32> {
   50     let key = match text.strip_prefix("0x").or_else(|| text.strip_prefix("0X")) {
   51         Some(hex) => u32::from_str_radix(hex, 16).ok()?,
   52         None => text.parse().ok()?,
   53     };
   54     (key != 0).then_some(key)
   55 }

   57 /// The page sizes a new segment may get, tried largest first: 1 GiB and 2
   58 /// MiB pages from the kernel's hugetlb pool, and normal (4 KiB) pages.
   59 #[derive(Clone, Copy, Debug, PartialEq, Eq)]
   60 pub struct ShmPages(u8);
   61 
   62 /// Normal pages (the `4k` of `ShmPages`).
   63 pub const NORMAL_PAGE: u64 = 1 << 12;
   64 
   65 impl ShmPages {
   66     /// All three sizes.
   67     pub const ALL: ShmPages = ShmPages(0b111);
   68     /// Each size, largest first, with its name in `parse`.
   69     const SIZES: [(u64, &'static str); 3] = [(1 << 30, "1g"), (1 << 21, "2m"), (NORMAL_PAGE, "4k")];
   70 
   71     /// The allowed page sizes in bytes, largest first.
   72     pub fn sizes(self) -> impl Iterator<Item = u64> {
   73         (0..Self::SIZES.len())
   74             .filter(move |&i| self.0 & (1 << i) != 0)
   75             .map(|i| Self::SIZES[i].0)
   76     }
   77 
   78     /// A comma-separated list of `1g`, `2m` and `4k` (any order, at least
   79     /// one), or `all`.
   80     #[must_use]
   81     pub fn parse(text: &str) -> Option<ShmPages> {
   82         if text.eq_ignore_ascii_case("all") {
   83             return Some(ShmPages::ALL);
   84         }
   85         let mut bits = 0;
   86         for word in text.split(',') {
   87             let word = word.trim();
   88             let i = Self::SIZES
   89                 .iter()
   90                 .position(|(_, name)| word.eq_ignore_ascii_case(name))?;
   91             bits |= 1 << i;
   92         }
   93         (bits != 0).then_some(ShmPages(bits))
   94     }
   95 }
   96 
   97 impl Default for ShmPages {
   98     fn default() -> ShmPages {
   99         ShmPages::ALL
  100     }
  101 }

  253     /// The last 4 KiB of a segment: the table file's trailer, then the
  254     /// segment's page size and source file (keys `Trailer::parse` ignores).
  255     pub(super) fn header_bytes(trailer: &Trailer, page: u64, source: &str) -> Vec<u8> {
  256         let mut bytes = trailer.to_bytes();
  257         let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
  258         // (One line of at most 2000 bytes: the trailer text is far shorter
  259         // than its 4 KiB.)
  260         let source = source.replace(char::is_control, "?");
  261         let source = &source[..source.floor_char_boundary(2000)];
  262         let extra = format!("{PAGE_KEY} {page}\n{SOURCE_KEY} {source}\n");
  263         bytes[end..end + extra.len()].copy_from_slice(extra.as_bytes());
  264         bytes
  265     }

  267     /// The `Header` in a segment's last 4 KiB, or `None` if there is no
  268     /// trailer there.
  269     pub(super) fn parse_header(bytes: &[u8]) -> Result<Option<Header>, TableError> {
  270         let Some(trailer) = Trailer::parse(bytes)? else {
  271             return Ok(None);
  272         };
  273         let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
  274         let text = String::from_utf8_lossy(&bytes[..end]);
  275         let value = |key: &str| {
  276             text.lines()
  277                 .find_map(|line| line.strip_prefix(key)?.strip_prefix(' '))
  278                 .map(str::to_string)
  279         };
  280         Ok(Some(Header {
  281             trailer,
  282             page_bytes: value(PAGE_KEY).and_then(|v| v.parse().ok()),
  283             source: value(SOURCE_KEY).unwrap_or_else(|| "?".to_string()),
  284         }))
  285     }

  312     pub(super) fn shmget(key: u32, bytes: u64, flags: i32) -> io::Result<i32> {
  313         let bytes = usize::try_from(bytes).map_err(io::Error::other)?;
  314         // SAFETY: no pointers.
  315         let id = unsafe { libc::shmget(key.cast_signed(), bytes, flags) };
  316         if id < 0 {
  317             Err(io::Error::last_os_error())
  318         } else {
  319             Ok(id)
  320         }
  321     }

  323     pub(super) fn stat(id: i32) -> io::Result<libc::shmid_ds> {
  324         // SAFETY: a valid out-pointer.
  325         unsafe {
  326             let mut ds: libc::shmid_ds = std::mem::zeroed();
  327             if libc::shmctl(id, libc::IPC_STAT, &raw mut ds) != 0 {
  328                 return Err(io::Error::last_os_error());
  329             }
  330             Ok(ds)
  331         }

  350     /// An attachment of a segment, detached on drop (unless made `Memory`).
  351     struct Attachment {
  352         ptr: *mut u8,
  353         ds: libc::shmid_ds,
  354     }
  355 
  356     impl Attachment {
  357         /// Attach segment `id`, read-only or writable (`ds` is its state
  358         /// before: `shm_nattch` does not count this attachment).
  359         fn new(id: i32, read_only: bool) -> io::Result<Attachment> {
  360             let ds = stat(id)?;
  361             let flags = if read_only { libc::SHM_RDONLY } else { 0 };
  362             // SAFETY: the kernel picks the address.
  363             let ptr = unsafe { libc::shmat(id, std::ptr::null(), flags) };
  364             if ptr.addr() == usize::MAX {
  365                 return Err(io::Error::last_os_error());
  366             }
  367             Ok(Attachment {
  368                 ptr: ptr.cast(),
  369                 ds,
  370             })
  371         }
  372 
  373         fn bytes(&self) -> u64 {
  374             self.ds.shm_segsz as u64
  375         }
  376 
  377         /// Its last 4 KiB (the trailer, once loaded).
  378         fn last_page(&self) -> Option<Vec<u8>> {
  379             let page = usize::try_from(file::TRAILER_BYTES).ok()?;
  380             let offset = usize::try_from(self.bytes()).ok()?.checked_sub(page)?;
  381             let mut bytes = vec![0u8; page];
  382             // SAFETY: the last 4 KiB of the attached segment. Its first byte
  383             // is written last, with Release (`write_trailer`): a reader that
  384             // sees it sees the rest, and the data.
  385             unsafe {
  386                 let start = self.ptr.add(offset);
  387                 bytes[0] = AtomicU8::from_ptr(start).load(Ordering::Acquire);
  388                 std::ptr::copy_nonoverlapping(start.add(1), bytes.as_mut_ptr().add(1), page - 1);
  389             }
  390             Some(bytes)
  391         }
  392 
  393         /// The attachment as `Memory`, unmapped on drop (`munmap` detaches a
  394         /// segment as `shmdt` does; the length is the mapping's, as segments
  395         /// are whole pages).
  396         fn into_memory(self) -> io::Result<Memory> {
  397             let len = usize::try_from(self.bytes()).map_err(io::Error::other)?;
  398             let memory = Memory { ptr: self.ptr, len };
  399             std::mem::forget(self);
  400             Ok(memory)
  401         }
  402     }

  404     impl Drop for Attachment {
  405         fn drop(&mut self) {
  406             // SAFETY: our attachment, detached once.
  407             unsafe {
  408                 libc::shmdt(self.ptr.cast());
  409             }
  410         }

  413     /// Write `trailer` into the last 4 KiB of a writable segment, its first
  414     /// byte last (with Release: a reader that sees it, `Attachment::last_page`,
  415     /// sees the rest and the data).
  416     fn write_trailer(segment: &Memory, trailer: &[u8]) {
  417         let offset = segment.len() - trailer.len();
  418         // SAFETY: the last 4 KiB of our writable attachment.
  419         unsafe {
  420             let start = segment.ptr().add(offset);
  421             std::ptr::copy_nonoverlapping(trailer.as_ptr().add(1), start.add(1), trailer.len() - 1);
  422             AtomicU8::from_ptr(start).store(trailer[0], Ordering::Release);
  423         }
  424     }

  426     /// What a segment holds, for the table of `trailer`.
  427     enum Found {
  428         /// This table, loaded.
  429         Same(Header),
  430         /// No trailer yet: being loaded by its maker, which is running.
  431         Loading(i32),
  432         /// No trailer, and its maker is gone: the load was interrupted.
  433         Interrupted(i32),
  434         /// Another table, or no table this code reads.
  435         Other(String),
  436     }

  438     fn examine(attachment: &Attachment, trailer: &Trailer) -> Found {
  439         let bytes = attachment.bytes();
  440         let creator = attachment.ds.shm_cpid;
  441         match attachment.last_page().map(|page| parse_header(&page)) {
  442             None => Found::Other(format!("a segment of {bytes} bytes, too small for a table")),
  443             Some(Err(error)) => {
  444                 Found::Other(format!("a segment with an unreadable trailer ({error})"))
  445             }
  446             Some(Ok(None)) if alive(creator) => Found::Loading(creator),
  447             Some(Ok(None)) => Found::Interrupted(creator),
  448             Some(Ok(Some(header))) => {
  449                 let whole = |page: u64| {
  450                     page > 0
  451                         && bytes.is_multiple_of(page)
  452                         && bytes >= header.trailer.data_bytes + file::TRAILER_BYTES
  453                 };
  454                 if header.trailer == *trailer && header.page_bytes.is_some_and(whole) {
  455                     Found::Same(header)
  456                 } else {
  457                     Found::Other(format!(
  458                         "{} (loaded from {})",
  459                         describe(&header.trailer),
  460                         header.source
  461                     ))
  462                 }
  463             }
  464         }
  465     }

  467     pub(super) fn open(
  468         path: &str,
  469         options: &LoadOptions,
  470         create: bool,
  471         notes: &mut Vec<String>,
  472     ) -> Result<Option<(Memory, Segment)>, TableError> {
  473         let Some(trailer) = file::read_trailer(path)?.filter(|t| t.checksum.is_some()) else {
  474             return if create {
  475                 Err(TableError::ShmNoChecksum {
  476                     path: path.to_string(),
  477                 })
  478             } else {
  479                 Ok(None)
  480             };
  481         };
  482         let key = options
  483             .shm_key
  484             .unwrap_or_else(|| key(trailer.k, trailer.base));
  485         // (A few rounds: other processes may make or remove it meanwhile.)
  486         for _ in 0..4 {
  487             let id = match shmget(key, 0, 0) {
  488                 Ok(id) => id,
  489                 Err(e) if is(&e, libc::ENOENT) && create => {
  490                     match make(key, path, &trailer, options, notes)? {
  491                         Some(made) => return Ok(Some(made)),
  492                         None => continue,
  493                     }
  494                 }
  495                 Err(e) if is(&e, libc::ENOENT) => return Ok(None),
  496                 Err(e) => return Err(call_error(key, "shmget")(e)),
  497             };
  498             let attachment = match Attachment::new(id, true) {
  499                 Ok(attachment) => attachment,
  500                 // (Removed meanwhile.)
  501                 Err(e) if is(&e, libc::EIDRM) || is(&e, libc::EINVAL) => continue,
  502                 Err(e) => return Err(call_error(key, "shmat")(e)),
  503             };
  504             match examine(&attachment, &trailer) {
  505                 Found::Same(header) => {
  506                     let segment = Segment {
  507                         key,
  508                         id,
  509                         bytes: attachment.bytes(),
  510                         page_bytes: header.page_bytes.unwrap_or(NORMAL_PAGE),
  511                         created: false,
  512                         source: header.source,
  513                         creator: attachment.ds.shm_cpid,
  514                     };
  515                     let memory = attachment.into_memory().map_err(call_error(key, "shmat"))?;
  516                     if options.prefault {
  517                         memory.prefault().map_err(call_error(key, "prefault"))?;
  518                     }
  519                     return Ok(Some((memory, segment)));
  520                 }
  521                 Found::Loading(pid) => return Err(TableError::ShmLoading { key, pid }),
  522                 // (Nobody else has it attached, and its maker is gone: no
  523                 // process can be filling it.)
  524                 Found::Interrupted(pid) if create && attachment.ds.shm_nattch == 0 => {
  525                     drop(attachment);
  526                     mark_removed(id).map_err(call_error(key, "shmctl(IPC_RMID)"))?;
  527                     notes.push(format!(
  528                         "removed segment {key:#010x}: its load by process {pid} was interrupted"
  529                     ));
  530                 }
  531                 Found::Interrupted(pid) => {
  532                     return Err(TableError::Shm {
  533                         key,
  534                         message: format!(
  535                             "has no trailer: its load by process {pid} was interrupted; \
  536                              remove it ({})",
  537                             remove_hint(path, key)
  538                         ),
  539                     });
  540                 }
  541                 Found::Other(held) => {
  542                     return Err(TableError::ShmOtherTable {
  543                         key,
  544                         path: path.to_string(),
  545                         held,
  546                         wanted: describe(&trailer),
  547                         hint: remove_hint(path, key),
  548                     });
  549                 }
  550             }
  551         }
  552         Err(TableError::Shm {
  553             key,
  554             message: "made and removed by other processes again and again; try once more".into(),
  555         })
  556     }

  558     pub(super) fn loaded(path: &str, options: &LoadOptions) -> bool {
  559         let Ok(Some(trailer)) = file::read_trailer(path) else {
  560             return false;
  561         };
  562         let key = options
  563             .shm_key
  564             .unwrap_or_else(|| key(trailer.k, trailer.base));
  565         trailer.checksum.is_some()
  566             && shmget(key, 0, 0)
  567                 .and_then(|id| Attachment::new(id, true))
  568                 .is_ok_and(|attachment| matches!(examine(&attachment, &trailer), Found::Same(_)))
  569     }

  596     /// Make segment `key` from the table file: on the first page size that
  597     /// can be had, the data copied in and its checksum checked, then the
  598     /// trailer. Returns it attached read-only, or `None` if another process
  599     /// made it first.
  600     fn make(
  601         key: u32,
  602         path: &str,
  603         trailer: &Trailer,
  604         options: &LoadOptions,
  605         notes: &mut Vec<String>,
  606     ) -> Result<Option<(Memory, Segment)>, TableError> {
  607         let mut made = None;
  608         for page in options.shm_pages.sizes() {
  609             let bytes = segment_bytes(trailer.data_bytes, page);
  610             let huge = huge_page(page);
  611             // Preflight: fail (or skip a page size) now, not deep into the copy.
  612             if let Err(why) = preflight(page, bytes) {
  613                 if huge {
  614                     notes.push(why);
  615                     continue;
  616                 }
  617                 return Err(TableError::Shm { key, message: why });
  618             }
  619             let page_flags = if huge {
  620                 libc::SHM_HUGETLB | (page.trailing_zeros() << SHM_HUGE_SHIFT).cast_signed()
  621             } else {
  622                 0
  623             };
  624             match shmget(
  625                 key,
  626                 bytes,
  627                 libc::IPC_CREAT | libc::IPC_EXCL | 0o600 | page_flags,
  628             ) {
  629                 Ok(id) => {
  630                     made = Some((id, page, bytes));
  631                     break;
  632                 }
  633                 Err(e) if is(&e, libc::EEXIST) => return Ok(None),
  634                 Err(e) if huge => notes.push(huge_page_failure(page, &e)),
  635                 Err(e) => return Err(call_error(key, "shmget")(e)),
  636             }
  637         }
  638         let Some((id, page, bytes)) = made else {
  639             return Err(TableError::Shm {
  640                 key,
  641                 message: format!("no allowed page size can be had: {}", notes.join("; ")),
  642             });
  643         };
  644         let guard = RemoveOnDrop(Some(id));
  645         let writable = Attachment::new(id, false)
  646             .map_err(call_error(key, "shmat"))?
  647             .into_memory()
  648             .map_err(call_error(key, "shmat"))?;
  649         if !huge_page(page) {
  650             // Transparent huge pages, where shared memory gets them
  651             // (/sys/kernel/mm/transparent_hugepage/shmem_enabled: advise).
  652             // SAFETY: advice on our own attachment.
  653             unsafe {
  654                 libc::madvise(writable.ptr().cast(), writable.len(), libc::MADV_HUGEPAGE);
  655             }
  656         }
  657         let table_file = std::fs::File::open(path).map_err(file::io_error(path))?;
  658         let data_len =
  659             usize::try_from(trailer.data_bytes).map_err(|_| TableError::TooLarge(trailer.k))?;
  660         super::super::read_parallel(&table_file, &writable, data_len)
  661             .map_err(file::io_error(path))?;
  662         // The copy is the file's table, and the file is not damaged.
  663         let data = (writable.bytes()?.get(..data_len)).ok_or(TableError::TooLarge(trailer.k))?;
  664         let got = file::checksum(data, crate::thread_count())?;
  665         let want = trailer.checksum.unwrap_or_default();
  666         if got != want {
  667             return Err(TableError::Checksum { got, want });
  668         }
  669         write_trailer(&writable, &header_bytes(trailer, page, path));
  670         // Read-only from now on (`ipcs -m` shows perms 400 once loaded, 600
  671         // while loading).
  672         let mut ds = stat(id).map_err(call_error(key, "shmctl(IPC_STAT)"))?;
  673         ds.shm_perm.mode = 0o400;
  674         // SAFETY: a valid buffer.
  675         if unsafe { libc::shmctl(id, libc::IPC_SET, &raw mut ds) } != 0 {
  676             return Err(call_error(key, "shmctl(IPC_SET)")(
  677                 io::Error::last_os_error(),
  678             ));
  679         }
  680         drop(writable);
  681         let read_only = Attachment::new(id, true).map_err(call_error(key, "shmat"))?;
  682         guard.disarm();
  683         let segment = Segment {
  684             key,
  685             id,
  686             bytes,
  687             page_bytes: page,
  688             created: true,
  689             source: path.to_string(),
  690             creator: std::process::id().cast_signed(),
  691         };
  692         let memory = read_only.into_memory().map_err(call_error(key, "shmat"))?;
  693         if options.prefault {
  694             memory.prefault().map_err(call_error(key, "prefault"))?;
  695         }
  696         Ok(Some((memory, segment)))
  697     }

  708     /// Room for a segment of `bytes` on pages of `page` bytes: the kernel's
  709     /// segment limits, then for huge pages the permission and free pages in
  710     /// the hugetlb pool. (Free memory for 4 KiB pages is the caller's check:
  711     /// `cli::preflight`.)
  712     fn preflight(page: u64, bytes: u64) -> Result<(), String> {
  713         kernel_limits(bytes)?;
  714         if !huge_page(page) {
  715             return Ok(());
  716         }
  717         // The permission first, as the kernel checks it first.
  718         if let Some(permission) = no_permission() {
  719             return Err(format!("{} pages: {permission}", page_name(page)));
  720         }
  721         pool_room(page, bytes)
  722     }

  724     /// `kernel.shmmax` (bytes per segment), `kernel.shmall` (pages in all
  725     /// segments) and `kernel.shmmni` (segments).
  726     fn kernel_limits(bytes: u64) -> Result<(), String> {
  727         if let Some(max) = read_number("/proc/sys/kernel/shmmax").filter(|&max| bytes > max) {
  728             return Err(format!(
  729                 "kernel.shmmax is {max} bytes, below the segment's {bytes}: raise it with \
  730                  `sudo sysctl kernel.shmmax={bytes}`"
  731             ));
  732         }
  733         // SAFETY: no pointers.
  734         let system_page =
  735             u64::try_from(unsafe { libc::sysconf(libc::_SC_PAGESIZE) }).unwrap_or(4096);
  736         let (count, used) = segments_in_use(system_page);
  737         let pages = bytes.div_ceil(system_page);
  738         if let Some(all) = read_number("/proc/sys/kernel/shmall").filter(|&all| used + pages > all)
  739         {
  740             return Err(format!(
  741                 "kernel.shmall is {all} pages of {} KiB, other segments use {used} and this one \
  742                  needs {pages}: raise it with `sudo sysctl kernel.shmall={}`",
  743                 system_page >> 10,
  744                 used + pages
  745             ));
  746         }
  747         if let Some(most) = read_number("/proc/sys/kernel/shmmni").filter(|&most| count >= most) {
  748             return Err(format!(
  749                 "kernel.shmmni: all {most} segments are in use (`ipcs -m`)"
  750             ));
  751         }
  752         Ok(())
  753     }
  754 
  755     /// Segments and the system pages they take, from /proc/sysvipc/shm.
  756     fn segments_in_use(system_page: u64) -> (u64, u64) {
  757         let text = std::fs::read_to_string("/proc/sysvipc/shm").unwrap_or_default();
  758         let sizes = text
  759             .lines()
  760             .skip(1)
  761             .filter_map(|line| line.split_whitespace().nth(3)?.parse::<u64>().ok());
  762         sizes.fold((0, 0), |(count, pages), size| {
  763             (count + 1, pages + size.div_ceil(system_page))
  764         })
  765     }

  767     /// Free pages of `page` bytes in the hugetlb pool (free less reserved,
  768     /// plus what overcommit allows) for a segment of `bytes`.
  769     fn pool_room(page: u64, bytes: u64) -> Result<(), String> {
  770         let pool = format!("/sys/kernel/mm/hugepages/hugepages-{}kB", page >> 10);
  771         let read = |what: &str| read_number(&format!("{pool}/{what}"));
  772         let Some(total) = read("nr_hugepages") else {
  773             return Err(format!(
  774                 "{} pages: this kernel has none ({pool} is missing)",
  775                 page_name(page)
  776             ));
  777         };
  778         let unused = |have: Option<u64>, taken: Option<u64>| {
  779             have.unwrap_or(0).saturating_sub(taken.unwrap_or(0))
  780         };
  781         let free = unused(read("free_hugepages"), read("resv_hugepages"))
  782             + unused(read("nr_overcommit_hugepages"), read("surplus_hugepages"));
  783         pool_verdict(page, bytes, total, free)
  784     }
  785 
  786     /// Whether `free` pages of the pool of `total` pages of `page` bytes
  787     /// hold a segment of `bytes`; if not, how to add them.
  788     pub(super) fn pool_verdict(page: u64, bytes: u64, total: u64, free: u64) -> Result<(), String> {
  789         let needed = bytes / page;
  790         if needed <= free {
  791             return Ok(());
  792         }
  793         Err(format!(
  794             "{} pages: {needed} needed, {free} free in the pool ({total} in all); to add them: \
  795              `echo {} | sudo tee /sys/kernel/mm/hugepages/hugepages-{}kB/nr_hugepages`",
  796             page_name(page),
  797             total + needed - free,
  798             page >> 10
  799         ))
  800     }

  802     /// Why `shmget` of `bytes` on huge pages of `page` bytes failed, and how
  803     /// to fix it.
  804     pub(super) fn huge_page_failure(page: u64, error: &io::Error) -> String {
  805         let name = page_name(page);
  806         match error.raw_os_error() {
  807             Some(libc::EPERM) => {
  808                 let group = read_number("/proc/sys/vm/hugetlb_shm_group").unwrap_or(0);
  809                 // SAFETY: no pointers.
  810                 let gid = unsafe { libc::getegid() };
  811                 format!("{name} pages: EPERM: {}", permission_message(group, gid))
  812             }
  813             Some(libc::ENOMEM) => format!(
  814                 "{name} pages: the pool has too few free pages (ENOMEM; see \
  815                  /sys/kernel/mm/hugepages/hugepages-{}kB)",
  816                 page >> 10
  817             ),
  818             _ => format!("{name} pages: shmget: {error}"),
  819         }
  820     }

  822     /// Why this process may not make huge-page segments, if it may not
  823     /// (the kernel allows them to the group `vm.hugetlb_shm_group` and to
  824     /// `CAP_IPC_LOCK`).
  825     fn no_permission() -> Option<String> {
  826         // (`EGG_SOLVER_SHM_DENY_HUGE`, for tests, undocumented: as if it were
  827         // missing, without asking the kernel.)
  828         let simulated = crate::env::var_os("EGG_SOLVER_SHM_DENY_HUGE").is_some();
  829         let group = read_number("/proc/sys/vm/hugetlb_shm_group");
  830         if simulated {
  831             // SAFETY: no pointers.
  832             let message = permission_message(group.unwrap_or(0), unsafe { libc::getegid() });
  833             return Some(format!("{message} (simulated: EGG_SOLVER_SHM_DENY_HUGE)"));
  834         }
  835         let group = group?;
  836         // SAFETY: no pointers; then a buffer of the size asked for.
  837         let (gid, groups) = unsafe {
  838             let n = usize::try_from(libc::getgroups(0, std::ptr::null_mut())).unwrap_or(0);
  839             let mut groups = vec![0; n];
  840             let got = libc::getgroups(i32::try_from(n).unwrap_or(0), groups.as_mut_ptr());
  841             groups.truncate(usize::try_from(got).unwrap_or(0));
  842             (libc::getegid(), groups)
  843         };
  844         let in_group = std::iter::once(gid)
  845             .chain(groups)
  846             .any(|g| u64::from(g) == group);
  847         // `CAP_IPC_LOCK` is capability 14 of the effective set.
  848         let status = std::fs::read_to_string("/proc/self/status").unwrap_or_default();
  849         let ipc_lock = status
  850             .lines()
  851             .find_map(|l| l.strip_prefix("CapEff:"))
  852             .and_then(|caps| u64::from_str_radix(caps.trim(), 16).ok())
  853             .is_some_and(|caps| caps & (1 << 14) != 0);
  854         (!in_group && !ipc_lock).then(|| permission_message(group, gid))
  855     }
  856 
  857     /// How to get the permission for huge-page segments.
  858     pub(super) fn permission_message(group: u64, gid: u32) -> String {
  859         format!(
  860             "no permission: huge-page segments need a group of yours in \
  861              vm.hugetlb_shm_group (now {group}) or CAP_IPC_LOCK; to allow your group: \
  862              `sudo sysctl vm.hugetlb_shm_group={gid}`"
  863         )
  864     }

  866     pub(super) fn remove(key: u32) -> Result<Option<Removed>, TableError> {
  867         let id = match shmget(key, 0, 0) {
  868             Ok(id) => id,
  869             Err(e) if is(&e, libc::ENOENT) => return Ok(None),
  870             Err(e) => return Err(call_error(key, "shmget")(e)),
  871         };
  872         let ds = stat(id).map_err(call_error(key, "shmctl(IPC_STAT)"))?;
  873         let held = match Attachment::new(id, true) {
  874             Ok(attachment) => match attachment.last_page().map(|page| parse_header(&page)) {
  875                 Some(Ok(Some(header))) => describe(&header.trailer),
  876                 Some(Ok(None)) => "no table yet (loading, or its load was interrupted)".into(),
  877                 _ => "no table".into(),
  878             },
  879             Err(e) => format!("unknown ({e})"),
  880         };
  881         mark_removed(id).map_err(call_error(key, "shmctl(IPC_RMID)"))?;
  882         #[allow(
  883             clippy::unnecessary_cast,
  884             reason = "shmatt_t is 32-bit on 32-bit targets"
  885         )]
  886         let attached = ds.shm_nattch as u64;
  887         Ok(Some(Removed {
  888             id,
  889             bytes: ds.shm_segsz as u64,
  890             held,
  891             attached,
  892         }))
  893     }
  894 }

 1112     /// A segment without a trailer is being loaded (its maker runs: `Shm`
 1113     /// says so, to try again; `Auto` loads the file) or was interrupted (its
 1114     /// maker is gone: `Shm` removes it and loads anew).
 1115     #[test]
 1116     fn unfinished_segments_are_found() {
 1117         let key = TestKey::new(2);
 1118         let table = TableFile::new("shm-unfinished", 0);
 1119         let bytes = segment_bytes(h48_bytes(0), NORMAL_PAGE);
 1120         let flags = libc::IPC_CREAT | libc::IPC_EXCL | 0o600;
 1121         let id = shmget(key.0, bytes, flags).unwrap();
 1122         let error = table.open(&key.options(LoadMode::Shm, "4k")).err().unwrap();
 1123         let me = std::process::id().cast_signed();
 1124         assert!(
 1125             matches!(error, TableError::ShmLoading { pid, .. } if pid == me),
 1126             "{error}"
 1127         );
 1128         let (_, report) = table.open(&key.options(LoadMode::Auto, "4k")).unwrap();
 1129         assert_eq!(report.segment, None);
 1130         assert!(
 1131             report.notes.iter().any(|n| n.contains("being loaded")),
 1132             "{:?}",
 1133             report.notes
 1134         );
 1135         assert!(remove(key.0).unwrap().is_some());
 1136         assert!(stat(id).is_err());
 1137         // Made by a child process that exits without loading it.
 1138         // SAFETY: the child only makes the segment and exits (no allocation,
 1139         // no locks).
 1140         unsafe {
 1141             let child = libc::fork();
 1142             if child == 0 {
 1143                 let ok =
 1144                     libc::shmget(key.0.cast_signed(), usize::try_from(bytes).unwrap(), flags) >= 0;
 1145                 libc::_exit(i32::from(!ok));
 1146             }
 1147             let mut status = 0;
 1148             assert_eq!(libc::waitpid(child, &raw mut status, 0), child);
 1149             assert_eq!(status, 0);
 1150         }
 1151         let (loaded, report) = table.open(&key.options(LoadMode::Shm, "4k")).unwrap();
 1152         assert!(report.segment.unwrap().created);
 1153         assert!(
 1154             report.notes.iter().any(|n| n.contains("was interrupted")),
 1155             "{:?}",
 1156             report.notes
 1157         );
 1158         assert_eq!(
 1159             &loaded.memory.bytes().unwrap()[..table.data.len()],
 1160             &table.data[..]
 1161         );
 1162     }
