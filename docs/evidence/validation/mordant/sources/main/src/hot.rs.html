    1 //! Fixed placement of the hot loops in the binary.
    2 //!
    3 //! On this machine (Zen 5), the 1-thread stream throughput can depend on
    4 //! where `bench::stream_thread` (the stream loop, with `run_lanes` inlined)
    5 //! starts within a 64-byte line. Functions are 16-byte aligned, so it can
    6 //! start at offset 0, 16, 32 or 48. On main abce87c, the same code at these
    7 //! offsets gave about 25.5, 25.0, 24.9 and 25.2 solves/s (3 layouts each).
    8 //! On 735dd25 the four pads were within the noise of each other (+-0.7%).
    9 //!
   10 //! The linker places functions one after another, so the offset changed
   11 //! with any change anywhere in the binary, and it moved A/B results by up
   12 //! to the size of most kernel changes. Now each hot function goes into a
   13 //! section of its own. `hot_section!` aligns that section to 64 bytes and
   14 //! starts it with a pad, so the function starts at the pad's offset in
   15 //! every build.
   16 //!
   17 //! A change to the hot code can change the best pad: re-sweep it with the
   18 //! build variables `EGG_SOLVER_STREAM_PAD` and `EGG_SOLVER_SHARE_PAD` (0, 16, 32
   19 //! or 48), and use the same pads on both sides of an A/B.
   20 
   21 /// Offset of `bench::stream_thread` in its 64-byte line: 16, where 735dd25
   22 /// had it (its four pads were within the noise of each other at 1T), and
   23 /// the best pad for m8's loop on def2cd0 (1T against main at 16, 2 pairs
   24 /// per pad: 0 -0.3 +0.6%, 16 +4.0 +0.5%, 32 -0.1 +1.2%, 48 +2.4 +1.3%).
   25 pub const STREAM_PAD: usize = {
   26     let pad = pad(
   27         crate::env::build_var(
   28             option_env!("EGG_SOLVER_STREAM_PAD"),
   29             option_env!("CUBEOPT_STREAM_PAD"),
   30         ),
   31         16,
   32     );
   33     // (A compile-time check: a build with another value fails.)
   34     assert!(is_pad(pad), "EGG_SOLVER_STREAM_PAD is 0, 16, 32 or 48");
   35     pad
   36 };

   38 /// Offset of the share mode's round loop (`share::Worker::rounds`) in its
   39 /// 64-byte line: 16. (With the loop visiting the lanes by pointer, m8 on
   40 /// def2cd0, 1T share against main at its pad 0, 2 pairs per pad: 0 +1.3
   41 /// +1.8%, 16 +5.3 +3.2%, 32 +1.6 +1.0%, 48 +1.7 +1.9%. Before: at 48T the

   56 /// The value of a pad build variable (decimal; `usize::MAX` if it is not
   57 /// a number), or `default`. (Evaluated at compile time only: the pads are
   58 /// constants.)
   59 const fn pad(var: Option<&str>, default: usize) -> usize {
   60     let Some(text) = var else {
   61         return default;
   62     };
   63     let mut value = 0usize;
   64     let mut digits = text.as_bytes();
   65     while let [digit, rest @ ..] = digits {
   66         if !digit.is_ascii_digit() || value > 64 {
   67             return usize::MAX;
   68         }
   69         value = value * 10 + (*digit - b'0') as usize;
   70         digits = rest;
   71     }
   72     value
   73 }

   75 /// Whether `pad` is a valid pad: 0, 16, 32 or 48.
   76 const fn is_pad(pad: usize) -> bool {
   77     pad < 64 && pad.is_multiple_of(16)
   78 }

   80 /// Starts section `$name` on a 64-byte boundary, followed by `$pad` bytes
   81 /// (int3, never executed). The hot function marked
   82 /// `#[unsafe(link_section = $name)]`, the section's only one, then starts
   83 /// at offset `$pad` of a 64-byte line. (Module-level asm is emitted before
   84 /// any function, and the function's own alignment, 16, divides the pad.)
   85 /// Only on x86-64 Linux (ELF section syntax, tuned on Zen 5): elsewhere the
   86 /// hot functions stay where the linker puts them (their `link_section` is
   87 /// set under the same condition).
   88 #[macro_export]
   89 macro_rules! hot_section {
   90     ($name:literal, $pad:expr) => {
   91         #[cfg(all(target_arch = "x86_64", target_os = "linux"))]
   92         ::core::arch::global_asm!(
   93             concat!(".pushsection ", $name, ",\"ax\",@progbits"),
   94             ".p2align 6",
   95             ".skip {pad}, 0xcc",
   96             ".popsection",
   97             pad = const $pad,
   98         );
   99     };
