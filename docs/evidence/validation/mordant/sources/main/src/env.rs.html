   18 /// Select the modern name when set, otherwise an existing legacy name.
   19 #[must_use]
   20 pub fn name(modern: &'static str) -> &'static str {
   21     if std::env::var_os(modern).is_some() {
   22         return modern;
   23     }
   24     ALIASES
   25         .iter()
   26         .find_map(|&(current, previous)| {
   27             (current == modern && std::env::var_os(previous).is_some()).then_some(previous)
   28         })
   29         .unwrap_or(modern)
   30 }

   32 /// Read a setting, preferring its modern name even when that value is invalid.
   33 ///
   34 /// # Errors
   35 /// If the selected variable is absent or is not Unicode.
   36 pub fn var(modern: &str) -> Result<String, std::env::VarError> {
   37     match std::env::var(modern) {
   38         Err(std::env::VarError::NotPresent) => {
   39             if let Some(suffix) = modern.strip_prefix("EGG_SOLVER_") {
   40                 std::env::var(format!("CUBEOPT_{suffix}"))
   41             } else {
   42                 Err(std::env::VarError::NotPresent)
   43             }
   44         }
   45         result => result,
   46     }
   47 }
   48 
   49 /// Read a possibly non-Unicode setting, with its legacy name as fallback.
   50 #[must_use]
   51 pub fn var_os(modern: &str) -> Option<OsString> {
   52     std::env::var_os(modern).or_else(|| {
   53         let suffix = modern.strip_prefix("EGG_SOLVER_")?;
   54         std::env::var_os(format!("CUBEOPT_{suffix}"))
   55     })
   56 }

   58 /// Select a compile-time setting without introducing runtime work.
   59 #[must_use]
   60 pub const fn build_var(
   61     modern: Option<&'static str>,
   62     previous: Option<&'static str>,
   63 ) -> Option<&'static str> {
   64     match modern {
   65         Some(value) => Some(value),
   66         None => previous,
   67     }
