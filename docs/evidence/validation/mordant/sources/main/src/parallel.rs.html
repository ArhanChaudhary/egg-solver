    1 //! Parallel loops that also run where there are no threads (WebAssembly
    2 //! without the threads proposal): the calling thread does one share of the
    3 //! work itself, so one thread spawns nothing. A host can also run them on
    4 //! threads of its own (`set_runner`).
    5 
    6 use std::sync::{Mutex, OnceLock, PoisonError};
    7 
    8 /// Runs `work(t)` for every t in `0..n`, t = 0 on the calling thread, and
    9 /// returns once all have returned.
   10 pub type Runner = fn(n: usize, work: &(dyn Fn(usize) + Sync));
   11 
   12 static RUNNER: OnceLock<Runner> = OnceLock::new();
   13 
   14 /// Have `on_threads` use `runner` instead of spawning threads for each call.
   15 /// The wasm threads build installs a pool of threads that never exit (in
   16 /// wasi-libc, a thread could stay stuck in its exit; see `wasm/src/threads.rs`).
   17 /// Returns false if a runner was set already.
   18 pub fn set_runner(runner: Runner) -> bool {
   19     RUNNER.set(runner).is_ok()
   20 }
   21 
   22 /// A worker thread of `on_threads` panicked. (Only where panics unwind, as
   23 /// in tests: the release profile aborts on a panic.)
   24 #[derive(Debug, thiserror::Error)]
   25 #[error("a worker thread panicked")]
   26 pub struct WorkerPanicked;
   27 
   28 /// `work(t)` for t in `0..threads` (at least one): t = 0 on the calling
   29 /// thread, the others on scoped threads (or the `set_runner` threads). A
   30 /// share whose thread cannot be started runs on the calling thread instead.
   31 /// Returns the results in order of t.
   32 ///
   33 /// # Errors
   34 /// If a worker thread panicked.
   35 pub fn on_threads<R: Send>(
   36     threads: usize,
   37     work: impl Fn(usize) -> R + Sync,
   38 ) -> Result<Vec<R>, WorkerPanicked> {
   39     let work = &work;
   40     if let Some(runner) = RUNNER.get() {
   41         let results: Vec<Mutex<Option<R>>> =
   42             (0..threads.max(1)).map(|_| Mutex::new(None)).collect();
   43         runner(results.len(), &|t| {
   44             if let Some(slot) = results.get(t) {
   45                 *slot.lock().unwrap_or_else(PoisonError::into_inner) = Some(work(t));
   46             }
   47         });
   48         // (A share that the runner did not run runs here.)
   49         return Ok((0..)
   50             .zip(results)
   51             .map(|(t, r)| {
   52                 let result = r.into_inner().unwrap_or_else(PoisonError::into_inner);
   53                 result.unwrap_or_else(|| work(t))
   54             })
   55             .collect());
   56     }
   57     std::thread::scope(|s| {
   58         let spawned: Vec<_> = (1..threads.max(1))
   59             .map(|t| {
   60                 (
   61                     t,
   62                     std::thread::Builder::new().spawn_scoped(s, move || work(t)),
   63                 )
   64             })
   65             .collect();
   66         let first = work(0);
   67         // Join every worker before propagating an error. A short-circuiting
   68         // Result collection would leave later handles to the scope, whose
   69         // automatic joins would panic if another worker also failed.
   70         let joined: Vec<_> = spawned
   71             .into_iter()
   72             .map(|(t, spawned)| match spawned {
   73                 Ok(handle) => handle.join().map_err(|_| WorkerPanicked),
   74                 Err(_) => Ok(work(t)),
   75             })
   76             .collect();
   77         std::iter::once(Ok(first)).chain(joined).collect()
   78     })
   79 }

   85         for threads in [0, 1, 2, 5] {
