    1 //! egg-solver as a WebAssembly module: a plain C ABI of a few functions, which
    2 //! `egg-solver.mjs` wraps for JavaScript (Bun and browsers).
    3 //!
    4 //! Strings go in as (pointer, length) of UTF-8 in buffers from
    5 //! `egg_solver_alloc`. Results come out as text in an output buffer
    6 //! (`egg_solver_output`, `egg_solver_output_len`): JSON after a call that returns 0,
    7 //! an error message after one that returns -1. A table file is copied by the
    8 //! caller straight into a buffer from `egg_solver_table_buffer`, then checked
    9 //! and installed by `egg_solver_table_load`.
   10 //!
   11 //! Built for `wasm32-wasip1` (one thread) or `wasm32-wasip1-threads` (std
   12 //! threads through wasi-threads: the share and stream modes run on several
   13 //! threads as natively). See README.md, "WebAssembly".

   43 /// The WASI reactor initializer, which a Rust cdylib does not get from
   44 /// wasi-libc (its `crt1-reactor`): the host calls it once, on the main
   45 /// thread, before any other export (`egg-solver.mjs` does). In the threads
   46 /// build it first sets up the main thread's pthread record. Without that,
   47 /// wasi-libc's thread list grew from a zeroed record and came apart: an
   48 /// exiting thread could find itself alone in the list and call `exit(0)`
   49 /// instead of finishing its exit, so a `join` on it waited forever (the
   50 /// intermittent hang of `wasm/test.mjs`, at ~1 in 3 runs under load). It
   51 /// then installs the thread pool (`threads.rs`).
   52 #[unsafe(no_mangle)]
   53 pub extern "C" fn _initialize() {
   54     static DONE: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
   55     unsafe extern "C" {
   56         #[cfg(wasm_threads)]
   57         fn __wasi_init_tp();
   58         fn __wasm_call_ctors();
   59     }
   60     if DONE.swap(true, std::sync::atomic::Ordering::Relaxed) {
   61         return;
   62     }
   63     // SAFETY: wasi-libc's and the linker's initializers, run once, first.
   64     unsafe {
   65         #[cfg(wasm_threads)]
   66         __wasi_init_tp();
   67         __wasm_call_ctors();
   68     }
   69     #[cfg(wasm_threads)]
   70     let _ = egg_solver::parallel::set_runner(threads::run);
   71 }

   96 /// The cocsep table of the state (`State::corners`), made now if it is not
   97 /// yet (it takes ~0.2 s).
   98 fn corners(corners: &mut Option<Cocsep>) -> Result<&Cocsep, String> {
   99     match corners {
  100         Some(corners) => Ok(corners),
  101         None => Ok(corners.insert(Cocsep::generate().map_err(|e| e.to_string())?)),
  102     }

  105 /// Run `f` on the state; its `Ok` text or `Err` message becomes the output.
  106 /// Returns 0 or -1.
  107 fn with_state(f: impl FnOnce(&mut State) -> Result<String, String>) -> i32 {
  108     let mut state = state();
  109     let (output, status) = match f(&mut state) {
  110         Ok(text) => (text, 0),
  111         Err(message) => (message, -1),
  112     };
  113     state.output = output;
  114     status
  115 }

  117 /// A string argument.
  118 ///
  119 /// # Safety
  120 /// `ptr` points to `len` readable bytes (or `len` is 0).
  121 unsafe fn text_argument<'a>(ptr: *const u8, len: usize) -> Result<&'a str, String> {
  122     if len == 0 {
  123         return Ok("");
  124     }
  125     // SAFETY: as the caller promises.
  126     let bytes = unsafe { std::slice::from_raw_parts(ptr, len) };
  127     std::str::from_utf8(bytes).map_err(|_| "the text is not UTF-8".to_string())
  128 }

  130 /// A cube from a scramble ("R U2 F' ...") or a facelet string (54 stickers
  131 /// of six symbols; see `CubieCube::from_facelets`). A scramble of 54 moves
  132 /// without suffixes of all six faces would read as facelets.
  133 fn parse_cube(text: &str) -> Result<CubieCube, String> {
  134     let stickers: Vec<char> = text.chars().filter(|c| !c.is_whitespace()).collect();
  135     let mut symbols = stickers.clone();
  136     symbols.sort_unstable();
  137     symbols.dedup();
  138     let is_facelets = stickers.len() == 54
  139         && symbols.len() == 6
  140         && !symbols.iter().any(|c| matches!(c, '1' | '2' | '3' | '\''));
  141     if is_facelets {
  142         return CubieCube::from_facelets(text).map_err(|e| e.to_string());
  143     }
  144     parse_moves(text)
  145         .and_then(|moves| CubieCube::SOLVED.apply_moves(&moves))
  146         .ok_or_else(|| format!("neither a scramble nor 54 facelets: {text:?}"))
  147 }

  149 /// A solved cube as JSON (`{"index":..,"error":..}` if it has no
  150 /// solution).
  151 fn result_json(index: usize, solution: Option<&[u8]>, stats: Stats, seconds: f64) -> String {
  152     let Some(solution) = solution else {
  153         return format!(r#"{{"index":{index},"error":"{NO_SOLUTION}"}}"#);
  154     };
  155     format!(
  156         r#"{{"index":{index},"moves":"{}","length":{},"nodes":{},"lookups":{},"seconds":{seconds:.6}}}"#,
  157         format_moves(solution),
  158         solution.len(),
  159         stats.nodes,
  160         stats.lookups
  161     )
  162 }

  178 /// Allocate `len` bytes for the caller (strings passed in); 0 if out of
  179 /// memory. Free with `egg_solver_free`.
  180 #[unsafe(no_mangle)]
  181 pub extern "C" fn egg_solver_alloc(len: usize) -> *mut u8 {
  182     let Ok(layout) = std::alloc::Layout::from_size_align(len.max(1), 1) else {
  183         return std::ptr::null_mut();
  184     };
  185     // SAFETY: a non-zero size.
  186     unsafe { std::alloc::alloc(layout) }
  187 }

  219 /// A buffer of `len` bytes for a whole table file (data and trailer), which
  220 /// the caller fills before `egg_solver_table_load`. Drops the current table
  221 /// first (wasm32 has room for one). Returns 0 if out of memory.
  222 #[unsafe(no_mangle)]
  223 pub extern "C" fn egg_solver_table_buffer(len: usize) -> *mut u8 {
  224     let mut state = state();
  225     state.table = None;
  226     state.incoming = None;
  227     state.incoming = Memory::try_anonymous(len);
  228     state
  229         .incoming
  230         .as_ref()
  231         .map_or(std::ptr::null_mut(), Memory::ptr)
  232 }

  234 /// Install the table file copied into the `egg_solver_table_buffer`: checks
  235 /// its trailer and size, and if `verify` is not 0 its checksum. Output: the
  236 /// table's `{"k":..,"base":..,"bytes":..}`.
  237 #[unsafe(no_mangle)]
  238 pub extern "C" fn egg_solver_table_load(verify: u32) -> i32 {
  239     with_state(|state| {
  240         let memory = state
  241             .incoming
  242             .take()
  243             .ok_or("no table buffer (egg_solver_table_buffer)")?;
  244         let table = H48Table::from_file_memory(memory, verify != 0, threads_for(0))
  245             .map_err(|e| e.to_string())?;
  246         let info = table_json(&table);
  247         state.table = Some(table);
  248         Ok(info)
  249     })
  250 }

  262 /// Generate table hk (k = 0..6) in memory (`base` 0: the default for `k`)
  263 /// on `threads` threads (0: all), replacing the current table. Output: as
  264 /// `egg_solver_table_load`.
  265 #[unsafe(no_mangle)]
  266 pub extern "C" fn egg_solver_generate(k: u32, base: u32, threads: u32) -> i32 {
  267     with_state(|state| {
  268         // (h7 fits in wasm32's 4 GiB, but not with the generator's own
  269         // memory: make it with the native `gen` and load the file.)
  270         if k > 6 {
  271             return Err(format!(
  272                 "h{k} is too big to generate here (h0..h6): use the native gen and load its file"
  273             ));
  274         }
  275         let base = match base {
  276             0 => default_base(k),
  277             b => u8::try_from(b)
  278                 .ok()
  279                 .filter(|&b| (1..=12).contains(&b))
  280                 .ok_or(format!("base {b}: 1..12"))?,
  281         };
  282         state.table = None;
  283         state.incoming = None;
  284         // (A table that does not fit would abort in the allocator: check
  285         // that it can be allocated first.)
  286         let bytes = usize::try_from(h48_bytes(k)).map_err(|_| format!("h{k} is too big"))?;
  287         drop(Memory::try_anonymous(bytes).ok_or(format!("no room for h{k} ({bytes} bytes)"))?);
  288         let corners = corners(&mut state.corners)?;
  289         let table = egg_solver::generate_table(corners, k, base, threads_for(threads), false)
  290             .map_err(|e| e.to_string())?;
  291         let info = table_json(&table);
  292         state.table = Some(table);
  293         Ok(info)
  294     })
  295 }

  297 /// Solve one cube (a scramble or facelets, see `parse_cube`) with all lanes
  298 /// of all threads on it (the share mode with one cube in progress: the
  299 /// lowest latency). `lanes` per thread (0: 16, see `lanes_for`), `threads`
  300 /// (0: all). Output: `{"index":0,"moves":..,"length":..,"nodes":..,
  301 /// "lookups":..,"seconds":..}`.
  302 ///
  303 /// # Safety
  304 /// `text` points to `len` readable bytes.
  305 #[unsafe(no_mangle)]
  306 pub unsafe extern "C" fn egg_solver_solve(
  307     text: *const u8,
  308     len: usize,
  309     lanes: u32,
  310     threads: u32,
  311 ) -> i32 {
  312     // SAFETY: as the caller promises.
  313     let text = unsafe { text_argument(text, len) };
  314     with_state(|state| {
  315         let cube = parse_cube(text?)?;
  316         let corners = corners(&mut state.corners)?;
  317         let table = state.table.as_ref().ok_or("no table loaded")?;
  318         let solver = Solver::new(corners, table);
  319         let lanes = lanes_for(lanes);
  320         let threads = threads_for(threads);
  321         let get = |i: usize| (i == 0).then_some(cube);
  322         let result = Mutex::new(None);
  323         let done = |i: usize, solution: Option<Vec<u8>>, stats: Stats, seconds: f64| {
  324             *result.lock().unwrap_or_else(PoisonError::into_inner) =
  325                 Some(solution.map(|solution| result_json(i, Some(&solution), stats, seconds)));
  326         };
  327         // (`share::solve_all`, on `on_threads`: the pool's threads in the
  328         // threads build, this thread alone in the other.)
  329         let pool = Pool::new(&get, 1, threads);
  330         let stop = AtomicBool::new(false);
  331         egg_solver::parallel::on_threads(threads, |_| {
  332             pool.run(&solver, lanes, &stop, |i, solution, stats, seconds| {
  333                 done(i, solution, stats, seconds);
  334             });
  335         })
  336         .map_err(|e| e.to_string())?;
  337         let result = result.into_inner().unwrap_or_else(PoisonError::into_inner);
  338         result
  339             .ok_or("not solved")?
  340             .ok_or_else(|| NO_SOLUTION.to_string())
  341     })
  342 }

  344 /// Solve many cubes, one per line, for throughput: the stream mode (each
  345 /// lane solves its own cube) on `threads` threads of `lanes` lanes. Output:
  346 /// a JSON array of the results (as `egg_solver_solve`) in input order.
  347 ///
  348 /// # Safety
  349 /// `text` points to `len` readable bytes.
  350 #[unsafe(no_mangle)]
  351 pub unsafe extern "C" fn egg_solver_solve_batch(
  352     text: *const u8,
  353     len: usize,
  354     lanes: u32,
  355     threads: u32,
  356 ) -> i32 {
  357     // SAFETY: as the caller promises.
  358     let text = unsafe { text_argument(text, len) };
  359     with_state(|state| {
  360         let cubes: Vec<CubieCube> = text?
  361             .lines()
  362             .filter(|l| !l.trim().is_empty())
  363             .map(parse_cube)
  364             .collect::<Result<_, _>>()?;
  365         let corners = corners(&mut state.corners)?;
  366         let table = state.table.as_ref().ok_or("no table loaded")?;
  367         let solver = Solver::new(corners, table);
  368         let lanes = lanes_for(lanes);
  369         let get = |i: usize| cubes.get(i).copied();
  370         let results = Mutex::new(vec![String::new(); cubes.len()]);
  371         let done = |i: usize, solution: Option<Vec<u8>>, stats: Stats, seconds: f64| {
  372             // (`i` is the index of a cube: `get` has no others.)
  373             let mut results = results.lock().unwrap_or_else(PoisonError::into_inner);
  374             if let Some(slot) = results.get_mut(i) {
  375                 *slot = result_json(i, solution.as_deref(), stats, seconds);
  376             }
  377         };
  378         stream(
  379             &solver,
  380             &get,
  381             lanes,
  382             threads_for(threads),
  383             &AtomicBool::new(false),
  384             &done,
  385         )?;
  386         let results = results.into_inner().unwrap_or_else(PoisonError::into_inner);
  387         Ok(format!("[{}]", results.join(",")))
  388     })
  389 }

  391 /// Stream throughput, as the native `bench <table> stream <warmup> <window>
  392 /// <threads> <lanes> <seed>`: an endless stream of random cubes (cube i
  393 /// from its own seed, the same cubes as `bench`), solves counted during
  394 /// `window` seconds after `warmup`. Output: `{"solves_per_second":..,
  395 /// "nodes_per_cube":..,"lookups_per_cube":..,"cubes":..,"threads":..,
  396 /// "lanes":..}`.
  397 #[unsafe(no_mangle)]
  398 pub extern "C" fn egg_solver_bench(
  399     seed: u32,
  400     warmup: f64,
  401     window: f64,
  402     lanes: u32,
  403     threads: u32,
  404 ) -> i32 {
  405     with_state(|state| {
  406         let corners = corners(&mut state.corners)?;
  407         let table = state.table.as_ref().ok_or("no table loaded")?;
  408         let solver = Solver::new(corners, table);
  409         let lanes = lanes_for(lanes);
  410         let threads = threads_for(threads);
  411         let seed = u64::from(seed);
  412         let cubes = |i: usize| {
  413             let mut rng = Rng(seed.wrapping_mul(0x9E37_79B9_7F4A_7C15) ^ (i as u64));
  414             rng.next_u64();
  415             Some(CubieCube::random(&mut rng))
  416         };
  417         let stop = AtomicBool::new(false);
  418         // (solves, nodes, lookups) in the window.
  419         let counted = Mutex::new((0u64, 0u64, 0u64));
  420         let start = std::time::Instant::now();
  421         let done = |_: usize, _: Option<Vec<u8>>, stats: Stats, _: f64| {
  422             let at = start.elapsed().as_secs_f64();
  423             if at >= warmup + window {
  424                 stop.store(true, std::sync::atomic::Ordering::Relaxed);
  425             } else if at >= warmup {
  426                 let mut counted = counted.lock().unwrap_or_else(PoisonError::into_inner);
  427                 counted.0 += 1;
  428                 counted.1 += stats.nodes;
  429                 counted.2 += stats.lookups;
  430             }
  431         };
  432         stream(&solver, &cubes, lanes, threads, &stop, &done)?;
  433         let (cubes_counted, nodes, lookups) =
  434             counted.into_inner().unwrap_or_else(PoisonError::into_inner);
  435         #[expect(clippy::cast_precision_loss, reason = "statistics")]
  436         let per_cube = |x: u64| x as f64 / cubes_counted.max(1) as f64;
  437         #[expect(clippy::cast_precision_loss, reason = "statistics")]
  438         let rate = cubes_counted as f64 / window;
  439         Ok(format!(
  440             r#"{{"solves_per_second":{rate:.4},"nodes_per_cube":{:.0},"lookups_per_cube":{:.0},"cubes":{cubes_counted},"threads":{threads},"lanes":{lanes}}}"#,
  441             per_cube(nodes),
  442             per_cube(lookups)
  443         ))
  444     })
  445 }

  447 /// The stream mode (`lanes::run_lanes`: each lane solves its own cube) on
  448 /// `threads` threads, until `cubes` has no more or `stop` is set. Every
  449 /// export runs this one compiled copy of the search: V8 (browsers, Node)
  450 /// first runs code with its baseline compiler and gives a function its
  451 /// optimized code from its next call on, so `egg-solver.mjs`'s warm-up, which
  452 /// solves a few cubes, gets the optimized search for all later calls.
  453 #[inline(never)]
  454 fn stream(
  455     solver: &Solver,
  456     cubes: &(dyn Fn(usize) -> Option<CubieCube> + Sync),
  457     lanes: usize,
  458     threads: usize,
  459     stop: &AtomicBool,
  460     done: &(dyn Fn(usize, Option<Vec<u8>>, Stats, f64) + Sync),
  461 ) -> Result<(), String> {
  462     let next = AtomicUsize::new(0);
  463     egg_solver::parallel::on_threads(threads, |_| {
  464         egg_solver::lanes::run_lanes(solver, cubes, &next, stop, lanes, done);
  465     })
  466     .map_err(|e| e.to_string())?;
  467     Ok(())
  468 }

  470 /// Lanes per thread: as asked, else 16. Lanes pay off in wasm too (the
  471 /// table load of one lane overlaps the work of the next ones; 8 lanes: +62%
  472 /// against 1), but already fully at 8: V8, h6, 100 depth-17 cubes, 1 thread,
  473 /// 1/8/16/24/40/64 lanes: 200.6/123.8/125.3/128.2/130.3/127.9 s. 16 lanes,
  474 /// with some margin, take 23 MB instead of the 56 MB of 40.
  475 fn lanes_for(asked: u32) -> usize {
  476     match asked {
  477         0 => 16,
  478         n => n as usize,
  479     }
  480 }

  482 /// `count` uniformly random cube states from `seed` (the same ones as the
  483 /// native `bench <table> random:<count>:<seed>`), as facelet strings.
  484 /// Output: a JSON array of strings.
  485 #[unsafe(no_mangle)]
  486 pub extern "C" fn egg_solver_random(seed: u32, count: u32) -> i32 {
  487     with_state(|_| {
  488         let mut rng = Rng(u64::from(seed));
  489         let mut json = String::from("[");
  490         for i in 0..count {
  491             let separator = if i == 0 { "" } else { "," };
  492             let _ = write!(
  493                 json,
  494                 r#"{separator}"{}""#,
  495                 CubieCube::random(&mut rng).to_facelets()
  496             );
  497         }
  498         json.push(']');
  499         Ok(json)
  500     })
