   48 /// Solve with the recursive solver, all threads.
   49 fn par_solve(cubes: &[CubieCube], table: &H48Table) -> Vec<Vec<u8>> {
   50     let solver = Solver::new(corners(), table);
   51     let out = std::sync::Mutex::new(vec![vec![]; cubes.len()]);
   52     let next = std::sync::atomic::AtomicUsize::new(0);
   53     std::thread::scope(|s| {
   54         for _ in 0..egg_solver::thread_count() {
   55             s.spawn(|| loop {
   56                 let i = next.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
   57                 if i >= cubes.len() {
   58                     break;
   59                 }
   60                 let (solution, _) = solver.solve(&cubes[i]);
   61                 out.lock().unwrap()[i] = solution.unwrap();
   62             });
   63         }
   64     });
   65     out.into_inner().unwrap()
   66 }

   68 /// All positions within `depth` moves, with their distance.
   69 fn ball(depth: u8) -> HashMap<[u8; 32], (Cube, u8)> {
   70     let mut seen: HashMap<[u8; 32], (Cube, u8)> = HashMap::new();
   71     let mut frontier = vec![Cube::solved()];
   72     seen.insert(Cube::solved().to_bytes(), (Cube::solved(), 0));
   73     for distance in 1..=depth {
   74         let mut next = Vec::new();
   75         for cube in frontier {
   76             for mv in Move::ALL {
   77                 let child = cube.apply_move(mv);
   78                 if let std::collections::hash_map::Entry::Vacant(e) = seen.entry(child.to_bytes()) {
   79                     e.insert((child, distance));
   80                     next.push(child);
   81                 }
   82             }
   83         }
   84         frontier = next;
   85     }
   86     seen
   87 }

   89 /// Generate h0 with a small base and check every entry of depth <= 5 against
   90 /// brute force (this exercises the BFS and the self-symmetry handling).
   91 #[test]
   92 fn h0_table_matches_brute_force() {
   93     let corners = corners();
   94     let base = 3u8;
   95     let (table, _) =
   96         unsafe { egg_solver::generate(corners, 0, base, None, &[], egg_solver::thread_count()) }
   97             .unwrap();
   98     // The distance of each table entry: the least over the cubes it covers.
   99     let mut best: HashMap<u64, u8> = HashMap::new();
  100     for &(cube, distance) in ball(5).values() {
  101         let index = corners.index(cube, 0);
  102         let entry = best.entry(index).or_insert(distance);
  103         *entry = (*entry).min(distance);
  104         // admissible
  105         let (code, group_min) = table.get(index).unwrap();
  106         let value = if code == 0 { group_min } else { base + code };
  107         assert!(
  108             value <= distance,
  109             "inadmissible: index {index} value {value} > distance {distance}"
  110         );
  111     }
  112     let mut want = [0u64; 3];
  113     for (&index, &distance) in &best {
  114         let (code, _) = table.get(index).unwrap();
  115         let expect = distance.saturating_sub(base).min(3);
  116         assert_eq!(code, expect, "index {index} depth {distance}");
  117         if distance <= base + 2 {
  118             want[usize::from(expect)] += 1;
  119         }
  120     }
  121     let counts = table.distribution().unwrap();
  122     assert_eq!(&counts[..3], &want[..], "entries with depth <= base+2");
  123 }

  135 /// Independent reference: plain IDA* with only the corner (cocsep) bound.
  136 fn reference_length(cube: &CubieCube, corners: &Cocsep) -> usize {
  137     /// A solution of exactly `depth` moves, not starting on `last_face`?
  138     fn search(cube: Cube, corners: &Cocsep, depth: u8, last_face: usize) -> bool {
  139         if depth == 0 {
  140             return cube.is_solved();
  141         }
  142         if entry_depth(corners.entries()[cocsep(cube)]) > depth {
  143             return false;
  144         }
  145         for mv in Move::ALL {
  146             if mv.index() / 3 == last_face {
  147                 continue;
  148             }
  149             if search(cube.apply_move(mv), corners, depth - 1, mv.index() / 3) {
  150                 return true;
  151             }
  152         }
  153         false
  154     }
  155     let cube = Cube::from_cubie(cube);
  156     (0..=20)
  157         .find(|&depth| search(cube, corners, depth, 6))
  158         .map(usize::from)
  159         .unwrap()
  160 }

  275 /// Solve with the work-sharing lanes (`lanes::share`): `threads` threads of
  276 /// `nlanes` lanes, at most `max_active` cubes at once. Returns the solutions
  277 /// and the pool's event counters.
  278 fn share_solve(
  279     cubes: &[CubieCube],
  280     table: &H48Table,
  281     threads: usize,
  282     nlanes: usize,
  283     max_active: usize,
  284 ) -> (Vec<Vec<u8>>, String) {
  285     let solver = Solver::new(corners(), table);
  286     let out = std::sync::Mutex::new(vec![None; cubes.len()]);
  287     let get = |i: usize| cubes.get(i).copied();
  288     let pool = egg_solver::lanes::share::Pool::new(&get, max_active, threads);
  289     let stop = std::sync::atomic::AtomicBool::new(false);
  290     std::thread::scope(|s| {
  291         for _ in 0..threads {
  292             s.spawn(|| {
  293                 // (A panic stops the other threads: they would wait for its
  294                 // work forever.)
  295                 let r = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
  296                     pool.run(&solver, nlanes, &stop, |i, sol, _, _| {
  297                         let mut o = out.lock().unwrap();
  298                         assert!(o[i].is_none(), "cube {i} reported twice");
  299                         o[i] = Some(sol);
  300                     });
  301                 }));
  302                 if let Err(e) = r {
  303                     stop.store(true, std::sync::atomic::Ordering::Relaxed);
  304                     std::panic::resume_unwind(e);
  305                 }
  306             });
  307         }
  308     });
  309     let solutions = out.into_inner().unwrap();
  310     let solutions = solutions
  311         .into_iter()
  312         .enumerate()
  313         .map(|(i, s)| {
  314             s.flatten()
  315                 .unwrap_or_else(|| panic!("cube {i} not reported or solved"))
  316         })
  317         .collect();
  318     (solutions, pool.counters())
  319 }

  508 /// hk generated depth-first (`gen::dfs`) with base 3 and a BFS prefix of
  509 /// `prefix_depth`, in `parts` row ranges (each in its own memory).
  510 fn dfs_table_in_parts(k: u32, prefix_depth: u8, parts: u64) -> H48Table {
  511     let settings = egg_solver::gen::dfs::Settings {
  512         k,
  513         base: 3,
  514         threads: egg_solver::thread_count(),
  515         prefix_depth,
  516         verbose: false,
  517     };
  518     let generator = egg_solver::gen::dfs::Generator::new(corners(), settings).unwrap();
  519     let memory = Memory::anonymous(usize::try_from(h48_bytes(k)).unwrap()).unwrap();
  520     let rows = h48_rows(k);
  521     let per = rows.div_ceil(parts);
  522     for start in (0..rows).step_by(usize::try_from(per).unwrap()) {
  523         let range = start..(start + per).min(rows);
  524         let part_mem =
  525             Memory::anonymous(usize::try_from((range.end - range.start) * ROW_BYTES).unwrap())
  526                 .unwrap();
  527         let part = egg_solver::gen::dfs::Part {
  528             // SAFETY: fresh writable memory, copied only after workers finish.
  529             words: unsafe { part_mem.words().unwrap() },
  530             first_slot: range.start * N_SLOTS as u64,
  531             end_slot: range.end * N_SLOTS as u64,
  532         };
  533         generator.generate(&range, &part).unwrap();
  534         let offset = usize::try_from(range.start * ROW_BYTES).unwrap();
  535         // SAFETY: the part's rows of the table's own memory.
  536         unsafe {
  537             std::ptr::copy_nonoverlapping(
  538                 part_mem.ptr(),
  539                 memory.ptr().add(offset),
  540                 part_mem.bytes().unwrap().len(),
  541             );
  542         }
  543     }
  544     H48Table::from_memory(memory, H48Meta { k, base: 3 }).unwrap()
  545 }

  547 fn dfs_table(k: u32, prefix_depth: u8) -> H48Table {
  548     dfs_table_in_parts(k, prefix_depth, 1)
  549 }

  551 /// The depth-first generator gives the BFS's h0 byte for byte (for any
  552 /// prefix depth), an h2 that matches brute force (codes, group minima, no
  553 /// entry too low), and hk that collapse to h(k-1) (`gen::collapse`).
  554 #[test]
  555 fn dfs_generator_matches_bfs_and_brute_force() {
  556     let cs = corners();
  557     let base = 3u8;
  558     let len = usize::try_from(h48_bytes(0)).unwrap();
  559     let bfs = unsafe { egg_solver::generate(cs, 0, base, None, &[], egg_solver::thread_count()) }
  560         .unwrap()
  561         .0;
  562     for prefix_depth in [1, 2, 5] {
  563         let t = dfs_table(0, prefix_depth);
  564         assert!(
  565             t.memory().bytes().unwrap()[..len] == bfs.memory().bytes().unwrap()[..len],
  566             "h0 with prefix depth {prefix_depth}"
  567         );
  568     }
  569     // h2 against all positions within 5 moves (base + 2: every code but 3
  570     // is decided by them).
  571     let k = 2;
  572     let t = dfs_table(k, 2);
  573     let mut best: HashMap<u64, u8> = HashMap::new();
  574     for &(c, d) in ball(5).values() {
  575         let e = best.entry(cs.index(c, k)).or_insert(d);
  576         *e = (*e).min(d);
  577     }
  578     let mut group_min: HashMap<u64, u8> = HashMap::new();
  579     for (&idx, &d) in &best {
  580         let (code, _) = t.get(idx).unwrap();
  581         assert_eq!(code, d.saturating_sub(base).min(3), "idx {idx} depth {d}");
  582         let g = group_min.entry(idx >> 6).or_insert(d);
  583         *g = (*g).min(d);
  584     }
  585     let dist = t.distribution().unwrap();
  586     assert_eq!(
  587         dist[0] + dist[1] + dist[2],
  588         best.len() as u64,
  589         "no entry too low"
  590     );
  591     for (&g, &d) in &group_min {
  592         assert_eq!(t.get(g << 6 | 2).unwrap().1, d, "group {g}");
  593     }
  594     // Collapse: h3 -> h2, h3 -> h0.
  595     let t3 = dfs_table(3, 2);
  596     // In parts (rows of several classes and parts of a class): the same.
  597     let n3 = usize::try_from(h48_bytes(3)).unwrap();
  598     for parts in [3, 7] {
  599         let t = dfs_table_in_parts(3, 2, parts);
  600         assert!(
  601             t.memory().bytes().unwrap()[..n3] == t3.memory().bytes().unwrap()[..n3],
  602             "h3 in {parts} parts"
  603         );
  604     }
  605     let threads = egg_solver::thread_count();
  606     for (small, want) in [(&t, 2), (&bfs, 0)] {
  607         let derived = egg_solver::gen::collapse::derive(&t3, want, threads).unwrap();
  608         let n = usize::try_from(h48_bytes(want)).unwrap();
  609         assert!(
  610             derived.memory().bytes().unwrap()[..n] == small.memory().bytes().unwrap()[..n],
  611             "h3 -> h{want}"
  612         );
  613         let diff = egg_solver::gen::collapse::compare(small, &t3, threads).unwrap();
  614         assert_eq!((diff.codes, diff.group_minima), (0, 0), "{diff:?}");
  615     }
  616 }

  682 /// (length, nodes) of each of `cubes` in stream lanes (3 threads of 5).
  683 fn lanes_lengths_and_nodes(solver: &Solver, cubes: &[CubieCube]) -> Vec<(usize, u64)> {
  684     let get = |i: usize| cubes.get(i).copied();
  685     let next = std::sync::atomic::AtomicUsize::new(0);
  686     let stop = std::sync::atomic::AtomicBool::new(false);
  687     let out = std::sync::Mutex::new(vec![(0, 0); cubes.len()]);
  688     std::thread::scope(|scope| {
  689         for _ in 0..3 {
  690             scope.spawn(|| {
  691                 egg_solver::lanes::run_lanes(solver, &get, &next, &stop, 5, |i, sol, stats, _| {
  692                     let sol = sol.unwrap();
  693                     assert_eq!(cubes[i].apply_moves(&sol), Some(CubieCube::SOLVED));
  694                     out.lock().unwrap()[i] = (sol.len(), stats.nodes);
  695                 });
  696             });
  697         }
  698     });
  699     out.into_inner().unwrap()
  700 }

  702 /// (length, nodes) of each of `cubes` in share mode (3 threads of 4 lanes,
  703 /// at most `max_active` cubes at once), and the pool's event counters.
  704 fn share_lengths_and_nodes(
  705     solver: &Solver,
  706     cubes: &[CubieCube],
  707     max_active: usize,
  708 ) -> (Vec<(usize, u64)>, String) {
  709     let get = |i: usize| cubes.get(i).copied();
  710     let stop = std::sync::atomic::AtomicBool::new(false);
  711     let pool = egg_solver::lanes::share::Pool::new(&get, max_active, 3);
  712     let out = std::sync::Mutex::new(vec![(0, 0); cubes.len()]);
  713     std::thread::scope(|scope| {
  714         for _ in 0..3 {
  715             scope.spawn(|| {
  716                 pool.run(solver, 4, &stop, |i, sol, stats, _| {
  717                     let sol = sol.unwrap();
  718                     assert_eq!(cubes[i].apply_moves(&sol), Some(CubieCube::SOLVED));
  719                     out.lock().unwrap()[i] = (sol.len(), stats.nodes);
  720                 });
  721             });
  722         }
  723     });
  724     (out.into_inner().unwrap(), pool.counters())
  725 }

  727 /// Symmetric start (`symstart`) keeps the optimal lengths in every mode
  728 /// (share mode: in its first pass, h0's iterations up to 14, and after it)
  729 /// and searches fewer nodes; random states are not filtered (same nodes).
  730 #[test]
  731 fn symmetric_start_keeps_optimal_lengths() {
  732     use egg_solver::symstart::SymFilter;
  733     let cubes: Vec<CubieCube> = SYMMETRIC_CASES
  734         .iter()
  735         .map(|(scramble, _)| {
  736             CubieCube::SOLVED
  737                 .apply_moves(&parse_moves(scramble).unwrap())
  738                 .unwrap()
  739         })
  740         .collect();
  741     let known: Vec<usize> = SYMMETRIC_CASES.iter().map(|case| case.1).collect();
  742     // (Only the cubes with a symmetry are filtered.)
  743     for (i, cubie) in cubes.iter().enumerate() {
  744         let cube = Cube::from_cubie(cubie);
  745         assert_eq!(SymFilter::new(cube, cube.inverse()).is_active(), i < 13);
  746     }
  747     let filtered = Solver::new(corners(), h0());
  748     let mut plain = Solver::new(corners(), h0());
  749     plain.sym_start = false;
  750     // Recursive solver: lengths, and the nodes of each cube.
  751     let solve_all = |solver: &Solver, cubes: &[CubieCube]| -> Vec<(usize, u64)> {
  752         cubes
  753             .iter()
  754             .map(|cube| {
  755                 let (solution, stats) = solver.solve(cube);
  756                 let solution = solution.unwrap();
  757                 assert_eq!(cube.apply_moves(&solution), Some(CubieCube::SOLVED));
  758                 (solution.len(), stats.nodes)
  759             })
  760             .collect()
  761     };
  762     let lengths = |results: &[(usize, u64)]| results.iter().map(|r| r.0).collect::<Vec<_>>();
  763     let nodes = |results: &[(usize, u64)]| results.iter().map(|r| r.1).sum::<u64>();
  764     let want = solve_all(&plain, &cubes);
  765     let got = solve_all(&filtered, &cubes);
  766     assert_eq!(lengths(&want), known);
  767     assert_eq!(lengths(&got), known);
  768     assert!(
  769         nodes(&got) < nodes(&want),
  770         "{} vs {}",
  771         nodes(&got),
  772         nodes(&want)
  773     );
  774     // Stream lanes, with and without the filter.
  775     let lanes_filtered = lanes_lengths_and_nodes(&filtered, &cubes);
  776     let lanes_plain = lanes_lengths_and_nodes(&plain, &cubes);
  777     assert_eq!(lengths(&lanes_filtered), known, "lanes");
  778     assert_eq!(lengths(&lanes_plain), known, "lanes, no filter");
  779     assert!(nodes(&lanes_filtered) < nodes(&lanes_plain), "lanes");
  780     // Share mode, batch and one cube at a time.
  781     let counter = |counters: &str, name: &str| -> u64 {
  782         let fields: Vec<&str> = counters.split_whitespace().collect();
  783         let i = fields.iter().position(|&field| field == name).unwrap();
  784         fields[i + 1].parse().unwrap()
  785     };
  786     for max_active in [usize::MAX, 1] {
  787         let (share_filtered, counters) = share_lengths_and_nodes(&filtered, &cubes, max_active);
  788         assert_eq!(
  789             lengths(&share_filtered),
  790             known,
  791             "share {max_active}: {counters}"
  792         );
  793         // (The first pass found the cubes of lengths 12 to 14.)
  794         assert!(counter(&counters, "pass_solutions") > 0, "{counters}");
  795         let (share_plain, _) = share_lengths_and_nodes(&plain, &cubes, max_active);
  796         assert_eq!(
  797             lengths(&share_plain),
  798             known,
  799             "share {max_active}, no filter"
  800         );
  801         // (Share mode's counts vary from run to run, by a few percent here; the
  802         // filter takes about 30% off.)
  803         assert!(
  804             nodes(&share_filtered) < nodes(&share_plain),
  805             "share {max_active}"
  806         );
  807     }
  808     // Random states: never filtered, identical search.
  809     let randoms: Vec<CubieCube> = dataset("depth15.txt").into_iter().take(10).collect();
  810     assert_eq!(solve_all(&filtered, &randoms), solve_all(&plain, &randoms));
  811 }
