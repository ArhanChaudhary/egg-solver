    1 //! Symmetric-start filtering: for a cube with a symmetry, search one job of
    2 //! each class of equivalent job prefixes (nissy's task filter, nxopt's `-s`,
    3 //! Rokicki's "drop symmetry duplicates in the first levels").
    4 //!
    5 //! Gains (x-h48's 14 symmetric and antisymmetric test positions, `solve -c
    6 //! 1`, on main fe926ed): 22.8x at 1T (9.75 -> 0.43 s) and 7.9x at 48T
    7 //! (267 -> 34 ms), nearly all of it the superflip's (71x at 1T, 21x at
    8 //! 48T); the 7 positions with only a symmetry 1.87x at 1T (-47% nodes) and
    9 //! 1.30x at 48T. Stream mode (8 threads, on f8a4919): 28.9x solves/s,
   10 //! -96.6% nodes.
   11 //!
   12 //! A job is a canonical move prefix `p` of the solution, searched from
   13 //! `X·p`. A job is dropped if a lexicographically smaller job `q` (move
   14 //! indices; same length) reaches an equivalent position: `X·q` is a
   15 //! conjugate of `X·p` or of its inverse (one of 96), so both are equally far
   16 //! from solved. Two rules find such a `q`:
   17 //!
   18 //! * `keep`: for each symmetry `t` of `X` (`t X t⁻¹ = X`), `q = canon(t(p))`
   19 //!   (`X·t(p) = t (X·p) t⁻¹`; `canon` puts adjacent opposite-face moves in
   20 //!   the canonical order, U before D). Cheap: no cube is touched.
   21 //! * `dedup`: exact classes (the least of the 96 conjugates as a key) of a
   22 //!   whole level of jobs; keeps the smallest job of each class. This finds
   23 //!   the equivalences that come from an antisymmetry (`t X t⁻¹ = X⁻¹`) of a
   24 //!   cube that also has symmetries (the superflip, which commutes with every
   25 //!   move: `X·p` and `X·p⁻¹`), and accidental ones.
   26 //!
   27 //! Which cubes: `keep` runs for every cube with a symmetry, `dedup` only if
   28 //! it also has an antisymmetry. On x-h48's symmetric timing set, `dedup` of
   29 //! the 7 cubes with only a symmetry dropped nothing beyond `keep` (1656 of
   30 //! 3240 3-move jobs kept either way) and took ~0.4 ms per cube at 1T; for
   31 //! the 6 with only an antisymmetry neither rule drops anything (it relates
   32 //! `X·p` to positions with `p` on the inverse side, which are not jobs), so
   33 //! they are not filtered (filtered by `dedup`: ~0.75 ms per cube at 1T, and
   34 //! 1.5-3.2x their time at 48T `-c 1` in one screening run). For the
   35 //! superflip, `dedup` takes the 3-move jobs from 60 to 44 and the 4-move
   36 //! ones from 553 to 380.
   37 //!
   38 //! Why one optimal solution is still found: let `g` be the smallest job of
   39 //! its length that is the prefix of a canonical optimal solution. If `g` was
   40 //! dropped for `q < g`, then `q` followed by an optimal solution of `X·q` is
   41 //! optimal, and its canonical form starts with `q` or with `q`'s last move
   42 //! swapped for an earlier opposite-face one: a smaller prefix of an optimal
   43 //! solution, a contradiction. Each prefix of `g` is the smallest of its own
   44 //! length (a smaller one would extend to a smaller `g`), so filtering every
   45 //! extension level never drops it either. The search inside a job (both
   46 //! sides, premoves) is unchanged.
   47 //!
   48 //! Random states never have a symmetry: none of 7M positions (2000 random
   49 //! states and their prefixes up to 3 moves deep). The check at a cube's start
   50 //! costs 95 conjugations (~0.3 µs), and nothing else happens unless the cube
   51 //! has a symmetry.
   52

   56 /// Exact classes are computed for levels of at most this many jobs (after
   57 /// `keep`): ~0.2 µs per job at 1T (3240 jobs in ~0.7 ms).
   58 const DEDUP_MAX_JOBS: usize = 8192;

   60 /// The jobs to keep for one cube (see the module docs).
   61 #[derive(Clone, Copy)]
   62 pub struct SymFilter {
   63     /// The cube's symmetries other than the identity: the first
   64     /// `n_symmetries` of `symmetries` (none: every job is kept and nothing
   65     /// else is read).
   66     n_symmetries: u8,
   67     symmetries: [u8; 47],
   68     /// The cube also has an antisymmetry: `dedup` runs.
   69     antisymmetric: bool,
   70 }
   71 
   72 impl SymFilter {
   73     /// Keeps every job.
   74     pub const NONE: SymFilter = SymFilter {
   75         n_symmetries: 0,
   76         symmetries: [0; 47],
   77         antisymmetric: false,
   78     };
   79 
   80     /// The filter of `cube` (with inverse `inverse`).
   81     #[must_use]
   82     pub fn new(cube: Cube, inverse: Cube) -> SymFilter {
   83         let mut filter = SymFilter::NONE;
   84         let mut antisymmetric = false;
   85         for sym in 0..48u8 {
   86             let conjugate = cube.transform(usize::from(sym));
   87             if sym > 0 && conjugate.equals(cube) {
   88                 filter.symmetries[usize::from(filter.n_symmetries)] = sym;
   89                 filter.n_symmetries += 1;
   90             }
   91             antisymmetric |= conjugate.equals(inverse);
   92         }
   93         filter.antisymmetric = antisymmetric;
   94         filter
   95     }
   96 
   97     /// Whether the filter drops anything (the cube has a symmetry).
   98     #[must_use]
   99     pub fn is_active(&self) -> bool {
  100         self.n_symmetries > 0
  101     }
  102 
  103     /// Whether `dedup` runs (the cube also has an antisymmetry). Without it,
  104     /// whether a job is kept depends on its moves alone (`keep`).
  105     #[must_use]
  106     pub fn dedups(&self) -> bool {
  107         self.is_active() && self.antisymmetric
  108     }
  109 
  110     /// Whether to keep the job with move prefix `moves` (nonempty,
  111     /// canonical, at most 4 moves): false if a symmetry of the cube maps it
  112     /// to a smaller one.
  113     #[must_use]
  114     pub fn keep(&self, moves: &[u8]) -> bool {
  115         if !self.is_active() {
  116             return true;
  117         }
  118         let conjugates = move_conjugates();
  119         for &sym in &self.symmetries[..usize::from(self.n_symmetries)] {
  120             let mut image = [0u8; 4];
  121             for (image_move, &mv) in image.iter_mut().zip(moves) {
  122                 *image_move = conjugates[usize::from(sym)][usize::from(mv)];
  123             }
  124             let image = &mut image[..moves.len()];
  125             // Canonical order of opposite-face pairs (no three consecutive
  126             // moves share an axis in a canonical sequence, so pairs are
  127             // disjoint: one pass).
  128             for i in 1..image.len() {
  129                 let (first_face, second_face) = (image[i - 1] / 3, image[i] / 3);
  130                 if first_face / 2 == second_face / 2 && first_face > second_face {
  131                     image.swap(i - 1, i);
  132                 }
  133             }
  134             if *image < *moves {
  135                 return false;
  136             }
  137         }
  138         true
  139     }
  140 
  141     /// Keep only the smallest job (by `moves`) of each class of equivalent
  142     /// positions (`position`) among `jobs`, a whole level (same length), in
  143     /// order. Does nothing unless the cube has a symmetry and an
  144     /// antisymmetry (`dedups`), or above `DEDUP_MAX_JOBS` jobs.
  145     pub fn dedup<T>(
  146         &self,
  147         jobs: &mut Vec<T>,
  148         moves: impl Fn(&T) -> &[u8],
  149         position: impl Fn(&T) -> Cube,
  150     ) {
  151         if !self.dedups() || jobs.len() > DEDUP_MAX_JOBS {
  152             return;
  153         }
  154         let keys: Vec<(u64, u128)> = jobs.iter().map(|job| class_key(position(job))).collect();
  155         // The smallest job of each class.
  156         let mut smallest: HashMap<(u64, u128), usize> = HashMap::with_capacity(jobs.len());
  157         for (i, key) in keys.iter().enumerate() {
  158             smallest
  159                 .entry(*key)
  160                 .and_modify(|best| {
  161                     if moves(&jobs[i]) < moves(&jobs[*best]) {
  162                         *best = i;
  163                     }
  164                 })
  165                 .or_insert(i);
  166         }
  167         let mut i = 0;
  168         jobs.retain(|_| {
  169             let keep = smallest[&keys[i]] == i;
  170             i += 1;
  171             keep
  172         });
  173     }
  174 }

  176 /// `conjugates[t][m]`: the move `t m t⁻¹` (`Cube::transform` by `t` of
  177 /// move `m`): each symmetry's move map, from build.rs.
  178 fn move_conjugates() -> &'static [[u8; 18]; 48] {
  179     &crate::simd::SYM_MOVE_MAP
  180 }
  181 
  182 /// The corner and edge bytes of a cube.
  183 fn position_key(cube: Cube) -> (u64, u128) {
  184     let bytes = cube.to_bytes();
  185     let mut corners = [0u8; 8];
  186     for (x, &byte) in corners.iter_mut().zip(&bytes) {
  187         *x = byte;
  188     }
  189     let mut edges = [0u8; 16];
  190     for (x, &byte) in edges.iter_mut().zip(bytes.iter().skip(16).take(12)) {
  191         *x = byte;
  192     }
  193     (u64::from_le_bytes(corners), u128::from_le_bytes(edges))
  194 }
  195 
  196 /// The same key for all cubes equivalent to `cube`: the least over its 48
  197 /// conjugates and those of its inverse.
  198 fn class_key(cube: Cube) -> (u64, u128) {
  199     let inverse = cube.inverse();
  200     (0..48)
  201         .flat_map(|sym| {
  202             [
  203                 position_key(cube.transform(sym)),
  204                 position_key(inverse.transform(sym)),
  205             ]
  206         })
  207         .fold((u64::MAX, u128::MAX), std::cmp::min)
  208 }

  223     /// Conjugation is a homomorphism:
  224     /// `t (X·m) t⁻¹ = (t X t⁻¹)·conjugates[t][m]`.
  225     #[test]
  226     fn move_conjugates_compose() {
  227         let conjugates = move_conjugates();
  228         let mut rng = Rng(5);
  229         for _ in 0..if cfg!(miri) { 2 } else { 50 } {
  230             let x = Cube::from_cubie(&CubieCube::random(&mut rng));
  231             for (sym, row) in conjugates.iter().enumerate() {
  232                 for (mv, &image_move) in row.iter().enumerate() {
  233                     let lhs = x.apply_move(Move::ALL[mv]).transform(sym);
  234                     let rhs = x
  235                         .transform(sym)
  236                         .apply_move(Move::ALL[usize::from(image_move)]);
  237                     assert!(lhs.equals(rhs), "sym {sym} move {mv}");
  238                 }
  239             }
  240         }
  241     }

  252     /// Jobs kept per length (1..=4) for cube `x`, extending level by level
  253     /// as the solvers do.
  254     fn kept_per_level(x: Cube) -> [usize; 5] {
  255         let filter = SymFilter::new(x, x.inverse());
  256         let mut kept = [0usize; 5];
  257         let mut level: Vec<Vec<u8>> = vec![vec![]];
  258         for kept_of_length in kept.iter_mut().skip(1) {
  259             let mut next = Vec::new();
  260             for prefix in &level {
  261                 for mv in 0..18u8 {
  262                     if let Some(&last) = prefix.last() {
  263                         let (last_face, face) = (last / 3, mv / 3);
  264                         if last_face == face || (last_face % 2 == 1 && face == last_face - 1) {
  265                             continue;
  266                         }
  267                     }
  268                     let mut child = prefix.clone();
  269                     child.push(mv);
  270                     if filter.keep(&child) {
  271                         next.push(child);
  272                     }
  273                 }
  274             }
  275             filter.dedup(
  276                 &mut next,
  277                 |prefix| prefix,
  278                 |prefix| {
  279                     prefix
  280                         .iter()
  281                         .fold(x, |cube, &mv| cube.apply_move(Move::ALL[usize::from(mv)]))
  282                 },
  283             );
  284             *kept_of_length = next.len();
  285             level = next;
  286         }
  287         kept
  288     }

  290     /// The superflip has all 48 symmetries and is its own inverse (and
  291     /// commutes with every move): one quarter turn and one half turn are
  292     /// left as first moves, and far fewer than nissy's 1038 of the 43254
  293     /// 4-move prefixes.
  294     #[test]
  295     fn superflip_jobs() {
  296         let x = cube("U R2 F B R B2 R U2 L B2 R U' D' R2 F R' L B2 U2 F2");
  297         let filter = SymFilter::new(x, x.inverse());
  298         assert!(filter.dedups() && filter.n_symmetries == 47);
  299         let kept = kept_per_level(x);
  300         assert_eq!(kept[1], 2, "{kept:?}");
  301         assert!(kept[4] <= 1038, "{kept:?}");
  302     }

  304     /// Only an antisymmetry: no filter (no symmetry for `keep`, and `dedup`
  305     /// would find nothing: see the module docs).
  306     #[test]
  307     fn antisymmetric_cube() {
  308         // A · t(A)⁻¹ is antisymmetric for an involution t (here y2).
  309         let a = cube("R U2 F' L D B2 R'");
  310         let sym = (0..48)
  311             .find(|&sym| {
  312                 let u = Cube::solved().apply_move(Move::U);
  313                 u.transform(sym).equals(u)
  314                     && Cube::solved()
  315                         .apply_move(Move::R)
  316                         .transform(sym)
  317                         .equals(Cube::solved().apply_move(Move::L))
  318             })
  319             .unwrap();
  320         let x = a.compose(a.transform(sym).inverse());
  321         assert!(x.transform(sym).equals(x.inverse()));
  322         let filter = SymFilter::new(x, x.inverse());
  323         assert!(filter.antisymmetric && filter.n_symmetries == 0);
  324         assert!(!filter.is_active() && !filter.dedups());
  325         assert!(filter.keep(&[3, 6, 0]));
  326         assert_eq!(kept_per_level(x), [0, 18, 243, 3240, 43254]);
  327     }
