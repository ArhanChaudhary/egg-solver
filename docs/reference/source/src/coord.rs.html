    1 //! Coordinates: the numbers that index the pruning tables.
    2 //!
    3 //! A pruning table stores, for each value of a coordinate (a function of
    4 //! the cube state), a lower bound on the number of moves to solve any cube
    5 //! with that value. The H48 coordinates (from nissy and cube48opt):
    6 //!
    7 //! * cocsep: corner orientation (3^7) x corner tetrad separation: which 4
    8 //!   of the 8 positions hold the corners `UFL UBR DFR DBL` (C(8,4) = 70;
    9 //!   half turns keep these 4 among themselves). Reduced by the 48
   10 //!   symmetries to 3393 classes.
   11 //! * msplit: which 4 of the 8 non-E-slice positions hold S-slice edges (70).
   12 //! * eo: edge orientation (2^11; the 12th flip follows from the others).
   13 //! * eslot: which 4 positions hold E-slice edges (C(12,4) = 495 values),
   14 //!   packed into 512 slots so that slots 64k and 64k+1 are free for a 4-bit
   15 //!   group minimum.
   16 //!
   17 //! The full h11 key (computed after conjugating the cube by the symmetry
   18 //! that maps its corners to the class representative) is
   19 //! `X = ((class * 70 + msplit) << 20) | (eo11 << 9) | eslot`. The table
   20 //! `hk` (k = 0..=11) keeps only the top `k` eo bits: its index is
   21 //! `((X >> (11 - k)) & !0x1FF) | (X & 0x1FF)` (cube48opt's layout).
   22 //!
   23 //! The hot path (`cocsep_x3`, `index_sym`) reads the cube's bytes with
   24 //! `pext` (gather the bits under a mask) and `movemask` (gather the top bit
   25 //! of each byte), then looks the parts up in small tables.

   55 /// Binomial coefficient `n` choose `k` (for small `n`: no overflow checks).
   56 const fn binomial(n: u32, k: u32) -> u32 {
   57     if k > n {
   58         return 0;
   59     }
   60     let mut result = 1;
   61     let mut i = 0;
   62     while i < k {
   63         // (Exact: a product of i + 1 consecutive numbers is divisible by (i + 1)!.)
   64         result = result * (n - i) / (i + 1);
   65         i += 1;
   66     }
   67     result
   68 }

   70 /// Colexicographic rank of a bit mask among the masks with as many set
   71 /// bits: the sum of `binomial(p, j)` over its `j`-th set bit `p`.
   72 const fn colex_rank(mask: u32) -> u32 {
   73     let mut rank = 0;
   74     let mut ones = 0;
   75     let mut bit = 0;
   76     while bit < 32 {
   77         if mask >> bit & 1 == 1 {
   78             ones += 1;
   79             rank += binomial(bit, ones);
   80         }
   81         bit += 1;
   82     }
   83     rank
   84 }

   86 /// E-slice mask -> slot, and slot -> mask: masks in descending order fill
   87 /// slots 2..510, skipping slots 64k and 64k+1 (reserved for the group
   88 /// minimum).
   89 const fn eslot_tables() -> ([u16; 4096], [u16; 512]) {
   90     let mut slot_of = [0xFFFFu16; 4096];
   91     let mut mask_of = [0u16; 512];
   92     let mut slot: u16 = 2;
   93     let mut mask: u16 = 4096;
   94     while mask > 0 {
   95         mask -= 1;
   96         if mask.count_ones() == 4 {
   97             slot_of[mask as usize] = slot;
   98             mask_of[slot as usize] = mask;
   99             if (slot + 1).is_multiple_of(64) {
  100                 slot += 3;
  101             } else {
  102                 slot += 1;
  103             }
  104         }
  105     }
  106     (slot_of, mask_of)
  107 }
  108 const ESLOT_TABLES: ([u16; 4096], [u16; 512]) = eslot_tables();

  110 /// Rank (0..70) of each 8-bit mask with 4 set bits; 0xFF for other masks.
  111 #[expect(
  112     clippy::cast_possible_truncation,
  113     reason = "ranks of 8-bit masks are below 70 (no `try_from` in const fns)"
  114 )]
  115 const fn rank8_table() -> [u8; 256] {
  116     let mut table = [0u8; 256];
  117     let mut mask: u8 = 0;
  118     loop {
  119         table[mask as usize] = if mask.count_ones() == 4 {
  120             colex_rank(mask as u32) as u8
  121         } else {
  122             0xFF
  123         };
  124         if mask == u8::MAX {
  125             return table;
  126         }
  127         mask += 1;
  128     }
  129 }
  130 
  131 /// The mask of each rank of `rank8_table`.
  132 const fn unrank8_table() -> [u8; 70] {
  133     let mut table = [0u8; 70];
  134     let mut mask: u8 = 0;
  135     loop {
  136         if mask.count_ones() == 4 {
  137             table[colex_rank(mask as u32) as usize] = mask;
  138         }
  139         if mask == u8::MAX {
  140             return table;
  141         }
  142         mask += 1;
  143     }
  144 }

  146 /// Base-3 value of `n` corner orientations given as 2-bit fields (as
  147 /// gathered by `pext`), times `scale`.
  148 const fn co_rank_table(scale: u16, n: usize) -> [u16; 256] {
  149     let mut table = [0u16; 256];
  150     let mut fields: u16 = 0;
  151     while fields < (1 << (2 * n)) {
  152         let mut rank = 0;
  153         let mut power = 1;
  154         let mut i = 0;
  155         while i < n {
  156             rank += ((fields >> (2 * i)) & 3) * power;
  157             power *= 3;
  158             i += 1;
  159         }
  160         table[fields as usize] = rank * scale;
  161         fields += 1;
  162     }
  163     table
  164 }

  179 /// `co_rank_table` pre-multiplied by `N_CSEP` (saves a multiply on the
  180 /// cube -> cocsep data dependency chain; part of m1-micro's +2.2% at 48T).
  181 #[expect(
  182     clippy::cast_possible_truncation,
  183     reason = "`N_CSEP` is 70 (no `try_from` in const fns)"
  184 )]
  185 const fn co_rank_table_x70(scale: u32, n: usize) -> [u32; 256] {
  186     let ranks = co_rank_table(1, n);
  187     let mut table = [0u32; 256];
  188     let mut i = 0;
  189     while i < 256 {
  190         table[i] = ranks[i] as u32 * scale * N_CSEP as u32;
  191         i += 1;
  192     }
  193     table
  194 }

  209 /// The corner bits of a cube: the orientations of corners 0..7 (2 bits
  210 /// each, 14 bits) and the tetrad mask (8 bits), by two `pext`s.
  211 #[cfg(backend = "x86")]
  212 #[inline(always)]
  213 fn corner_fields(cube: Cube) -> (u64, u64) {
  214     let corners = cube.corners_u64();
  215     let (co_fields, tetrads): (u64, u64);
  216     unsafe {
  217         let tabs = &crate::simd::HOT_TABLES;
  218         // (Miri cannot run inline asm: the same pexts as intrinsics.)
  219         #[cfg(miri)]
  220         {
  221             co_fields = _pext_u64(corners, tabs.co_pext_mask);
  222             tetrads = _pext_u64(corners, tabs.tetrad_pext_mask);
  223         }
  224         // (The masks are memory operands: no 64-bit immediates to load;
  225         // +1.7% when added, part of m4-micro's +7.7% at 48T.)
  226         #[cfg(not(miri))]
  227         std::arch::asm!(
  228             "pext {co2}, {x}, qword ptr [{t} + {oc}]",
  229             "pext {sep}, {x}, qword ptr [{t} + {os}]",
  230             co2 = out(reg) co_fields,
  231             sep = lateout(reg) tetrads,
  232             x = in(reg) corners,
  233             t = in(reg) tabs,
  234             oc = const std::mem::offset_of!(crate::simd::HotTables, co_pext_mask),
  235             os = const std::mem::offset_of!(crate::simd::HotTables, tetrad_pext_mask),
  236             options(pure, readonly, nostack, preserves_flags)
  237         );
  238         // (A range hint, not an index here: the lookups of `cocsep` and
  239         // `cocsep_x3` are unchecked already.)
  240         std::hint::assert_unchecked(co_fields < 1 << 14 && tetrads < 1 << 8);
  241     }
  242     (co_fields, tetrads)
  243 }

  246 #[must_use]
  247 #[inline(always)]

  249     clippy::cast_possible_truncation,

  281 /// Edge masks: (E-slice positions, S-slice positions, flipped edges), 12
  282 /// bits each.
  283 #[cfg(backend = "x86")]
  284 #[must_use]
  285 #[inline(always)]
  286 pub fn edge_masks(cube: Cube) -> (u32, u32, u32) {
  287     // (Shifting a bit of each byte into the top bit lets `movemask` gather
  288     // it; edges are bytes 16..28.)
  289     let mask = |bits: i32| (bits.cast_unsigned() >> 16) & 0xFFF;
  290     unsafe {
  291         (
  292             mask(_mm256_movemask_epi8(_mm256_slli_epi16(cube.0, 4))),
  293             mask(_mm256_movemask_epi8(_mm256_slli_epi16(cube.0, 5))),
  294             mask(_mm256_movemask_epi8(_mm256_slli_epi16(cube.0, 3))),
  295         )
  296     }
  297 }

  299 /// Edge part of the h11 key of an (already transformed) cube:
  300 /// `(msplit << 20) | (eo11 << 9) | eslot`.
  301 #[must_use]
  302 #[inline(always)]
  303 #[cfg_attr(not(backend = "x86"), allow(unused_unsafe))]
  304 pub fn edge_key(cube: Cube) -> u64 {
  305     let (e_slice, s_slice, flips) = edge_masks(cube);
  306     unsafe {
  307         // (msplit ranks the S-slice positions among the non-E positions.)
  308         #[cfg(backend = "x86")]
  309         let s_among_non_e = _pext_u32(s_slice, !e_slice & 0xFFF);
  310         #[cfg(not(backend = "x86"))]
  311         let s_among_non_e = pext12(s_slice, !e_slice & 0xFFF);
  312         let msplit = u64::from(RANK8[s_among_non_e as usize]);
  313         let eslot = u64::from(ESLOT[e_slice as usize]);
  314         (msplit << 20) | (u64::from(flips >> 1) << 9) | eslot
  315     }
  316 }

  318 /// Edge part of the h11 key of `cube` conjugated by symmetry `sym`; equals
  319 /// `edge_key(cube.transform_edges(sym))`. Only the edge classes and
  320 /// orientations of the conjugate are built: one shuffle classifies each
  321 /// piece `p` as `T[p]` (E/S flags, eo of T), xor adds the cube's eo and
  322 /// Tinv's (by source position: one three-way xor), and a second shuffle
  323 /// applies the position permutation `Tinv`. (Instead of a full
  324 /// conjugation: +5.5% at 48T when added, edge-conj.)
  325 #[must_use]
  326 #[inline(always)]
  327 pub fn edge_key_sym(cube: Cube, sym: usize) -> Option<u64> {
  328     let (msplit, eo11, eslot) = edge_parts_sym(cube, sym)?;
  329     Some((msplit << 20) | (eo11 << 9) | eslot)
  330 }

  332 /// (msplit, eo11, eslot) of `cube` conjugated by `sym` (see `edge_key_sym`).
  333 #[must_use]
  334 #[inline(always)]
  335 pub fn edge_parts_sym(cube: Cube, sym: usize) -> Option<(u64, u64, u64)> {
  336     if sym >= 48 {
  337         return None;
  338     }
  339     let (msplit, eo_bits, eslot) = edge_parts_sym_raw(cube, sym);
  340     Some((msplit, u64::from(eo_bits >> 17), eslot))
  341 }

  343 /// `edge_parts_sym` with the eo part unshifted: the eo of edges 0..11 in
  344 /// bits 16..28 (all else zero).
  345 #[cfg(backend = "x86")]
  346 #[expect(
  347     clippy::cast_possible_truncation,
  348     reason = "x86-64: the u64 pext result indexes a 256-entry table"
  349 )]
  350 #[inline(always)]
  351 fn edge_parts_sym_raw(cube: Cube, sym: usize) -> (u64, u32, u64) {
  352     unsafe {
  353         let tabs = &crate::simd::HOT_TABLES;
  354         let entry = tabs.sym_edges.get_unchecked(sym);
  355         let classify = crate::simd::load(&entry.classes);
  356         let positions = crate::simd::load(&entry.perm);
  357         // Tinv's eo (by source position) is folded into the xor that adds
  358         // the cube's eo: one vpternlog, and no xor after the position
  359         // shuffle (part of m7's µop stack: +1.5% at 1T averaged over code
  360         // offsets, even at 48T).
  361         let position_eo = crate::simd::load(&entry.perm_eo);
  362         let classes = _mm256_xor_si256(
  363             _mm256_xor_si256(_mm256_shuffle_epi8(classify, cube.0), cube.0),
  364             position_eo,
  365         );
  366         let conj = _mm256_shuffle_epi8(classes, positions);
  367         // Masks of edge positions 0..12, in bits 16..28 (all else zero).
  368         let e_mask = _mm256_movemask_epi8(conj).cast_unsigned();
  369         // asm: LLVM would turn movemask(conj + conj) into vptestmb + kmovd,
  370         // which is ~1.5 cycles longer on Zen 5 (this mask feeds the pext;
  371         // part of m1-micro's +2.2% at 48T).
  372         let doubled = _mm256_add_epi8(conj, conj);
  373         // (A 64-bit output: the 32-bit write zero-extends, and the 64-bit
  374         // pext below then needs no zero extension of `s_mask`.)
  375         let s_mask: u64;
  376         #[cfg(miri)]
  377         {
  378             s_mask = u64::from(_mm256_movemask_epi8(doubled).cast_unsigned());
  379         }
  380         #[cfg(not(miri))]
  381         std::arch::asm!("vpmovmskb {s:e}, {zz}", s = out(reg) s_mask, zz = in(ymm_reg) doubled,
  382             options(pure, nomem, nostack, preserves_flags));
  383         let eo_bits = _mm256_movemask_epi8(_mm256_slli_epi16(conj, 3)).cast_unsigned();
  384         // The non-E positions (`e_mask` is within bits 16..28): the mask of
  385         // the pext, and (shifted) the index of the complemented eslot
  386         // table, which saves a copy of `e_mask`.
  387         let non_e = e_mask ^ 0x0FFF_0000;
  388         // (64-bit pext: no zero extension of the index.)
  389         let msplit = u64::from(
  390             *tabs
  391                 .rank8
  392                 .get_unchecked(_pext_u64(s_mask, u64::from(non_e)) as usize),
  393         );
  394         let eslot = u64::from(*tabs.eslot_complement.get_unchecked((non_e >> 16) as usize));
  395         (msplit, eo_bits, eslot)
  396     }
  397 }

  399 /// `index_from_parts(class, edge_parts_sym(cube, sym), k)`, with the shifts
  400 /// by `k` done as a multiply by `row_mul` = `1 << (k + 9)` and a pext by
  401 /// `eo_mask` = the top `k` eo bits, `((1 << k) - 1) << (28 - k)` (memory
  402 /// operands: no variable shift counts to keep in registers). (The index
  403 /// straight from the parts: part of packed-cocsep's +2.8% at 48T; the
  404 /// multiply and pext: part of m6-micro's index patch.)
  405 #[must_use]
  406 #[inline(always)]
  407 pub(crate) fn index_sym(cube: Cube, sym: usize, class: usize, row_mul: u64, eo_mask: u32) -> u64 {
  408     let (msplit, eo_bits, eslot) = edge_parts_sym_raw(cube, sym);
  409     let row = class as u64 * N_MSPLIT as u64 + msplit;
  410     #[cfg(backend = "x86")]
  411     let eo = u64::from(unsafe { _pext_u32(eo_bits, eo_mask) });
  412     #[cfg(not(backend = "x86"))]
  413     let eo = u64::from(pext_contiguous(eo_bits, eo_mask));
  414     (row * row_mul) | (eo << 9) | eslot
  415 }

  417 /// hk index of class `class` and edge parts `(msplit, eo11, eslot)`; equals
  418 /// `reduce_key((class * 70 + msplit) << 20 | eo11 << 9 | eslot, k)` but with
  419 /// a shorter dependency chain from msplit.
  420 #[must_use]
  421 #[inline(always)]
  422 pub fn index_from_parts(class: usize, parts: (u64, u64, u64), k: u32) -> u64 {
  423     let (msplit, eo11, eslot) = parts;
  424     let row = class as u64 * N_MSPLIT as u64 + msplit;
  425     (row << (k + 9)) | ((eo11 >> (11 - k)) << 9) | eslot
  426 }

  435 /// Reduce an h11 key to an hk index (drop the low `11 - k` eo bits).
  436 #[must_use]
  437 #[inline(always)]
  438 pub fn reduce_key(key: u64, k: u32) -> u64 {
  439     ((key >> (11 - k)) & !0x1FF) | (key & 0x1FF)

  442 /// Inverse of `reduce_key` for k in {0, 11}: (class, h11 edge key).
  443 #[must_use]
  444 #[inline(always)]
  445 pub fn split_index(index: u64, k: u32) -> (usize, u64) {
  446     let eslot = index & 0x1FF;
  447     let rest = index >> 9;
  448     let eo = (rest & u64::MAX.unbounded_shr(64 - k)) << (11 - k);
  449     let row = rest >> k;
  450     let class = (row / N_MSPLIT as u64) as usize;
  451     let msplit = row % N_MSPLIT as u64;
  452     (class, (msplit << 20) | (eo << 9) | eslot)
  453 }

  466 /// Internal inverse for an existing coordinate (a real slot, msplit < 70).
  467 pub(crate) fn cube_from_edge_key_internal(corners: Cube, edge_key: u64) -> Cube {
  468     let eslot = (edge_key & 0x1FF) as usize;
  469     let eo11 = ((edge_key >> 9) & 0x7FF) as u32;
  470     let msplit = (edge_key >> 20) as usize;
  471     let e_slice = u32::from(SLOT_EMASK[eslot]);
  472     #[cfg(backend = "x86")]
  473     let s_slice = unsafe { _pdep_u32(u32::from(UNRANK8[msplit]), !e_slice & 0xFFF) };
  474     #[cfg(not(backend = "x86"))]
  475     let s_slice = pdep12(u32::from(UNRANK8[msplit]), !e_slice & 0xFFF);
  476     // (The 12th flip makes the number of flips even.)
  477     let eo12 = (eo11 << 1) | (eo11.count_ones() & 1);
  478     let mut bytes = corners.to_bytes();
  479     // Next piece of each slice: E 8..12, S 4..8, M 0..4.
  480     let (mut next_e, mut next_s, mut next_m) = (8u8, 4u8, 0u8);
  481     for position in 0..12 {
  482         let next = if e_slice >> position & 1 == 1 {
  483             &mut next_e
  484         } else if s_slice >> position & 1 == 1 {
  485             &mut next_s
  486         } else {
  487             &mut next_m
  488         };
  489         let piece = *next;
  490         *next += 1;
  491         let flipped = u8::from(eo12 >> position & 1 == 1);
  492         bytes[16 + position] = piece | (flipped << 4);
  493     }
  494     bytes[8..16].fill(0);
  495     bytes[28..32].fill(0);
  496     Cube::from_bytes_internal(bytes)
  497 }
