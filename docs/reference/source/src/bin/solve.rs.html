    1 //! Solve scrambles read from stdin (one per line, e.g. `R3 B1 F2 L2 U3`, as
    2 //! in `cube_solver_test`) optimally with the work-sharing lanes
    3 //! (`lanes::share`).
    4 //!
    5 //! ```text
    6 //! Usage: solve [OPTIONS] <TABLE> < scrambles
    7 //!   -t  threads (default: all cores)
    8 //!   -l  interleaved lanes per thread (default: from `<table>.tune` if
    9 //!       `bench <table> tune` made one, else `tables::file::untuned_lanes`)
   10 //!   -c  cubes in progress at once: 1 solves one cube at a time with all
   11 //!       threads (latency); 0 (default) solves the whole input as a batch
   12 //!   -n  solve only the first `count` scrambles

   14 //! (`solve --help` has all options.)
   15 //!
   16 //! Output, one line per cube in completion order, then a summary:
   17 //!
   18 //! ```text
   19 //! solved <index> <length> <nodes> <seconds> <solution>
   20 //! total <cubes> cubes wall <seconds> wall/cube <seconds> nodes/cube <n> ...

   22 //!
   23 //! `seconds` is the cube's time from its start to its solution (with -c 1,
   24 //! its latency). Every solution is verified; a wrong one is reported as
   25 //! `BAD <index>` and makes the exit status 1. With `--json`, the same as
   26 //! JSON lines (`Line`).
   27 //!

   29 //! can (what fast.sh did): the largest table found, all threads on the one

   33 //! later runs attach at once (`--shm`; README.md "Shared memory"), `solve

   65 // `share::Worker::rounds`'s section starts on a page boundary, `SHARE_LINE`
   66 // bytes before the library's 64-byte pad (a binary's module asm comes before
   67 // the library's), so the round loop starts at page offset 0x310, as on main,
   68 // whatever code or data is added before it (`.text` starts after the
   69 // unwind tables, which grow with every function). Only in `solve`: `bench`
   70 // keeps main's layout (s9).
   71 #[cfg(all(target_arch = "x86_64", target_os = "linux"))]
   72 const SHARE_LINE: usize = 0x300;
   73 #[cfg(all(target_arch = "x86_64", target_os = "linux"))]
   74 core::arch::global_asm!(
   75     ".pushsection .text.hot_share,\"ax\",@progbits",
   76     ".p2align 12",
   77     ".skip {pad}, 0xcc",
   78     ".popsection",
   79     pad = const SHARE_LINE,
   80 );

   82 /// `share::solve_all` with `Pool::run_pinned`: each worker's state and stack
   83 /// frames at fixed offsets within a page (main's offsets, so -c 1 is
   84 /// unchanged; see `Pool::run_pinned`).
   85 fn solve_all_pinned(
   86     solver: &Solver,
   87     cubes: &(dyn Fn(usize) -> Option<CubieCube> + Sync),
   88     threads: usize,
   89     lanes: usize,
   90     max_active: usize,
   91     done: &(dyn Fn(usize, Option<Vec<u8>>, Stats, f64) + Sync),
   92 ) -> anyhow::Result<()> {
   93     let pool = Pool::new(cubes, max_active, threads);
   94     let stop = AtomicBool::new(false);
   95     std::thread::scope(|scope| {
   96         let mut spawned = Vec::with_capacity(threads.max(1));
   97         let mut result = Ok(());
   98         for _ in 0..threads.max(1) {
   99             let thread = std::thread::Builder::new()
  100                 .spawn_scoped(scope, || pool.run_pinned(solver, lanes, &stop, done));
  101             match thread {
  102                 Ok(thread) => spawned.push(thread),
  103                 Err(error) => {
  104                     stop.store(true, Ordering::Relaxed);
  105                     result = Err(anyhow::Error::from(error).context("cannot start a thread"));
  106                     break;
  107                 }
  108             }
  109         }
  110         let mut all_ok = true;
  111         for thread in spawned {
  112             if thread.join().is_err() {
  113                 stop.store(true, Ordering::Relaxed);
  114                 all_ok = false;
  115             }
  116         }
  117         result?;
  118         anyhow::ensure!(all_ok, "a thread panicked");
  119         Ok(())
  120     })
  121 }

  268 fn main() -> ExitCode {
  269     let args = Args::parse();
  270     if let Some(shell) = args.completions {
  271         cli::print_completions::<Args>(shell);
  272         return ExitCode::SUCCESS;
  273     }
  274     args.verbosity.init_logging();
  275     let result = if let Some(path) = &args.shm_load {
  276         // SAFETY: Mapped/shared CLI inputs are required to remain unchanged while loaded (README contract).
  277         unsafe { args.load.shm_load(path) }
  278     } else if let Some(path) = &args.shm_remove {
  279         // SAFETY: the table segment follows the documented storage contract.
  280         unsafe { args.load.shm_remove(path) }
  281     } else if args.fast {
  282         run_fast(&args)
  283     } else {
  284         run(&args)
  285     };
  286     cli::exit_status("solve", result)
  287 }

  289 /// The cubes of the scrambles on stdin, one per line (blank lines
  290 /// skipped), at most `count`.
  291 fn read_scrambles(count: usize) -> anyhow::Result<Vec<CubieCube>> {
  292     let mut cubes = Vec::new();
  293     for (n, line) in std::io::stdin().lock().lines().enumerate() {
  294         let line = line.context("cannot read stdin")?;
  295         if line.trim().is_empty() {
  296             continue;
  297         }
  298         if cubes.len() >= count {
  299             break;
  300         }
  301         let cube = parse_moves(&line)
  302             .and_then(|moves| CubieCube::SOLVED.apply_moves(&moves))
  303             .ok_or_else(|| BadInput(format!("line {}: cannot parse {line:?}", n + 1)))?;
  304         cubes.push(cube);
  305     }
  306     Ok(cubes)
  307 }

  309 #[expect(clippy::cast_precision_loss, reason = "statistics")]
  310 fn run(args: &Args) -> anyhow::Result<ExitCode> {
  311     let path = (args.table.as_deref())
  312         .or(args.table_flag.as_deref())
  313         .context("no table")?;
  314     let cubes = read_scrambles(args.count.unwrap_or(usize::MAX))?;
  315     let corners = Cocsep::generate()?;
  316     // SAFETY: Mapped/shared CLI inputs are required to remain unchanged while loaded (README contract).
  317     let table = unsafe { args.load.open(path) }?;
  318     let threads = args.threads;
  319     let lanes = cli::lanes(args.lanes, path, table.meta().k, threads);
  320     let concurrency = args.concurrency;
  321     log::info!(
  322         "{} cubes, at most {concurrency} at once (0: all)",
  323         cubes.len()
  324     );
  325     let solver = Solver::new(&corners, &table);
  326     let get = |i: usize| cubes.get(i).copied();
  327     let max_active = if concurrency == 0 {
  328         usize::MAX
  329     } else {
  330         concurrency
  331     };
  332     let json = args.json;
  333     let stdout = Mutex::new(std::io::stdout());
  334     let nodes = AtomicU64::new(0);
  335     let bad = AtomicBool::new(false);
  336     let n_done = AtomicUsize::new(0);
  337     let done = |i: usize, solution: Option<Vec<u8>>, stats: Stats, seconds: f64| {
  338         // (No solution, or one that does not solve the cube: BAD.)
  339         let solves_cube = |moves: &[u8]| {
  340             cubes.get(i).and_then(|cube| cube.apply_moves(moves)) == Some(CubieCube::SOLVED)
  341         };
  342         let ok = solution.as_deref().is_some_and(solves_cube);
  343         let solution = solution.unwrap_or_default();
  344         nodes.fetch_add(stats.nodes, Ordering::Relaxed);
  345         n_done.fetch_add(1, Ordering::Relaxed);
  346         let mut out = stdout.lock().unwrap_or_else(PoisonError::into_inner);
  347         if !ok {
  348             bad.store(true, Ordering::Relaxed);
  349         }
  350         if json {
  351             let line = Line::Solved {
  352                 index: i,
  353                 moves: format_moves(&solution),
  354                 length: solution.len(),
  355                 nodes: stats.nodes,
  356                 lookups: stats.lookups,
  357                 seconds,
  358                 ok,
  359             };
  360             let _ = serde_json::to_writer(&mut *out, &line);
  361             let _ = writeln!(out);
  362         } else {
  363             if !ok {
  364                 let _ = writeln!(out, "BAD {i}");
  365             }
  366             let _ = writeln!(
  367                 out,
  368                 "solved {i} {} {} {seconds:.6} {}",
  369                 solution.len(),
  370                 stats.nodes,
  371                 format_moves(&solution)
  372             );
  373         }
  374         let _ = out.flush();
  375     };
  376     let start = Instant::now();
  377     solve_all_pinned(&solver, &get, threads, lanes, max_active, &done)?;
  378     let wall = start.elapsed().as_secs_f64();
  379     let n = cubes.len();
  380     let reported = n_done.load(Ordering::Relaxed);
  381     anyhow::ensure!(reported == n, "{reported} of {n} cubes reported");
  382     let wall_per_cube = wall / n.max(1) as f64;
  383     let nodes_per_cube = nodes.load(Ordering::Relaxed) as f64 / n.max(1) as f64;
  384     if json {
  385         print_json(&Line::Total {
  386             cubes: n,
  387             wall,
  388             wall_per_cube,
  389             nodes_per_cube,
  390             threads,
  391             lanes,
  392             concurrency,
  393         })?;
  394     } else {
  395         println!(
  396             "total {n} cubes wall {wall:.6} wall/cube {wall_per_cube:.6} nodes/cube {nodes_per_cube:.0} threads {threads} lanes {lanes} concurrency {concurrency}"
  397         );
  398     }
  399     Ok(if bad.load(Ordering::Relaxed) {
  400         ExitCode::FAILURE
  401     } else {
  402         ExitCode::SUCCESS
  403     })
  404 }

  422 /// A random scramble of `length` moves: no face twice in a row, and no X Y X
  423 /// with Y opposite X (e.g. U D U, which is U2 D).
  424 fn random_scramble(rng: &mut Rng, length: usize) -> String {
  425     // (Opposite faces have the same index mod 3: U D, R L, F B.)
  426     const FACES: [char; 6] = ['U', 'R', 'F', 'D', 'L', 'B'];
  427     const SUFFIXES: [&str; 3] = ["", "2", "'"];
  428     let below = |rng: &mut Rng, n: usize| rng.below_usize(n);
  429     let (mut previous, mut before) = (None, None);
  430     let mut moves = Vec::with_capacity(length);
  431     for _ in 0..length {
  432         let face = loop {
  433             let face = below(rng, 6);
  434             let opposite_return =
  435                 previous.is_some_and(|p: usize| p % 3 == face % 3) && before == Some(face);
  436             if previous != Some(face) && !opposite_return {
  437                 break face;
  438             }
  439         };
  440         moves.push(format!("{}{}", FACES[face], SUFFIXES[below(rng, 3)]));
  441         (before, previous) = (previous, Some(face));
  442     }
  443     moves.join(" ")
  444 }

  446 /// A seed from the clock and the process id. (The monotonic clock: reading
  447 /// `SystemTime` would change the code of the worker threads' `Instant::now`
  448 /// calls, which then pass the clock id instead of having it folded in.)
  449 fn time_seed() -> u64 {
  450     let mut hasher = std::hash::DefaultHasher::new();
  451     std::time::Instant::now().hash(&mut hasher);
  452     std::process::id().hash(&mut hasher);
  453     hasher.finish()
  454 }

  472 /// `solve --fast`: one scramble as fast as this machine can (fast.sh, as a
  473 /// flag): the largest table, all threads on the one cube (`-c 1`), a
  474 /// warm-up cube first in the same process (threads, page tables and TLBs
  475 /// are hot when the scramble starts). Prints fast.sh's lines; the time is
  476 /// the scramble's own solve time.
  477 #[expect(clippy::cast_precision_loss, reason = "statistics")]
  478 fn run_fast(args: &Args) -> anyhow::Result<ExitCode> {
  479     // (A first argument that is a move, not a file, starts the scramble.)
  480     let mut words = args.scramble.clone();
  481     let mut table_arg = args.table.clone();
  482     if let Some(first) =
  483         table_arg.take_if(|first| !Path::new(first).is_file() && parse_moves(first).is_some())
  484     {
  485         words.insert(0, first);
  486     }
  487     let path = (table_arg.or_else(|| args.table_flag.clone()))
  488         .or_else(largest_table)
  489         .context("no table found; generate one, e.g.: target/release/gen 8 auto tables/h8")?;
  490     let scramble = if words.is_empty() {
  491         random_scramble(&mut Rng(time_seed()), 25)
  492     } else {
  493         words.join(" ")
  494     };
  495     let parse = |text: &str| {
  496         parse_moves(text)
  497             .and_then(|moves| CubieCube::SOLVED.apply_moves(&moves))
  498             .ok_or_else(|| BadInput(format!("cannot parse {text:?}")))
  499     };
  500     let mut cubes = Vec::new();
  501     let warmup = args.warmup.as_deref().unwrap_or(WARMUP);
  502     if !warmup.trim().is_empty() {
  503         cubes.push(parse(warmup)?);
  504     }
  505     cubes.push(parse(&scramble)?);
  506     let threads = args.threads;
  507     if !args.json {
  508         println!("table    {path}");
  509         println!("threads  {threads} (all on this one cube)");
  510         println!("scramble {scramble}");
  511     }
  512     let corners = Cocsep::generate()?;
  513     // SAFETY: Mapped/shared CLI inputs are required to remain unchanged while loaded (README contract).
  514     let table = unsafe { args.load.open(&path) }?;
  515     let lanes = cli::lanes(args.lanes, &path, table.meta().k, threads);
  516     let solver = Solver::new(&corners, &table);
  517     let get = |i: usize| cubes.get(i).copied();
  518     // (The scramble, after the warm-up cube if any.)
  519     let (last, &cube) = cubes
  520         .iter()
  521         .enumerate()
  522         .next_back()
  523         .context("no scramble")?;
  524     let result = Mutex::new(None);
  525     let done = |i: usize, solution: Option<Vec<u8>>, stats: Stats, seconds: f64| {
  526         if i == last {
  527             *result.lock().unwrap_or_else(PoisonError::into_inner) =
  528                 Some((solution, stats, seconds));
  529         }
  530     };
  531     solve_all_pinned(&solver, &get, threads, lanes, 1, &done)?;
  532     let result = result.into_inner().unwrap_or_else(PoisonError::into_inner);
  533     let (solution, stats, seconds) = result.context("not solved")?;
  534     let solution = solution.context("no solution within 20 moves: is the table damaged?")?;
  535     anyhow::ensure!(
  536         cube.apply_moves(&solution) == Some(CubieCube::SOLVED),
  537         "BAD: {} does not solve the scramble",
  538         format_moves(&solution)
  539     );
  540     // (The seconds of solve's line, as fast.sh read them.)
  541     let seconds: f64 = format!("{seconds:.6}").parse()?;
  542     let rate = if seconds > 0.0 {
  543         stats.nodes as f64 / seconds
  544     } else {
  545         0.0
  546     };
  547     if args.json {
  548         print_json(&FastReport {
  549             table: &path,
  550             threads,
  551             lanes,
  552             scramble: &scramble,
  553             solution: format_moves(&solution),
  554             length: solution.len(),
  555             nodes: stats.nodes,
  556             lookups: stats.lookups,
  557             seconds,
  558             ms: seconds * 1000.0,
  559             nodes_per_second: rate,
  560         })?;
  561     } else {
  562         println!("solution {}", format_moves(&solution));
  563         println!("length   {} moves (optimal, verified)", solution.len());
  564         println!(
  565             "time     {:.3} ms   ({:.1} M nodes, {:.0} M nodes/s)",
  566             seconds * 1000.0,
  567             stats.nodes as f64 / 1e6,
  568             rate / 1e6
  569         );
  570     }
  571     Ok(ExitCode::SUCCESS)
  572 }
