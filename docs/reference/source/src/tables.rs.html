    1 //! Pruning tables: the cocsep (corner) table and the big H48 table.
    2 //!
    3 //! A pruning table maps a coordinate of the cube to a lower bound on its
    4 //! distance to solved. IDA* (see `solver`) prunes a node when its bound
    5 //! exceeds the moves left. Two tables are used:
    6 //!
    7 //! * `Cocsep`: the exact distance of each cocsep coordinate (corners only),
    8 //!   plus its symmetry class, built at startup in a fraction of a second.
    9 //! * `H48Table`: the distance of the full H48 coordinate (corner class,
   10 //!   msplit, eo, eslot; see `coord`), 2 bits per entry, generated by `gen`
   11 //!   and loaded from a file (58 GiB for h11).

   26 /// The cocsep table: per cocsep coordinate, its symmetry class, the
   27 /// symmetry that maps it to the class representative ("sym to rep") and its
   28 /// depth (distance to solved).
   29 ///
   30 /// ```compile_fail,E0616
   31 /// fn corrupt(table: &mut egg_solver::tables::Cocsep) {
   32 ///     table.entries = Vec::new().into_boxed_slice();
   33 /// }
   34 /// ```
   35 pub struct Cocsep {
   36     /// `entries[cocsep] = class << 16 | sym_to_rep << 8 | depth` (see
   37     /// `entry_*`).
   38     pub(crate) entries: Box<[u32]>,
   39     /// Representative cube (corners only meaningful) per class.
   40     pub(crate) representatives: Box<[Cube]>,
   41     /// Bitmask of the symmetries fixing the representative's cocsep, per
   42     /// class.
   43     pub(crate) self_symmetries: Box<[u64]>,
   44     /// Depth distribution of the 153090 cocsep coordinates.
   45     pub(crate) distribution: Box<[u64]>,
   46     /// Hot-path copy of `entries` packed into 3 bytes per entry (25% less
   47     /// L2 footprint): `class << 10 | sym_to_rep << 4 | depth` (+1 pad byte),
   48     /// on a transparent huge page (one TLB entry instead of ~113). (+2.8% at
   49     /// 48T when added, with the direct index, packed-cocsep.)
   50     packed: Memory,
   51 }

  121     /// Build the table: a breadth-first search (BFS) over the cocsep
  122     /// coordinate, then the symmetry classes.
  123     ///
  124     /// # Errors
  125     /// If the BFS does not find the expected number of coordinates or
  126     /// classes, or depths above 15 (a bug in the cube model), or the 2 MiB
  127     /// of packed entries cannot be allocated.
  128     pub fn generate() -> Result<Cocsep, TableError> {
  129         // BFS over the cocsep coordinate, keeping a cube for each.
  130         let mut depth = vec![0xFFu8; N_COCSEP];
  131         let mut cube_of = vec![Cube::solved(); N_COCSEP];
  132         let mut order = Vec::with_capacity(N_COCSEP);
  133         let solved = cocsep(Cube::solved());
  134         depth[solved] = 0;
  135         order.push(solved);
  136         let mut head = 0;
  137         while head < order.len() {
  138             let i = order[head];
  139             head += 1;
  140             for mv in Move::ALL {
  141                 let next = cube_of[i].apply_move(mv);
  142                 let j = cocsep(next);
  143                 if depth[j] == 0xFF {
  144                     depth[j] = depth[i] + 1;
  145                     cube_of[j] = next;
  146                     order.push(j);
  147                 }
  148             }
  149         }
  150         let model_error = |what, found, expected| TableError::CocsepModel {
  151             what,
  152             found,
  153             expected,
  154         };
  155         if order.len() != N_COCSEP {
  156             return Err(model_error("coordinates", order.len(), N_COCSEP));
  157         }
  158         // (Depths are below 16: they fit the packed entries' 4 bits.)
  159         let mut distribution = vec![0u64; 16].into_boxed_slice();
  160         for &d in &depth {
  161             let count = distribution.get_mut(usize::from(d)).ok_or(model_error(
  162                 "a depth of",
  163                 usize::from(d),
  164                 15,
  165             ))?;
  166             *count += 1;
  167         }
  168         // Classes, in BFS order: the first coordinate of a class is its
  169         // representative; its 48 conjugates are the class's members.
  170         let mut class = vec![u32::MAX; N_COCSEP];
  171         let mut sym_to_rep = vec![0u8; N_COCSEP];
  172         // (Filled by class index: `into_boxed_slice` of a pushed Vec keeps a
  173         // shrink path, +654 bytes of solve::main.)
  174         let mut representatives = vec![Cube::solved(); N_CLASSES].into_boxed_slice();
  175         let mut self_symmetries = vec![0u64; N_CLASSES].into_boxed_slice();
  176         let mut n_classes = 0;
  177         for &i in &order {
  178             if class[i] != u32::MAX {
  179                 continue;
  180             }
  181             // (Below `N_CLASSES`, so it fits in the entries and indexes the
  182             // class arrays.)
  183             let new_class = u32::try_from(n_classes)
  184                 .ok()
  185                 .filter(|_| n_classes < N_CLASSES)
  186                 .ok_or(model_error("more classes than", n_classes + 1, N_CLASSES))?;
  187             let rep = cube_of[i];
  188             let mut fixing = 0u64;
  189             for (sym, &inverse) in crate::simd::SYM_INVERSES.iter().enumerate() {
  190                 let j = cocsep(rep.transform(sym));
  191                 if class[j] == u32::MAX {
  192                     class[j] = new_class;
  193                     sym_to_rep[j] = inverse;
  194                 }
  195                 if j == i {
  196                     fixing |= 1 << sym;
  197                 }
  198             }
  199             debug_assert_eq!(sym_to_rep[i], 0);
  200             representatives[n_classes] = rep;
  201             self_symmetries[n_classes] = fixing;
  202             n_classes += 1;
  203         }
  204         if n_classes != N_CLASSES {
  205             return Err(model_error("classes", n_classes, N_CLASSES));
  206         }
  207         let entries: Box<[u32]> = (0..N_COCSEP)
  208             .map(|i| (class[i] << 16) | (u32::from(sym_to_rep[i]) << 8) | u32::from(depth[i]))
  209             .collect();
  210         // 2 MiB: the kernel aligns it so that it gets one huge page.
  211         let packed = Memory::anonymous(PACKED_LEN)?;
  212         // SAFETY: `3 * N_COCSEP + 1` bytes of the fresh 2 MiB (`PACKED_LEN`),
  213         // ours alone.
  214         let bytes = unsafe { std::slice::from_raw_parts_mut(packed.ptr(), 3 * N_COCSEP + 1) };
  215         let packed_entries = class.iter().zip(&sym_to_rep).zip(&depth);
  216         for (slot, ((&class, &sym_to_rep), &depth)) in
  217             bytes.as_chunks_mut::<3>().0.iter_mut().zip(packed_entries)
  218         {
  219             let entry = (class << 10) | (u32::from(sym_to_rep) << 4) | u32::from(depth);
  220             let [first, second, third, _] = entry.to_le_bytes();
  221             *slot = [first, second, third];
  222         }
  223         Ok(Cocsep {
  224             entries,
  225             representatives,
  226             self_symmetries,
  227             distribution,
  228             packed,
  229         })
  230     }

  232     /// A cheap stand-in for unit tests: corner depths from a BFS cut after
  233     /// `depth` moves (deeper coordinates get `depth + 1`, still a lower bound
  234     /// and still a function of the coset), all in one class with no symmetry.
  235     #[cfg(test)]
  236     pub(crate) fn truncated(depth: u8) -> Cocsep {
  237         let mut depths = vec![depth + 1; N_COCSEP];
  238         depths[cocsep(Cube::solved())] = 0;
  239         let mut frontier = vec![Cube::solved()];
  240         for d in 1..=depth {
  241             let mut next = Vec::new();
  242             for cube in frontier {
  243                 for mv in Move::ALL {
  244                     let child = cube.apply_move(mv);
  245                     let coord = cocsep(child);
  246                     if depths[coord] > d {
  247                         depths[coord] = d;
  248                         next.push(child);
  249                     }
  250                 }
  251             }
  252             frontier = next;
  253         }
  254         // Packed entries `class << 10 | sym_to_rep << 4 | depth` with class
  255         // and sym_to_rep 0: just the depth byte.
  256         let packed = Memory::anonymous(PACKED_LEN).unwrap();
  257         let bytes = unsafe { std::slice::from_raw_parts_mut(packed.ptr(), 3 * N_COCSEP + 1) };
  258         let mut distribution = vec![0u64; 16].into_boxed_slice();
  259         for (i, &d) in depths.iter().enumerate() {
  260             bytes[3 * i] = d;
  261             distribution[usize::from(d)] += 1;
  262         }
  263         Cocsep {
  264             entries: depths.iter().map(|&d| u32::from(d)).collect(),
  265             representatives: vec![Cube::solved(); N_CLASSES].into_boxed_slice(),
  266             self_symmetries: vec![1; N_CLASSES].into_boxed_slice(),
  267             distribution,
  268             packed,
  269         }
  270     }

  318 /// wasm32 linear memory never shrinks: the pages of the largest dropped
  319 /// `Memory` (address, bytes), for the next one that fits (e.g. a table
  320 /// loaded again).
  321 #[cfg(all(not(unix), target_arch = "wasm32", not(wasm_threads)))]

  324 unsafe impl Send for Memory {}

  328     /// Anonymous zeroed memory, transparent huge pages requested.
  329     ///
  330     /// # Errors
  331     /// If `mmap` fails.
  332     #[cfg(unix)]
  333     pub fn anonymous(len: usize) -> Result<Memory, TableError> {
  334         if isize::try_from(len).is_err() {
  335             return Err(TableError::TooLargeForSlice(len));
  336         }
  337         // (Miri supports only these flags, and no `madvise`.)
  338         #[cfg(not(miri))]
  339         let flags = libc::MAP_PRIVATE | libc::MAP_ANONYMOUS | libc::MAP_NORESERVE;
  340         #[cfg(miri)]
  341         let flags = libc::MAP_PRIVATE | libc::MAP_ANONYMOUS;
  342         unsafe {
  343             let p = libc::mmap(
  344                 std::ptr::null_mut(),
  345                 len,
  346                 libc::PROT_READ | libc::PROT_WRITE,
  347                 flags,
  348                 -1,
  349                 0,
  350             );
  351             if p == libc::MAP_FAILED {
  352                 return Err(TableError::Alloc {
  353                     bytes: len,
  354                     source: std::io::Error::last_os_error(),
  355                 });
  356             }
  357             if p.is_null() {
  358                 libc::munmap(p, len);
  359                 return Err(TableError::Alloc {
  360                     bytes: len,
  361                     source: std::io::Error::other("mapping returned a null address"),
  362                 });
  363             }
  364             #[cfg(all(not(miri), target_os = "linux"))]
  365             libc::madvise(p, len, libc::MADV_HUGEPAGE);
  366             Ok(Memory {
  367                 ptr: p.cast::<u8>(),
  368                 len,
  369             })
  370         }
  371     }

  395     /// Zeroed page-aligned memory from wasi-libc (`posix_memalign`), or
  396     /// `None` if there is not enough. The threads build's memory is all there
  397     /// from the start and never grows (see `wasm/build.rs`), and C's
  398     /// allocator supplies page alignment. Requests above `isize::MAX` are
  399     /// rejected because the solver uses in-bounds pointer arithmetic.
  400     #[cfg(all(target_arch = "wasm32", wasm_threads))]
  401     #[must_use]
  402     pub fn try_anonymous(len: usize) -> Option<Memory> {
  403         isize::try_from(len).ok()?;
  404         let mut ptr = std::ptr::null_mut();
  405         // SAFETY: an out-pointer; the block is freed in `drop`.
  406         if unsafe { libc::posix_memalign(&raw mut ptr, 4096, len.max(1)) } != 0 {
  407             return None;
  408         }
  409         let ptr = ptr.cast::<u8>();
  410         // SAFETY: `len` bytes of the fresh block.
  411         unsafe { std::ptr::write_bytes(ptr, 0, len) };
  412         Some(Memory { ptr, len })
  413     }

  415     /// Zeroed whole pages of wasm32 linear memory (`memory.grow`), or `None`
  416     /// if the engine has no more (at most 4 GiB in all). Not from the
  417     /// allocator: pages can be reused without growing the allocator's heap.
  418     /// A single block must still fit in `isize::MAX` (under 2 GiB on wasm32).
  419     #[cfg(all(not(unix), target_arch = "wasm32", not(wasm_threads)))]
  420     #[must_use]
  421     pub fn try_anonymous(len: usize) -> Option<Memory> {
  422         const PAGE: usize = 1 << 16;
  423         isize::try_from(len).ok()?;
  424         let pages_len = len.max(1).checked_next_multiple_of(PAGE)?;
  425         let mut spare = SPARE_PAGES
  426             .lock()
  427             .unwrap_or_else(std::sync::PoisonError::into_inner);
  428         let (address, pages_len) = if spare.1 >= pages_len {
  429             let (address, spare_len) = std::mem::take(&mut *spare);
  430             // SAFETY: the pages of a dropped `Memory`, used by nobody else.
  431             unsafe {
  432                 std::ptr::write_bytes(std::ptr::with_exposed_provenance_mut::<u8>(address), 0, len);
  433             }
  434             (address, spare_len)
  435         } else {
  436             let first_page = core::arch::wasm32::memory_grow::<0>(pages_len / PAGE);
  437             if first_page == usize::MAX {
  438                 return None;
  439             }
  440             // (Fresh pages are zero.)
  441             (first_page * PAGE, pages_len)
  442         };
  443         Some(Memory {
  444             ptr: std::ptr::with_exposed_provenance_mut(address),
  445             len,
  446             pages_len,
  447         })
  448     }

  495     #[cfg(unix)]
  496     fn map(path: &str, len: usize, create: bool, prefault: bool) -> std::io::Result<Memory> {
  497         use std::os::unix::io::AsRawFd;
  498         isize::try_from(len).map_err(std::io::Error::other)?;
  499         // (Opened tables are mapped read-only: a stray write faults instead
  500         // of corrupting the shared table file for every process.)
  501         let file = std::fs::OpenOptions::new()
  502             .read(true)
  503             .write(create)
  504             .create(create)
  505             .truncate(false)
  506             .open(path)?;
  507         let hugetlbfs = file::is_hugetlbfs(path);
  508         if create {
  509             // hugetlbfs needs the size rounded up to the page size.
  510             let page = 1usize << 30;
  511             let rounded = if hugetlbfs {
  512                 len.div_ceil(page) * page
  513             } else {
  514                 len
  515             };
  516             file.set_len(rounded as u64)?;
  517         }
  518         let file_len = usize::try_from(file.metadata()?.len()).map_err(std::io::Error::other)?;
  519         isize::try_from(file_len).map_err(std::io::Error::other)?;
  520         if file_len < len {
  521             return Err(std::io::Error::other(format!(
  522                 "table file {path} too small: {file_len} < {len}"
  523             )));
  524         }
  525         let protection = if create {
  526             libc::PROT_READ | libc::PROT_WRITE
  527         } else {
  528             libc::PROT_READ
  529         };
  530         // (The kernel populates with one thread: right for hugetlbfs, whose
  531         // pages are in memory already, and new files; other files are read
  532         // in parallel by `prefault`.)
  533         let populate_in_kernel = prefault && (create || hugetlbfs);
  534         #[cfg(target_os = "linux")]
  535         let flags = if populate_in_kernel {
  536             libc::MAP_SHARED | libc::MAP_POPULATE
  537         } else {
  538             libc::MAP_SHARED
  539         };
  540         #[cfg(not(target_os = "linux"))]
  541         let flags = libc::MAP_SHARED;
  542         // SAFETY: a fresh mapping of `file_len` bytes of an open file.
  543         let p = unsafe {
  544             libc::mmap(
  545                 std::ptr::null_mut(),
  546                 file_len,
  547                 protection,
  548                 flags,
  549                 file.as_raw_fd(),
  550                 0,
  551             )
  552         };
  553         if p == libc::MAP_FAILED {
  554             return Err(std::io::Error::last_os_error());
  555         }
  556         if p.is_null() {
  557             // SAFETY: this is the mapping just returned by mmap.
  558             unsafe {
  559                 libc::munmap(p, file_len);
  560             }
  561             return Err(std::io::Error::other("mapping returned a null address"));
  562         }
  563         let memory = Memory {
  564             ptr: p.cast::<u8>(),
  565             len: file_len,
  566         };
  567         // Huge pages where the kernel and file system support them (for
  568         // regular files: page cache folios of 2 MiB, mapped whole).
  569         #[cfg(target_os = "linux")]
  570         // SAFETY: advice on our own mapping.
  571         unsafe {
  572             libc::madvise(p, file_len, libc::MADV_HUGEPAGE);
  573         }
  574         if prefault && !(cfg!(target_os = "linux") && populate_in_kernel) {
  575             memory.prefault()?;
  576         }
  577         Ok(memory)
  578     }

  580     /// Fault every page of the mapping in, reading in parallel (16 threads;
  581     /// a range whose thread cannot be started is read on this one).
  582     ///
  583     /// # Errors
  584     /// If a thread panicked.
  585     #[cfg(unix)]
  586     fn prefault(&self) -> std::io::Result<()> {
  587         let per = self.len.div_ceil(16).next_multiple_of(1 << 21);
  588         std::thread::scope(|s| {
  589             let mut threads = Vec::with_capacity(16);
  590             for start in (0..self.len).step_by(per.max(1)) {
  591                 let len = per.min(self.len - start);
  592                 // (A raw pointer is not `Send`: pass the address.)
  593                 let address = self.ptr as usize + start;
  594                 // SAFETY (both calls): a range of our own mapping.
  595                 let fault = move || unsafe { prefault_range(address, len) };
  596                 match std::thread::Builder::new().spawn_scoped(s, fault) {
  597                     Ok(thread) => threads.push(thread),
  598                     Err(_) => unsafe { prefault_range(address, len) },
  599                 }
  600             }
  601             let mut result = Ok(());
  602             for thread in threads {
  603                 if thread.join().is_err() {
  604                     result = Err(std::io::Error::other("a prefault thread panicked"));
  605                 }
  606             }
  607             result
  608         })
  609     }

  667 /// Fault in `len` bytes from `address`: `MADV_POPULATE_READ`, else a read of
  668 /// a byte per 4 KiB (the smallest page size).
  669 ///
  670 /// # Safety
  671 /// `address..address + len` is mapped (and readable).
  672 #[cfg(unix)]
  673 unsafe fn prefault_range(address: usize, len: usize) {
  674     #[cfg(target_os = "linux")]
  675     // SAFETY: advice on a mapped range (the caller's contract).
  676     unsafe {
  677         let range = address as *mut libc::c_void;
  678         if libc::madvise(range, len, libc::MADV_POPULATE_READ) == 0 {
  679             return;
  680         }
  681     }
  682     let mut sum = 0u8;
  683     for offset in (0..len).step_by(4096) {
  684         // SAFETY: within the range (the caller's contract).
  685         sum ^= unsafe { std::ptr::read_volatile((address + offset) as *const u8) };
  686     }
  687     std::hint::black_box(sum);
  688 }

  714                 .unwrap_or_else(std::sync::PoisonError::into_inner);

  723 // H48 table (cube48opt layout): 128-byte rows of 512 2-bit slots, one row
  724 // per (class, msplit, eoK); the row is indexed by eslot. Each 16-byte group
  725 // (64 slots) reserves slots 0-1 for a 4-bit minimum over the group. (Neutral
  726 // at h11 when added; -71% nodes with h0, eslice-groups.)

  729 pub const ROW_BYTES: u64 = 128;
  730 pub const WORDS_PER_ROW: usize = 16;

  732 /// Rows of table hk: one per (class, msplit, top `k` eo bits).
  733 #[must_use]
  734 pub fn h48_rows(k: u32) -> u64 {
  735     (N_CLASSES * N_MSPLIT) as u64 * (1u64 << k)
  736 }
  737 /// Size of the index space (including unused slots).
  738 #[must_use]
  739 pub fn h48_slots(k: u32) -> u64 {
  740     h48_rows(k) * N_SLOTS as u64
  741 }
  742 /// Size of table hk in bytes.
  743 #[must_use]
  744 pub fn h48_bytes(k: u32) -> u64 {
  745     h48_rows(k) * ROW_BYTES
  746 }

  748 /// Default base value per k (`gen <k> auto`; cube48opt's, confirmed for h4..h8
  749 /// by nodes/cube on random:200:2026 and by solve speed). Base 9 for h4..h6
  750 /// (23-67% fewer nodes than base 8, 7-44% fewer than base 10; h6 at 1T and 8
  751 /// threads 15-20% faster than with base 10); h7: base 10 has 3.5% fewer
  752 /// nodes, but with 40 lanes is 4-9% faster at 1T and 6.3% slower at 8
  753 /// threads: a tie, so base 9 (generated in 30 s instead of 5 min on 8
  754 /// cores); h8: base 10 (10% fewer nodes than base 9); h9, h10 as h8 and h11
  755 /// (base 11: +10% nodes, x-h48); h0..h3 not measured.
  756 #[must_use]
  757 pub fn default_base(k: u32) -> u8 {
  758     [7, 8, 8, 8, 9, 9, 9, 9, 10, 10, 10, 10][k as usize]
  759 }

  761 /// u64 word index and bit shift of an entry.
  762 #[must_use]
  763 #[inline(always)]
  764 pub fn entry_position(index: u64) -> (usize, u32) {
  765     ((index >> 5) as usize, (index & 31) as u32 * 2)
  766 }

  768 /// Mask of the low bits of the 2-bit slots of word `w` that are real
  769 /// entries.
  770 #[must_use]
  771 #[inline(always)]
  772 pub fn entry_low_bits(w: u64) -> u64 {
  773     let mut mask = 0x5555_5555_5555_5555u64;
  774     if w.is_multiple_of(2) {
  775         mask &= !0x5; // slots 0, 1: group minimum
  776     }
  777     if w % WORDS_PER_ROW as u64 == WORDS_PER_ROW as u64 - 1 {
  778         mask &= !(1 << 62); // slot 511 unused
  779     }
  780     mask
  781 }

  856     /// Returns (2-bit code, group minimum), or `None` outside the table.
  857     #[must_use]
  858     #[inline(always)]
  859     #[expect(clippy::cast_ptr_alignment, reason = "tables are page aligned")]
  860     pub fn get(&self, index: u64) -> Option<(u8, u8)> {
  861         if index >= h48_slots(self.meta.k) {
  862             return None;
  863         }
  864         let w = usize::try_from(index >> 5).ok()?;
  865         if w >= self.memory.len() / 8 {
  866             return None;
  867         }
  868         // SAFETY: w and the preceding group word are within this aligned,
  869         // initialized allocation, which is immutable while borrowed.
  870         unsafe {
  871             let words = self.memory.ptr().cast_const().cast::<u64>();
  872             let word = *words.add(w);
  873             // (The group's first word holds the minimum in slots 0-1.)
  874             let group_min = *words.add(w & !1) & 15;
  875             Some((((word >> ((index & 31) * 2)) & 3) as u8, group_min as u8))
  876         }
  877     }

  892     /// Distribution of the 2-bit codes over real entries.

  894     /// # Errors
  895     /// If the table is larger than its memory or a slice can be, or a
  896     /// counting thread panicked.
  897     pub fn distribution(&self) -> Result<[u64; 4], TableError> {
  898         // SAFETY: distribution only loads the words; it never mutates them.
  899         let words = unsafe { self.memory.words()? };
  900         let len = usize::try_from(h48_bytes(self.meta.k) / 8)
  901             .map_err(|_| TableError::TooLarge(self.meta.k))?;
  902         let words = words.get(..len).ok_or(TableError::TooLarge(self.meta.k))?;
  903         Ok(distribution(words, crate::thread_count())?)

  907 /// Distribution of the 2-bit codes over the real entries of whole rows of a
  908 /// table (`words` from the start of a row).
  909 ///
  910 /// # Errors
  911 /// If a thread panicked.
  912 pub fn distribution(
  913     words: &[AtomicU64],
  914     threads: usize,
  915 ) -> Result<[u64; 4], crate::parallel::WorkerPanicked> {
  916     let per_thread = words.len().div_ceil(threads.max(1)).max(1);
  917     let mut total = [0u64; 4];
  918     let chunks: Vec<&[AtomicU64]> = words.chunks(per_thread).collect();
  919     let counts = crate::parallel::on_threads(chunks.len(), |t| {
  920         let mut counts = [0u64; 4];
  921         for (w, word) in ((t * per_thread) as u64..).zip(chunks.get(t).copied().unwrap_or(&[])) {
  922             let bits = word.load(Ordering::Relaxed);
  923             let mask = entry_low_bits(w);
  924             let lo = bits & mask;
  925             let hi = (bits >> 1) & mask;
  926             counts[3] += u64::from((lo & hi).count_ones());
  927             counts[2] += u64::from((!lo & hi).count_ones());
  928             counts[1] += u64::from((lo & !hi).count_ones());
  929             counts[0] += u64::from((mask & !lo & !hi).count_ones());
  930         }
  931         counts
  932     })?;
  933     for thread_counts in counts {
  934         for (sum, count) in total.iter_mut().zip(thread_counts) {
  935             *sum += count;
  936         }
  937     }
  938     Ok(total)
  939 }

  990 /// from its sidecar meta file (`meta_path`).

  994 /// most 12, or the trailer's data size is not that of `k`.

 1005     let text = std::fs::read_to_string(meta_path(path)).map_err(|source| TableError::NoMeta {

 1012         .split_whitespace()
 1013         .map(|x| x.parse().map_err(|_| invalid()))
 1014         .collect::<Result<_, _>>()?;
 1015     let [k, base] = numbers[..] else {
 1016         return Err(invalid());
 1017     };

 1081         let len = usize::try_from(h48_bytes(meta.k)).map_err(|_| TableError::TooLarge(meta.k))?;
 1082         // A file of another `k` (a stale or wrong meta) would give wrong,
 1083         // possibly too high, bounds: its size must match (plus a trailer;
 1084         // hugetlbfs files are rounded up to whole 1 GiB pages).
 1085         let io_error = file::io_error(path);
 1086         let file_len = std::fs::metadata(path).map_err(io_error)?.len();
 1087         if file_len < len as u64 || file_len >= len as u64 + file::TRAILER_BYTES + file::HUGE_PAGE {
 1088             return Err(TableError::WrongSize {
 1089                 path: path.to_string(),
 1090                 bytes: file_len,
 1091                 k: meta.k,
 1092                 needed: len as u64,
 1093             });
 1094         }
 1095         let mut report = LoadReport {
 1096             mode: options.mode_for(path),
 1097             segment: None,
 1098             notes: Vec::new(),
 1099         };
 1100         if matches!(options.mode, file::LoadMode::Auto | file::LoadMode::Shm) {
 1101             let create = options.mode == file::LoadMode::Shm;
 1102             if let Some((memory, segment)) = shm::open(path, options, create, &mut report.notes)? {
 1103                 report.mode = file::LoadMode::Shm;
 1104                 report.segment = Some(segment);
 1105                 return Ok((H48Table { memory, meta }, report));
 1106             }
 1107         }
 1108         let io_error = file::io_error(path);
 1109         if report.mode == file::LoadMode::Map {
 1110             // SAFETY: the caller promises an immutable, untruncated file.
 1111             let memory =
 1112                 unsafe { Memory::map_read_only(path, len, options.prefault) }.map_err(io_error)?;
 1113             return Ok((H48Table { memory, meta }, report));
 1114         }
 1115         let memory = Memory::anonymous(len)?;
 1116         let table_file = std::fs::File::open(path).map_err(io_error)?;
 1117         #[cfg(unix)]
 1118         read_parallel(&table_file, &memory, len).map_err(file::io_error(path))?;
 1119         #[cfg(not(unix))]
 1120         {
 1121             // SAFETY: `len` bytes of our own memory.
 1122             let buf = unsafe { std::slice::from_raw_parts_mut(memory.ptr(), len) };
 1123             std::io::Read::read_exact(&mut &table_file, buf).map_err(file::io_error(path))?;
 1124         }
 1125         Ok((H48Table { memory, meta }, report))
 1126     }

 1128     /// A table from the bytes of a table file already in `memory` (the data,
 1129     /// then its trailer in the last 4 KiB; e.g. a file fetched into
 1130     /// WebAssembly memory), checked as `open` checks a file: the trailer's
 1131     /// `k`, base and data size against the size, and with `verify` the
 1132     /// data's checksum, if the trailer records one (`threads` threads).
 1133     ///
 1134     /// # Errors
 1135     /// If there is no valid trailer, the size does not match it, or the
 1136     /// checksum differs.
 1137     pub fn from_file_memory(
 1138         memory: Memory,
 1139         verify: bool,
 1140         threads: usize,
 1141     ) -> Result<H48Table, TableError> {
 1142         let file_len = memory.len() as u64;
 1143         let trailer_start = file_len
 1144             .checked_sub(file::TRAILER_BYTES)
 1145             .ok_or(TableError::NotATableFile(file_len))?;
 1146         // Use bounded chunks for checksums and trailer reads.
 1147         let piece = |start: u64, len: u64| -> &[u8] {
 1148             // (Offsets within the memory, whose length is a `usize`, are
 1149             // `usize`s too; out of range, the piece is empty.)
 1150             match (usize::try_from(start), usize::try_from(len)) {
 1151                 (Ok(start), Ok(len))
 1152                     if start
 1153                         .checked_add(len)
 1154                         .is_some_and(|end| end <= memory.len()) =>
 1155                 {
 1156                     // SAFETY: within the memory (checked), which is ours and
 1157                     // not written meanwhile.
 1158                     unsafe { std::slice::from_raw_parts(memory.ptr().add(start), len) }
 1159                 }
 1160                 _ => &[],
 1161             }
 1162         };
 1163         let trailer = file::Trailer::parse(piece(trailer_start, file::TRAILER_BYTES))?
 1164             .ok_or(TableError::NoTrailer)?;
 1165         let meta = checked_meta(trailer.k, u32::from(trailer.base))
 1166             .filter(|meta| trailer.data_bytes == h48_bytes(meta.k))
 1167             .ok_or(TableError::UnusableTrailer(trailer))?;
 1168         // (As in `open_with`; hugetlbfs files have up to a page of slack.)
 1169         let data_len = trailer.data_bytes;
 1170         if trailer_start < data_len || trailer_start >= data_len + file::HUGE_PAGE {
 1171             return Err(TableError::WrongMemorySize {
 1172                 bytes: file_len,
 1173                 k: meta.k,
 1174                 needed: data_len,
 1175             });
 1176         }
 1177         if let (true, Some(want)) = (verify, trailer.checksum) {
 1178             // (Pieces of whole checksum chunks: the same chunk hashes as one
 1179             // slice of all the data.)
 1180             let piece_len = (file::CHECKSUM_CHUNK << 10) as u64;
 1181             let mut hashes = Vec::new();
 1182             for i in 0..data_len.div_ceil(piece_len) {
 1183                 let start = i * piece_len;
 1184                 hashes.extend(file::chunk_hashes(
 1185                     piece(start, piece_len.min(data_len - start)),
 1186                     threads,
 1187                 )?);
 1188             }
 1189             let got = file::combine_hashes(data_len, &hashes);
 1190             if got != want {
 1191                 return Err(TableError::Checksum { got, want });
 1192             }
 1193         }
 1194         Ok(H48Table { memory, meta })
 1195     }

 1254 /// Read the first `len` bytes of `table_file` into `memory`, in parallel (16
 1255 /// threads: a disk or page cache copy of up to tens of GiB; a range whose
 1256 /// thread cannot be started is read on this one), each 64 MiB piece dropped
 1257 /// from the page cache once copied (the copy is what is used).
 1258 ///
 1259 /// # Errors
 1260 /// If a read fails, or a thread panicked.
 1261 #[cfg(unix)]
 1262 fn read_parallel(table_file: &std::fs::File, memory: &Memory, len: usize) -> std::io::Result<()> {
 1263     let threads = 16usize;
 1264     // (Whole 2 MiB pages per thread.)
 1265     let per_thread = len.div_ceil(threads).next_multiple_of(1 << 21);
 1266     let len = len.min(memory.len());
 1267     std::thread::scope(|s| {
 1268         let mut spawned = Vec::with_capacity(threads);
 1269         let mut result = Ok(());
 1270         for t in 0..threads {
 1271             let start = t * per_thread;
 1272             let end = ((t + 1) * per_thread).min(len);
 1273             if start >= end {
 1274                 continue;
 1275             }
 1276             // SAFETY (both calls): `start..end` is within the memory, and
 1277             // each range is read once, by one thread.
 1278             let read = move || unsafe { read_range(table_file, memory, start, end) };
 1279             match std::thread::Builder::new().spawn_scoped(s, read) {
 1280                 Ok(thread) => spawned.push(thread),
 1281                 Err(_) => {
 1282                     result = result.and(unsafe { read_range(table_file, memory, start, end) });
 1283                 }
 1284             }
 1285         }
 1286         for thread in spawned {
 1287             let read = thread
 1288                 .join()
 1289                 .unwrap_or_else(|_| Err(std::io::Error::other("a reading thread panicked")));
 1290             result = result.and(read);
 1291         }
 1292         result
 1293     })
 1294 }

 1296 /// Read bytes `start..end` of `table_file` into the same bytes of `memory`.
 1297 ///
 1298 /// # Safety
 1299 /// `start..end` is within `memory`, and nothing else accesses those bytes
 1300 /// meanwhile.
 1301 #[cfg(unix)]
 1302 unsafe fn read_range(
 1303     table_file: &std::fs::File,
 1304     memory: &Memory,
 1305     start: usize,
 1306     end: usize,
 1307 ) -> std::io::Result<()> {
 1308     use std::os::unix::fs::FileExt;
 1309     // SAFETY: as the caller promises.
 1310     let buf = unsafe { std::slice::from_raw_parts_mut(memory.ptr().add(start), end - start) };
 1311     for (i, piece) in buf.chunks_mut(64 << 20).enumerate() {
 1312         let offset = (start + (i << 26)) as u64;
 1313         table_file.read_exact_at(piece, offset)?;
 1314         // (Advice only: skipped if the numbers do not fit.)
 1315         #[cfg(target_os = "linux")]
 1316         if let (Ok(offset), Ok(len)) = (
 1317             libc::off_t::try_from(offset),
 1318             libc::off_t::try_from(piece.len()),
 1319         ) {
 1320             use std::os::unix::io::AsRawFd;
 1321             // SAFETY: advice on an open file.
 1322             unsafe {
 1323                 libc::posix_fadvise(
 1324                     table_file.as_raw_fd(),
 1325                     offset,
 1326                     len,
 1327                     libc::POSIX_FADV_DONTNEED,
 1328                 );
 1329             }
 1330         }
 1331     }
 1332     Ok(())
 1333 }
