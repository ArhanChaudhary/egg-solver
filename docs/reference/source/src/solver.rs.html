    1 //! IDA* optimal solver using the H48 table: the plain recursive search,
    2 //! one cube at a time. `lanes` runs the same search interleaved over many
    3 //! cubes, which is much faster; this one is the reference.
    4 //!
    5 //! IDA* (iterative deepening A*): for target lengths `T` = 1, 2, ..., a
    6 //! depth-first search for a solution of exactly `T` moves that skips any
    7 //! node whose lower bound exceeds its remaining moves. The first target
    8 //! with a solution gives an optimal one.
    9 //!
   10 //! The search runs on both the cube and its inverse (a solution of the
   11 //! inverse is the reversed, inverted solution of the cube): a node either
   12 //! applies a move to the normal cube (appended to the solution) or to the
   13 //! inverse cube (a "premove", prepended in inverted form). Each node moves
   14 //! the side with fewer allowed moves, and both sides' table bounds prune.
   15 //!
   16 //! The first moves of each iteration are split into "jobs" (prefixes of up
   17 //! to 4 moves), sorted by how promising they were in the last iteration,
   18 //! so that the last iteration tends to find its solution early (-36% nodes,
   19 //! +46% at 48T on a batch when added, job-order).
   20

   27 /// All 18 moves (the low 18 bits).
   28 pub const ALL_MOVES: u32 = u32::MAX.unbounded_shr(32 - 18);
   29 
   30 /// The largest target (iteration depth) searched: every position is solved
   31 /// in at most 20 moves (God's number), so needing a later iteration means
   32 /// an unsolvable cube or a table whose bounds are too high, and the cube is
   33 /// given up (no solution). It also bounds the search stacks' depth.
   34 pub const MAX_TARGET: u8 = 20;
   35 /// Quarter turns only (moves with m % 3 != 1).
   36 pub const QUARTER_TURNS: u32 = 0b101_101_101_101_101_101;
   37 
   38 /// Moves allowed after a move on `face` (on the same side of the search):
   39 /// not the same face again (two turns of a face are one turn or none), and
   40 /// after the second face of an opposite pair (D, L, B) not the first one
   41 /// (U D = D U: only one order is searched).
   42 const fn allowed_after_table() -> [u32; 6] {
   43     let mut table = [0u32; 6];
   44     let mut face = 0;
   45     while face < 6 {
   46         let mut moves = ALL_MOVES & !(7 << (3 * face));
   47         if face % 2 == 1 {
   48             moves &= !(7 << (3 * (face - 1)));
   49         }
   50         table[face] = moves;
   51         face += 1;
   52     }
   53     table
   54 }
   55 /// Compile-time copy of `ALLOWED_AFTER` (for `simd::HOT_TABLES`).
   56 pub(crate) const ALLOWED_AFTER_CONST: [u32; 6] = allowed_after_table();
   57 /// Moves allowed after a move on each face (see `allowed_after_table`).
   58 pub static ALLOWED_AFTER: [u32; 6] = ALLOWED_AFTER_CONST;

   60 /// The tables of a search, and the hot path's lookups.
   61 ///
   62 /// ```compile_fail,E0616
   63 /// fn replace<'a>(solver: &mut egg_solver::solver::Solver<'a>, table: &'a egg_solver::tables::H48Table) {
   64 ///     solver.table = table;
   65 /// }
   66 /// ```
   67 pub struct Solver<'a> {
   68     pub(crate) corners: &'a Cocsep,
   69     pub(crate) table: &'a H48Table,
   70     /// `table.ptr()`, cached (saves a dependent load per prefetch/lookup;
   71     /// with `value` and `prefetch`, part of m1-micro's +2.2% at 48T).
   72     table_ptr: *const u8,
   73     /// `corners.packed_ptr()`, cached (saves a dependent load per index).
   74     packed_ptr: *const u8,
   75     /// `coord::index_sym` constants of the table's `k` (with `packed_ptr`,
   76     /// part of m6-micro's index patch: +0.7% at 1T, -0.4% at 48T).
   77     row_mul: u64,
   78     eo_mask: u32,
   79     /// The table's base: codes 1..=3 mean `base + code`.
   80     pub(crate) base: u8,
   81     /// Symmetric-start filtering of the job prefixes (`symstart`; here -29%
   82     /// nodes on the 21 positions of `symmetric_start_keeps_optimal_lengths`,
   83     /// h0); off only to compare against the unfiltered search.
   84     pub sym_start: bool,
   85 }

  100 /// A canonical move prefix searched as a unit; jobs are re-ordered between
  101 /// iterations by `stat`, an estimate of how close their subtree got to
  102 /// solved (sum of 2^(12-h) over evaluated nodes), so that in the last
  103 /// iteration the solution tends to be found early.
  104 #[derive(Clone, Copy)]
  105 struct Job {
  106     moves: [u8; 4],
  107     len: u8,
  108     /// The cube after the prefix, and its inverse.
  109     cube: Cube,
  110     inverse: Cube,
  111     stat: u64,
  112     /// Lower bounds of the job root (normal, inverse); computed once.
  113     normal_bound: u8,
  114     inverse_bound: u8,
  115 }

  117 /// Face weight compensating for D/L/B prefixes having fewer continuations.
  118 #[inline(always)]
  119 fn job_weight(j: &Job) -> u64 {
  120     if (j.moves[j.len as usize - 1] / 3).is_multiple_of(2) {
  121         47525
  122     } else {
  123         58206
  124     }

  199     /// (table index, corner lower bound) of a cube.
  200     #[inline(always)]
  201     pub(crate) fn index(&self, cube: Cube) -> (u64, u8) {
  202         // `self.corners.packed(cocsep(cube))`
  203         // SAFETY: `cocsep_x3` is 3 * a coordinate below `N_COCSEP`, and the
  204         // packed table has 3 * N_COCSEP + 1 bytes (4-byte reads).
  205         let packed = unsafe {
  206             self.packed_ptr
  207                 .add(cocsep_x3(cube))
  208                 .cast::<u32>()
  209                 .read_unaligned()
  210         };
  211         let index = index_sym(
  212             cube,
  213             packed_sym_to_rep(packed),
  214             packed_class(packed),
  215             self.row_mul,
  216             self.eo_mask,
  217         );
  218         (index, packed_depth(packed))
  219     }

  221     /// The lower bound of table entry `index`: `base + code`, or for code 0
  222     /// (at most `base`) the larger of the group minimum and the corner bound.
  223     #[inline(always)]
  224     #[expect(clippy::cast_ptr_alignment, reason = "tables are page aligned")]
  225     pub(crate) fn value(&self, index: u64, corner_bound: u8) -> u8 {
  226         // `H48Table::get` through the cached pointer.
  227         // SAFETY: indexes are of the table (`index`), whose words are all
  228         // in its memory.
  229         let (code, group_min) = unsafe {
  230             let words = self.table_ptr.cast::<u64>();
  231             let w = (index >> 5) as usize;
  232             let word = *words.add(w);
  233             let group_min = *words.add(w & !1) & 15;
  234             (((word >> ((index & 31) * 2)) & 3) as u8, group_min as u8)
  235         };
  236         if code == 0 {
  237             group_min.max(corner_bound)
  238         } else {
  239             self.base + code
  240         }
  241     }

  243     /// Prefetch table entry `index` (`H48Table::prefetch`).
  244     #[inline(always)]
  245     pub(crate) fn prefetch(&self, index: u64) {
  246         // NTA: each line is read once; less L2 pollution (cocsep data).
  247         // (+2.6% at 48T when added, prefetch-nta.)
  248         // SAFETY: an entry of the table (indices are below its size).
  249         unsafe { crate::prefetch::read_once(self.table_ptr.wrapping_add((index >> 2) as usize)) };
  250     }

  252     /// Lower bound for a single cube (no pipelining).
  253     #[must_use]
  254     pub fn lower_bound(&self, cube: Cube) -> u8 {
  255         let (index, corner_bound) = self.index(cube);
  256         self.value(index, corner_bound)
  257     }

  259     /// Length of the job prefixes for an iteration with target `target`:
  260     /// longer jobs for deeper targets (more of them to sort), up to 4.
  261     pub(crate) fn job_len(&self, target: u8) -> usize {
  262         let top = self.base + 3;
  263         if target > top {
  264             ((target - top) as usize).min(4)
  265         } else {
  266             0
  267         }
  268     }

  270     /// Extend jobs by one move; children inherit the parent's stat. Jobs
  271     /// that `filter` drops (symmetric start) are left out.
  272     fn extend_jobs(jobs: &[Job], filter: &SymFilter) -> Vec<Job> {
  273         let mut out = Vec::with_capacity(jobs.len() * 15);
  274         for j in jobs {
  275             let allowed = if j.len == 0 {
  276                 ALL_MOVES
  277             } else {
  278                 ALLOWED_AFTER[(j.moves[j.len as usize - 1] / 3) as usize]
  279             };
  280             let mut todo = allowed;
  281             while let Some(mv) = Move::lowest(todo) {
  282                 todo ^= todo.isolate_lowest_one();
  283                 let mut child = *j;
  284                 child.moves[child.len as usize] = mv as u8;
  285                 child.len += 1;
  286                 if !filter.keep(&child.moves[..usize::from(child.len)]) {
  287                     continue;
  288                 }
  289                 child.cube = j.cube.apply_move(mv);
  290                 child.inverse = j.inverse.premove_inverse(mv);
  291                 out.push(child);
  292             }
  293         }
  294         filter.dedup(
  295             &mut out,
  296             |job| &job.moves[..usize::from(job.len)],
  297             |job| job.cube,
  298         );
  299         out
  300     }

  302     /// Compute the root bounds of jobs, prefetching in batches (+2.4%
  303     /// throughput, +1.8% on a 48T batch when added, job-bounds).
  304     fn job_bounds(&self, jobs: &mut [Job]) {
  305         for chunk in jobs.chunks_mut(32) {
  306             let mut lookups = [(0u64, 0u8); 64];
  307             for (n, j) in chunk.iter().enumerate() {
  308                 lookups[2 * n] = self.index(j.cube);
  309                 lookups[2 * n + 1] = self.index(j.inverse);
  310                 self.prefetch(lookups[2 * n].0);
  311                 self.prefetch(lookups[2 * n + 1].0);
  312             }
  313             for (n, j) in chunk.iter_mut().enumerate() {
  314                 j.normal_bound = self.value(lookups[2 * n].0, lookups[2 * n].1);
  315                 j.inverse_bound = self.value(lookups[2 * n + 1].0, lookups[2 * n + 1].1);
  316             }
  317         }
  318     }

  320     /// Find one optimal solution: `None` if there is none within
  321     /// `MAX_TARGET` moves (an unsolvable cube, or a table whose bounds are
  322     /// too high).
  323     #[must_use]
  324     pub fn solve(&self, cube: &CubieCube) -> (Option<Vec<u8>>, Stats) {
  325         let cube = Cube::from_cubie(cube);
  326         if cube.is_solved() {
  327             return (Some(vec![]), Stats::default());
  328         }
  329         let inverse = cube.inverse();
  330         let normal_bound = self.lower_bound(cube);
  331         let inverse_bound = self.lower_bound(inverse);
  332         let filter = self.sym_filter(cube, inverse);
  333         let mut state = SearchState {
  334             moves: [0; 32],
  335             n_moves: 0,
  336             premoves: [0; 32],
  337             n_premoves: 0,
  338             stats: Stats::default(),
  339             stat: 0,
  340         };
  341         let mut jobs = vec![Job {
  342             moves: [0; 4],
  343             len: 0,
  344             cube,
  345             inverse,
  346             stat: 0,
  347             normal_bound,
  348             inverse_bound,
  349         }];
  350         let mut target = normal_bound.max(inverse_bound).max(1);
  351         loop {
  352             // (Also keeps the moves within `SearchState::moves`.)
  353             if target > MAX_TARGET {
  354                 return (None, state.stats);
  355             }
  356             let job_len = self.job_len(target);
  357             while jobs.first().is_some_and(|j| usize::from(j.len) < job_len) {
  358                 jobs = Self::extend_jobs(&jobs, &filter);
  359                 self.job_bounds(&mut jobs);
  360             }
  361             if job_len > 0 {
  362                 // stable sort keeps the canonical order among equal keys
  363                 jobs.sort_by_key(|j| std::cmp::Reverse(j.stat * job_weight(j)));
  364             }
  365             for j in &mut jobs {
  366                 let remaining_depth = target - j.len;
  367                 let (normal_bound, inverse_bound) = (j.normal_bound, j.inverse_bound);
  368                 state.stat = 0;
  369                 if normal_bound > remaining_depth
  370                     || inverse_bound > remaining_depth
  371                     || (j.len > 0 && j.cube.is_solved())
  372                 {
  373                     j.stat = 0;
  374                     continue;
  375                 }
  376                 let mut normal_moves = if j.len == 0 {
  377                     ALL_MOVES
  378                 } else {
  379                     ALLOWED_AFTER[(j.moves[j.len as usize - 1] / 3) as usize]
  380                 };
  381                 let mut inverse_moves = ALL_MOVES;
  382                 // A tight bound on one side: a half turn on the other side
  383                 // keeps it (see `search`), so only quarter turns can help.
  384                 if inverse_bound == remaining_depth {
  385                     normal_moves &= QUARTER_TURNS;
  386                 }
  387                 if normal_bound == remaining_depth {
  388                     inverse_moves &= QUARTER_TURNS;
  389                 }
  390                 for (to, &mv) in state
  391                     .moves
  392                     .iter_mut()
  393                     .zip(&j.moves)
  394                     .take(usize::from(j.len))
  395                 {
  396                     *to = mv;
  397                 }
  398                 state.n_moves = j.len as usize;
  399                 state.n_premoves = 0;
  400                 let found = self.search(
  401                     &mut state,
  402                     j.cube,
  403                     j.inverse,
  404                     normal_bound,
  405                     inverse_bound,
  406                     normal_moves,
  407                     inverse_moves,
  408                     remaining_depth,
  409                 ) == FOUND;
  410                 j.stat = state.stat;
  411                 if found {
  412                     let mut solution: Vec<u8> = state.moves[..state.n_moves].to_vec();
  413                     for k in (0..state.n_premoves).rev() {
  414                         solution.push(inverse_move(state.premoves[k]));
  415                     }
  416                     return (Some(solution), state.stats);
  417                 }
  418             }
  419             target += 1;
  420         }
  421     }

  423     /// Search for a solution of exactly `remaining_depth` (> 0) more moves.
  424     /// The bounds `normal_bound`, `inverse_bound` are at most
  425     /// `remaining_depth`; `normal_moves`, `inverse_moves` are the moves
  426     /// allowed on each side.
  427     ///
  428     /// Returns `FOUND`, or a lower bound on the distance of this node learned
  429     /// from its children (0 if nothing was learned). Every child value is a
  430     /// lower bound on the child's true distance, and the true distance
  431     /// changes by at most one per move, so (a) a child value `v` bounds this
  432     /// node by `v - 1` and (b) bounds same-face siblings (one move apart) by
  433     /// `v - 1` as well. (Both: -17.7% nodes, +7.5% throughput when added,
  434     /// sibling-prune.)
  435     #[allow(clippy::too_many_arguments)]
  436     fn search(
  437         &self,
  438         state: &mut SearchState,
  439         cube: Cube,
  440         inverse: Cube,
  441         normal_bound: u8,
  442         inverse_bound: u8,
  443         normal_moves: u32,
  444         inverse_moves: u32,
  445         remaining_depth: u8,
  446     ) -> u8 {
  447         if remaining_depth == 1 {
  448             return Self::last_move(state, cube, normal_moves & inverse_moves);
  449         }
  450         let child_remaining = remaining_depth - 1;
  451         // Move the side with fewer moves to try (ties: the normal cube; as
  452         // the other search techniques of the first version, not measured
  453         // separately).
  454         let normal = normal_moves.count_ones() <= inverse_moves.count_ones();
  455         let (moved, other, allowed, other_side_bound) = if normal {
  456             (cube, inverse, normal_moves, inverse_bound)
  457         } else {
  458             (inverse, cube, inverse_moves, normal_bound)
  459         };
  460         // Best known lower bound per face over that face's children.
  461         let mut face_bound = [0u8; 6];
  462         let (mut children, n_children) = Self::generate_children(
  463             state,
  464             (moved, other),
  465             allowed,
  466             other_side_bound,
  467             child_remaining,
  468             &mut face_bound,
  469         );
  470         let children = &mut children[..n_children];
  471         self.bound_children(state, children, &mut face_bound, child_remaining);
  472         // Lower bound for this node from all children.
  473         let mut lower_bound = face_bound
  474             .iter()
  475             .copied()
  476             .max()
  477             .unwrap_or(0)
  478             .saturating_sub(1);
  479         if lower_bound > remaining_depth {
  480             return lower_bound;
  481         }
  482         for child in &*children {
  483             let face = (child.mv / 3) as usize;
  484             // Pruned directly, or by a same-face sibling that is >= 2 over.
  485             if !child.live || face_bound[face] > child_remaining + 1 {
  486                 continue;
  487             }
  488             // Solved before the target depth: a shorter solution exists.
  489             if child.moved.is_solved() {
  490                 continue;
  491             }
  492             // The child's two sides, back in normal/inverse terms.
  493             let (mut child_normal_moves, mut child_inverse_moves, child_cube, child_inverse);
  494             let (child_normal_bound, child_inverse_bound);
  495             if normal {
  496                 child_normal_moves = ALLOWED_AFTER[face];
  497                 child_inverse_moves = inverse_moves;
  498                 child_cube = child.moved;
  499                 child_inverse = child.other;
  500                 child_normal_bound = child.moved_bound;
  501                 child_inverse_bound = child.other_bound;
  502                 state.moves[state.n_moves] = child.mv;
  503                 state.n_moves += 1;
  504             } else {
  505                 child_normal_moves = normal_moves;
  506                 child_inverse_moves = ALLOWED_AFTER[face];
  507                 child_cube = child.other;
  508                 child_inverse = child.moved;
  509                 child_normal_bound = child.other_bound;
  510                 child_inverse_bound = child.moved_bound;
  511                 state.premoves[state.n_premoves] = child.mv;
  512                 state.n_premoves += 1;
  513             }
  514             // A tight bound on one side: a half turn on the other side keeps
  515             // that side's coset, hence its bound, so it cannot finish in time.
  516             if child_inverse_bound == child_remaining {
  517                 child_normal_moves &= QUARTER_TURNS;
  518             }
  519             if child_normal_bound == child_remaining {
  520                 child_inverse_moves &= QUARTER_TURNS;
  521             }
  522             let result = self.search(
  523                 state,
  524                 child_cube,
  525                 child_inverse,
  526                 child_normal_bound,
  527                 child_inverse_bound,
  528                 child_normal_moves,
  529                 child_inverse_moves,
  530                 child_remaining,
  531             );
  532             if result == FOUND {
  533                 return FOUND;
  534             }
  535             if normal {
  536                 state.n_moves -= 1;
  537             } else {
  538                 state.n_premoves -= 1;
  539             }
  540             // Bound learned for the child: prunes its same-face siblings and
  541             // possibly this whole node.
  542             face_bound[face] = face_bound[face].max(result);
  543             lower_bound = lower_bound.max(result.saturating_sub(1));
  544             if lower_bound > remaining_depth {
  545                 return lower_bound;
  546             }
  547         }
  548         lower_bound
  549     }

  551     /// The last move of a search: is one of `moves` (allowed on both sides)
  552     /// a solution? (Nodes with one move left always move the normal cube.)
  553     fn last_move(state: &mut SearchState, cube: Cube, moves: u32) -> u8 {
  554         let mut todo = moves;
  555         while let Some(mv) = Move::lowest(todo) {
  556             todo ^= todo.isolate_lowest_one();
  557             state.stats.nodes += 1;
  558             if cube.apply_move(mv).is_solved() {
  559                 state.moves[state.n_moves] = mv as u8;
  560                 state.n_moves += 1;
  561                 return FOUND;
  562             }
  563         }
  564         0
  565     }
  566 
  567     /// Generate the children of a node: the `allowed` moves applied to
  568     /// `moved` (the node's cube of the chosen side), premoved on `other`.
  569     /// Returns them and their number.
  570     ///
  571     /// A half turn on this side does not change the other side's coset
  572     /// (every half turn lies in the subgroup whose cosets the table
  573     /// indexes), so its child keeps the node's other-side bound: children
  574     /// whose bound is already too high are not generated.
  575     #[inline(always)]
  576     fn generate_children(
  577         state: &mut SearchState,
  578         (moved, other): (Cube, Cube),
  579         allowed: u32,
  580         other_side_bound: u8,
  581         child_remaining: u8,
  582         face_bound: &mut [u8; 6],
  583     ) -> ([Child; 18], usize) {
  584         let mut children = [Child {
  585             moved,
  586             other,
  587             index: 0,
  588             mv: 0,
  589             moved_bound: 0,
  590             other_bound: 0,
  591             live: false,
  592         }; 18];
  593         let mut n_children = 0;
  594         let mut todo = allowed;
  595         while let Some(mv) = Move::lowest(todo) {
  596             todo ^= todo.isolate_lowest_one();
  597             let other_bound = if mv.index() % 3 == 1 {
  598                 other_side_bound
  599             } else {
  600                 UNKNOWN
  601             };
  602             if other_bound != UNKNOWN && other_bound > child_remaining {
  603                 face_bound[mv.index() / 3] = face_bound[mv.index() / 3].max(other_bound);
  604                 continue;
  605             }
  606             state.stats.nodes += 1;
  607             children[n_children] = Child {
  608                 moved: moved.apply_move(mv),
  609                 other: other.premove_inverse(mv),
  610                 index: 0,
  611                 mv: mv as u8,
  612                 moved_bound: 0,
  613                 other_bound,
  614                 live: true,
  615             };
  616             n_children += 1;
  617         }
  618         (children, n_children)
  619     }
  620 
  621     /// Look up the bounds of the children, pruning the ones that exceed
  622     /// `child_remaining` and noting each face's best bound. The lookups
  623     /// are pipelined: all indices are computed and prefetched before the
  624     /// first value is read (other side first, then the moved side).
  625     #[inline(always)]
  626     fn bound_children(
  627         &self,
  628         state: &mut SearchState,
  629         children: &mut [Child],
  630         face_bound: &mut [u8; 6],
  631         child_remaining: u8,
  632     ) {
  633         let max_value = self.base + 3;
  634         if child_remaining > max_value {
  635             // No table value can prune: skip the lookups.
  636             for child in children {
  637                 child.moved_bound = 0;
  638                 if child.other_bound == UNKNOWN {
  639                     child.other_bound = 0;
  640                 }
  641             }
  642             return;
  643         }
  644         // Stage 1: other-side index + prefetch.
  645         for child in &mut *children {
  646             if child.other_bound == UNKNOWN {
  647                 let (index, corner_bound) = self.index(child.other);
  648                 if corner_bound > child_remaining {
  649                     child.live = false;
  650                     let face = (child.mv / 3) as usize;
  651                     face_bound[face] = face_bound[face].max(corner_bound);
  652                     continue;
  653                 }
  654                 child.index = index;
  655                 child.other_bound = corner_bound; // temporarily the corner bound
  656                 self.prefetch(index);
  657                 child.moved_bound = UNKNOWN; // marks "other_bound pending"
  658             }
  659         }
  660         // Stage 2: other-side value, moved-side index + prefetch.
  661         for child in &mut *children {
  662             if !child.live {
  663                 continue;
  664             }
  665             let face = (child.mv / 3) as usize;
  666             if child.moved_bound == UNKNOWN {
  667                 state.stats.lookups += 1;
  668                 child.other_bound = self.value(child.index, child.other_bound);
  669                 if child.other_bound > child_remaining {
  670                     child.live = false;
  671                     face_bound[face] = face_bound[face].max(child.other_bound);
  672                     continue;
  673                 }
  674             }
  675             let (index, corner_bound) = self.index(child.moved);
  676             if corner_bound > child_remaining {
  677                 child.live = false;
  678                 face_bound[face] = face_bound[face].max(corner_bound);
  679                 continue;
  680             }
  681             child.index = index;
  682             child.moved_bound = corner_bound;
  683             self.prefetch(index);
  684         }
  685         // Stage 3: moved-side value.
  686         for child in &mut *children {
  687             if !child.live {
  688                 continue;
  689             }
  690             state.stats.lookups += 1;
  691             child.moved_bound = self.value(child.index, child.moved_bound);
  692             state.stat +=
  693                 (4096u64 >> child.other_bound.min(63)) + (4096u64 >> child.moved_bound.min(63));
  694             let face = (child.mv / 3) as usize;
  695             face_bound[face] = face_bound[face]
  696                 .max(child.moved_bound)
  697                 .max(child.other_bound);
  698             if child.moved_bound > child_remaining {
  699                 child.live = false;
  700             }
  701         }
  702     }
  703 }

  712     /// The hot-path index (`Solver::index`: `cocsep_x3`, packed entries,
  713     /// `index_sym`) equals `Cocsep::index` for every `k`, symmetric cubes too.
  714     #[test]
  715     #[cfg_attr(miri, ignore = "Cocsep::generate is too slow under Miri")]
  716     fn hot_index_matches_reference_index() {
  717         let corners = Cocsep::generate().unwrap();
  718         let mut rng = Rng(31);
  719         let mut cubes = vec![CubieCube::SOLVED];
  720         for scramble in [
  721             "U2 D2 R2 L2 F2 B2",
  722             "U D' R L' F B' U D'",
  723             "U R2 F B R B2 R U2 L B2 R U' D' R2 F R' L B2 U2 F2",
  724         ] {
  725             cubes.push(
  726                 CubieCube::SOLVED
  727                     .apply_moves(&parse_moves(scramble).unwrap())
  728                     .unwrap(),
  729             );
  730         }
  731         for _ in 0..40 {
  732             cubes.push(CubieCube::random(&mut rng));
  733         }
  734         // All conjugates and inverse conjugates (the index picks a symmetry).
  735         let variants: Vec<CubieCube> = cubes
  736             .iter()
  737             .flat_map(|c| [*c, c.inverse()])
  738             .flat_map(|c| symmetries().iter().map(move |s| conjugate(&c, s)))
  739             .collect();
  740         for k in [0, 7, 8, 11] {
  741             // (`index` does not read the table: a page stands in for it, so
  742             // that this also runs where h11 cannot be allocated, e.g. wasm32.)
  743             let table = H48Table {
  744                 memory: Memory::anonymous(4096).unwrap(),
  745                 meta: H48Meta { k, base: 10 },
  746             };
  747             let solver = Solver::new(&corners, &table);
  748             for c in &variants {
  749                 let cube = Cube::from_cubie(c);
  750                 let (index, corner_bound) = solver.index(cube);
  751                 assert_eq!(index, corners.index(cube, k), "k {k}: {c:?}");
  752                 assert_eq!(corner_bound, entry_depth(corners.entries[cocsep(cube)]));
  753             }
  754         }
  755     }
