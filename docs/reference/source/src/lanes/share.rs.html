    1 //! Cooperative batch and latency mode: the lanes of all threads share the
    2 //! work of a set of active cubes.
    3 //!
    4 //! In stream mode (`run_lanes`) every lane solves its own cube: ideal for an
    5 //! endless stream, but a finite batch waits for its slowest cubes, each in a
    6 //! single lane. Here:
    7 //!
    8 //! * A cube's search is the same sequence of iterations (target T) of sorted
    9 //!   jobs as in stream mode, but the job list lives in a shared `CubeCtx`.
   10 //!   Threads claim jobs in chunks (one atomic per chunk, smaller chunks as
   11 //!   the iteration runs out) and start them in their lanes at job
   12 //!   boundaries.
   13 //! * A cube's first iterations, up to the first one with 3-move jobs (16
   14 //!   with h11), run as one pass on those jobs: each job once per iteration,
   15 //!   claimed in the order of the iterations but without barriers between
   16 //!   them. A solution is reported once the iterations below its length have
   17 //!   all ended.
   18 //! * Iteration T+1 is built (jobs extended, sorted by the statistics of
   19 //!   iteration T) by the thread that finishes the last job of iteration T. A
   20 //!   solution found in iteration T is therefore optimal and is reported at
   21 //!   once; lanes still running jobs of the cube drop them (lanes are checked
   22 //!   once per round after a cube is solved). If threads are waiting for work,
   23 //!   a long job list is extended and sorted by all of them (`build_task`).
   24 //!   Jobs whose bounds exceed the iteration are sorted last and not claimed.
   25 //! * With few cubes in progress (at most threads / 8, e.g. one at a time),
   26 //!   builds are always staged and threads stay with the cubes they work on.
   27 //! * A lane whose job ends takes, in order: a piece donated to its thread,
   28 //!   another job of the same cube, a job its thread already claimed, a fresh
   29 //!   cube (at most `max_active` cubes are in progress), jobs of the oldest
   30 //!   cube with unclaimed jobs, or (while its thread has at most
   31 //!   `TAIL_LANES` active lanes) part of the remaining moves of a busy lane
   32 //!   of its thread: the upper half of the faces left at its shallowest node
   33 //!   with two faces left (cube48opt's `split_ida_node`). Otherwise it goes
   34 //!   idle.
   35 //! * A thread without active lanes asks for work in its mailbox; busy
   36 //!   threads check once per round whether some thread is waiting and donate
   37 //!   it a split of one of their lanes (a copy of the path to the split
   38 //!   node; the piece's slot is written and read in the order of the
   39 //!   mailbox states). Waiting threads spin briefly, then yield.
   40 //! * The job buffers of reported cubes are reused by new ones, the other
   41 //!   build buffers by later builds: builds touch no fresh pages.
   42 //!
   43 //! Each job runs exactly once; its statistic is the sum over its pieces. A
   44 //! cube's node count includes work done on it after its solution was found
   45 //! by another lane. The search kernel is the one of stream mode: `advance`
   46 //! of a `Lane::shared` lane only differs in returning at the end of a job,
   47 //! and everything here runs at job boundaries or once per round.
   48 //!
   49 //! Gains, 48 threads, h11, `random_state`: the mode itself (l5) took a batch
   50 //! from 7.86 ms to 1.26 ms per cube (bench random:1000 from 187 to 795
   51 //! solves/s), and one cube at a time from 306 ms (one lane) to 3.6 ms.
   52 //! l6-share-a and l6-share-2 brought one cube at a time to ~2.2 ms; c10
   53 //! (first pass, tail lanes, refills), c11 (touched buffers, split
   54 //! iterations) and m8 (unsafe pass, lanes by pointer) to ~1.6 ms. The
   55 //! comments at each optimization give its own measured gain.

  317 /// A reader-writer spin lock. The threads of a pool have cores of their
  318 /// own and hold it briefly: waiting must not sleep (with `std`'s lock,
  319 /// readers queued behind a writer sleep in the kernel, for tens of us).
  320 // Instead of std's RwLock: part of l6-share-a (-38.8% -c 1 latency at 48T).
  321 struct SpinRw<T> {
  322     /// Readers, plus `SPIN_W` while a writer holds or waits for the lock.
  323     state: AtomicU32,
  324     data: UnsafeCell<T>,
  325 }

  335 impl<T> SpinRw<T> {
  336     fn new(v: T) -> SpinRw<T> {
  337         SpinRw {
  338             state: AtomicU32::new(0),
  339             data: UnsafeCell::new(v),
  340         }
  341     }
  342 
  343     fn read(&self) -> SpinRead<'_, T> {
  344         let mut i = 0;
  345         loop {
  346             let state = self.state.load(Relaxed);
  347             if state & SPIN_W == 0
  348                 && self
  349                     .state
  350                     .compare_exchange_weak(state, state + 1, Acquire, Relaxed)
  351                     .is_ok()
  352             {
  353                 return SpinRead(self);
  354             }
  355             spin_wait(&mut i);
  356         }
  357     }
  358 
  359     /// (New readers wait while a writer waits for the others to leave.)
  360     fn write(&self) -> SpinWrite<'_, T> {
  361         let mut i = 0;
  362         while self.state.fetch_or(SPIN_W, Acquire) & SPIN_W != 0 {
  363             while self.state.load(Relaxed) & SPIN_W != 0 {
  364                 spin_wait(&mut i);
  365             }
  366         }
  367         while self.state.load(Acquire) != SPIN_W {
  368             spin_wait(&mut i);
  369         }
  370         SpinWrite(self)
  371     }
  372 }

  374 /// One wait of a spin lock: spins, then yields (if threads outnumber the
  375 /// cores, the holder may be waiting for one).
  376 // Yields after 256 spins: with pure spinning and more threads than cores,
  377 // the pool stalled (l6-share-a; a fix, not measured as a gain).
  378 #[inline(always)]
  379 fn spin_wait(i: &mut u32) {
  380     *i += 1;
  381     if *i < 256 {
  382         std::hint::spin_loop();
  383     } else {
  384         std::thread::yield_now();
  385     }
  386 }

  515 /// What a first pass does with a solution (`PassState::offer`).
  516 enum Offer {
  517     /// It is optimal (the phases below its own ended): report it.
  518     Report(Vec<u8>),
  519     /// The shortest so far: kept until the phases below it end.
  520     Candidate,
  521     /// One as short was found before: of no use.
  522     Useless,
  523 }
  524 
  525 impl PassState {
  526     /// A solution of phase `phase` was found (its entry has not ended).
  527     fn offer(&mut self, phase: u32, solution: Vec<u8>) -> Offer {
  528         // (Compared with the shortest one found, not with the candidate:
  529         // between `ended` taking it and its report, a solution of a later
  530         // phase, longer, could be reported first.)
  531         if self.shortest != 0 && phase + 1 >= self.shortest {
  532             return Offer::Useless;
  533         }
  534         self.shortest = phase + 1;
  535         if phase <= self.phases_ended {
  536             return Offer::Report(solution);
  537         }
  538         self.candidate = solution;
  539         Offer::Candidate
  540     }
  541 
  542     /// Count the leading phases (of `phases`) that ended (`ended(phase)`);
  543     /// returns the candidate, taken, if it is optimal now.
  544     fn ended(&mut self, phases: u32, ended: impl Fn(u32) -> bool) -> Option<Vec<u8>> {
  545         while self.phases_ended < phases && ended(self.phases_ended) {
  546             self.phases_ended += 1;
  547         }
  548         (!self.candidate.is_empty() && self.shortest <= self.phases_ended + 1)
  549             .then(|| std::mem::take(&mut self.candidate))
  550     }
  551 }

  599 impl CubeCtx {
  600     /// The current iteration. Only while the caller has outstanding jobs.
  601     #[inline(always)]
  602     fn inner(&self) -> &Inner {
  603         // SAFETY: the iteration cannot end (and be rebuilt) while one of
  604         // its jobs is outstanding.
  605         unsafe { &*self.inner.get() }
  606     }
  607 
  608     fn unclaimed(&self) -> u32 {
  609         let (count, next) = claim_fields(self.claim.load(Relaxed));
  610         count.saturating_sub(next)
  611     }
  612 
  613     fn task_unclaimed(&self) -> bool {
  614         let (_, count, next) = task_fields(self.task.load(Relaxed));
  615         next < count
  616     }
  617 
  618     /// Unclaimed jobs or build tasks.
  619     fn open(&self) -> bool {
  620         !self.done.load(Relaxed) && (self.unclaimed() > 0 || self.task_unclaimed())
  621     }
  622 
  623     /// Claim up to `chunk` jobs of the current iteration: (first, count).
  624     fn claim(&self, chunk: u32) -> Option<(u32, u32)> {
  625         let (count, next) = claim_fields(self.claim.load(Acquire));
  626         // (Checked first so that exhausted counters are not bumped forever.)
  627         if next >= count || self.done.load(Relaxed) {
  628             if self.trace && self.rarely_written.claims_ran_out_ns.load(Relaxed) == 0 {
  629                 let ns = u64::try_from(self.started.elapsed().as_nanos()).unwrap_or(u64::MAX);
  630                 let _ = self
  631                     .rarely_written
  632                     .claims_ran_out_ns
  633                     .compare_exchange(0, ns, Relaxed, Relaxed);
  634             }
  635             return None;
  636         }
  637         let (count, next) = claim_fields(self.claim.fetch_add(u64::from(chunk), AcqRel));
  638         (next < count).then(|| (next, chunk.min(count - next)))
  639     }
  640 }

  642 /// The job count and the next unclaimed job of a claim word.
  643 #[expect(clippy::cast_possible_truncation, reason = "the halves of the word")]
  644 fn claim_fields(word: u64) -> (u32, u32) {
  645     ((word >> 32) as u32, word as u32)
  646 }

  648 /// The stage, task count and next unclaimed task of a task word.
  649 #[expect(clippy::cast_possible_truncation, reason = "fields of the word")]
  650 fn task_fields(word: u64) -> (usize, usize, usize) {
  651     (
  652         (word >> 56) as usize,
  653         (word >> 32) as usize & 0xFF_FFFF,
  654         word as u32 as usize,

  658 /// `*stat += v` for f32 bits.
  659 fn add_f32(stat: &AtomicU32, v: f32) {
  660     let mut old = stat.load(Relaxed);
  661     while let Err(now) =
  662         stat.compare_exchange_weak(old, (f32::from_bits(old) + v).to_bits(), Relaxed, Relaxed)
  663     {
  664         old = now;
  665     }

  845     /// Work on the pool with `nlanes` interleaved lanes until all cubes are
  846     /// solved or `stop` is set. Calls `done(index, solution, stats,
  847     /// seconds)` (see `Done`) for the cubes whose last worker is this thread.
  848     pub fn run(
  849         &self,
  850         solver: &Solver,
  851         nlanes: usize,
  852         stop: &AtomicBool,
  853         mut done: impl FnMut(usize, Option<Vec<u8>>, Stats, f64),
  854     ) {
  855         let id = self.next_thread_id.fetch_add(1, Relaxed);
  856         if id < WARM_BUFFERS {
  857             let inner = Inner::touched();
  858             let mut bufs = Bufs::sized();
  859             bufs.pooled = true;
  860             self.spare.lock().push(inner);
  861             self.bufs.lock().push(bufs);
  862         }
  863         // (Built here, not by a `Worker::new`: that shrank this thread's
  864         // stack frame by 320 bytes, with `rounds` unchanged, and took solve
  865         // -c 1 on 17-move scrambles from 0.59 to 0.64 ms at 48T, depth-17
  866         // states from 0.65 to 0.72 ms (r8, 4 paired holds).)
  867         let mut worker = Worker {
  868             pool: self,
  869             solver,
  870             id,
  871             done: &mut done,
  872             lanes: (0..nlanes.max(1)).map(|_| SharedLane::new()).collect(),
  873             active_lanes: 0,
  874             locals: Vec::new(),
  875             scratch: Box::new(Lane::new()),
  876             bufs: Bufs::own(),
  877             epoch: 0,
  878             solved: 0,
  879             split_wait: 0,
  880             split_backoff: 0,
  881             donate_wait: 0,
  882             donate_backoff: 0,
  883             counts: [0; NCOUNT],
  884         };
  885         worker.run(stop);
  886         let mut counters = self.counters.lock();
  887         for (sum, count) in counters.iter_mut().zip(worker.counts) {
  888             *sum += count;
  889         }
  890     }

  910     /// Make sure `ctx` is on the open list (new jobs or build tasks).
  911     fn publish(&self, ctx: &Arc<CubeCtx>) {
  912         let mut open = self.open.write();
  913         // Drop the cubes with nothing left to claim (else nobody may: when
  914         // no lane is idle).
  915         if open.len() >= 64 && open.len().is_power_of_two() {
  916             open.retain(keep_open);
  917         }
  918         // (`listed` changes only under the write lock.)
  919         if !ctx.rarely_written.listed.load(Relaxed) {
  920             ctx.rarely_written.listed.store(true, Relaxed);
  921             open.push(ctx.clone());
  922         }
  923         self.open_count.store(open.len(), Relaxed);
  924         drop(open);
  925         self.epoch.fetch_add(1, Release);
  926     }

  928     /// The oldest cube with unclaimed jobs (`task`: build tasks).
  929     fn find_open(&self, task: bool) -> Option<Arc<CubeCtx>> {
  930         if self.open_count.load(Relaxed) == 0 {
  931             return None;
  932         }
  933         let open = self.open.read();
  934         let mut dead = 0;
  935         let mut found = None;
  936         for ctx in open.iter() {
  937             if !ctx.open() {
  938                 dead += 1;
  939                 continue;
  940             }
  941             let wanted = if task {
  942                 ctx.task_unclaimed()
  943             } else {
  944                 ctx.unclaimed() > 0
  945             };
  946             if wanted {
  947                 found = Some(ctx.clone());
  948                 break;
  949             }
  950         }
  951         drop(open);
  952         if dead > 0 {
  953             // (Not a try: under the read locks of the threads looking for
  954             // work it would rarely succeed, and the dead cubes would keep
  955             // them all taking the lock. Part of l6-share-a: -38.8% -c 1
  956             // latency at 48T.)
  957             let mut open = self.open.write();
  958             open.retain(keep_open);
  959             self.open_count.store(open.len(), Relaxed);
  960         }
  961         found
  962     }

 1023 /// Solve all cubes of `cubes` (until it returns None) with `threads` threads
 1024 /// of `nlanes` lanes, at most `max_active` cubes at a time; `done` is called
 1025 /// once per cube (from any thread; see `Done`).
 1026 ///
 1027 /// # Errors
 1028 /// If a thread cannot be started (the others are stopped), or panicked.
 1029 pub fn solve_all(
 1030     solver: &Solver,
 1031     cubes: &(dyn Fn(usize) -> Option<CubieCube> + Sync),
 1032     threads: usize,
 1033     nlanes: usize,
 1034     max_active: usize,
 1035     done: &(dyn Fn(usize, Option<Vec<u8>>, Stats, f64) + Sync),
 1036 ) -> std::io::Result<()> {
 1037     let pool = Pool::new(cubes, max_active, threads);
 1038     let stop = AtomicBool::new(false);
 1039     std::thread::scope(|scope| {
 1040         let mut spawned = Vec::with_capacity(threads.max(1));
 1041         let mut result = Ok(());
 1042         for _ in 0..threads.max(1) {
 1043             match std::thread::Builder::new()
 1044                 .spawn_scoped(scope, || pool.run(solver, nlanes, &stop, done))
 1045             {
 1046                 Ok(thread) => spawned.push(thread),
 1047                 Err(error) => {
 1048                     stop.store(true, Relaxed);
 1049                     result = Err(error);
 1050                     break;
 1051                 }
 1052             }
 1053         }
 1054         for thread in spawned {
 1055             if thread.join().is_err() {
 1056                 result = Err(std::io::Error::other("a share thread panicked"));
 1057             }
 1058         }
 1059         result
 1060     })
 1061 }

 1165 /// Build the next iteration of `ctx` (the first one if `first`) with
 1166 /// `scratch`'s extension, and publish it. If threads are waiting for work, a
 1167 /// long job list is extended and sorted by all of them instead
 1168 /// (`build_task`): returns true if that was started (the caller then helps).
 1169 fn build(
 1170     scratch: &mut Lane,
 1171     own_bufs: &mut Bufs,
 1172     solver: &Solver,
 1173     pool: &Pool,
 1174     ctx: &Arc<CubeCtx>,
 1175     first: bool,
 1176 ) -> bool {
 1177     let now = Instant::now();
 1178     // SAFETY: no job of the cube is outstanding: nobody else accesses
 1179     // `inner` until the iteration (or a build task) is published.
 1180     let inner = unsafe { &mut *ctx.inner.get() };
 1181     if pool.trace && !first {
 1182         trace_iteration(pool, ctx, inner, now);
 1183     }
 1184     inner.build_started = now;
 1185     inner.extended = now;
 1186     // (The next target, after the first.)
 1187     inner.target += u8::from(!first);
 1188     // The iterations up to the first one with jobs of `FIRST_PASS_JOB_LEN`
 1189     // moves (from a target at most `first_pass_span` below) run as one pass
 1190     // on its jobs, without barriers (`Inner::phases`): the ones below it
 1191     // are cheap and nearly never the last (17 of 1000 random states end in
 1192     // them), but their barriers, tails and builds each took ~0.03-0.12 ms of
 1193     // -c 1 at 48 threads. As one pass: -7.4% -c 1, -7.1% -c 0, +2.0% bench
 1194     // share at 48T (8 paired holds); 15/16/17-move scrambles -18/-43/-22%.
 1195     // (A job prefix cannot solve the cube: the target is a lower bound above
 1196     // the prefix length.)
 1197     let pass_target = solver.base + 3 + FIRST_PASS_JOB_LEN;
 1198     inner.phases = 1;
 1199     if inner.target < pass_target
 1200         && pass_target <= MAX_TARGET
 1201         && inner.target + pool.first_pass_span >= pass_target
 1202         && usize::from(inner.target) > solver.job_len(pass_target)
 1203     {
 1204         inner.phases = u32::from(pass_target - inner.target) + 1;
 1205         inner.target = pass_target;
 1206     }
 1207     inner.sort_by_bounds = first;
 1208     // The stack depth stays below the target, at most `MAX_TARGET` (< `MAXD`; only
 1209     // a first bound above it is lowered: nothing is found, `settle` gives up).
 1210     inner.target = inner.target.min(MAX_TARGET);
 1211     let target = inner.target;
 1212     // Stream mode's prefixes: 4-move prefixes from iteration 15 on cost +8%
 1213     // -c 1 latency at 48T (l6-share-2).
 1214     let job_len = solver.job_len(target);
 1215     // (With few cubes, threads are about to wait for this one.) Staging the
 1216     // builds of few cubes before anyone waits: part of l6-share-2 (-5.6% -c 1
 1217     // latency at 48T for the series; not measured alone).
 1218     let parallel = pool.few() || pool.hungry.load(Relaxed) != 0;
 1219     let len = inner.jobs[0].len as usize;
 1220     // Sorted by the statistics as they are (not copied to the jobs first)?
 1221     // (The serial copy took ~90 us per build; part of l6-share-a's -38.8% -c 1
 1222     // latency at 48T, not measured alone.)
 1223     let from_atomics =
 1224         !first && len >= job_len && job_len > 0 && parallel && inner.jobs.len() >= SORT_MIN;
 1225     plan_leads(
 1226         inner,
 1227         first,
 1228         job_len == MAX_JOB_LEN && parallel,
 1229         from_atomics,
 1230     );
 1231     if !first && !from_atomics {
 1232         copy_stats(inner);
 1233     }
 1234     if len < job_len {
 1235         // (`scratch` gets the cube's buffer and ours, and returns them.)
 1236         scratch.root = ctx.root;
 1237         scratch.root_inverse = ctx.root_inverse;
 1238         scratch.run_state.sym_filter = ctx.sym_filter;
 1239         std::mem::swap(&mut scratch.jobs, &mut inner.jobs);
 1240         std::mem::swap(&mut scratch.jobs_tmp, &mut own_bufs.out);
 1241         // Moves left to add, in parallel. (Also for a first pass, and with a
 1242         // symmetric-start filter that only `keep`s: the tasks then count the
 1243         // kept children. On x-h48's 7 test positions with only a symmetry:
 1244         // -9.9% -c 1 at 48T against extending them serially, median of 4
 1245         // paired rounds of 70 cubes. Not with a filter that dedups, whose
 1246         // classes span the tasks; such cubes, like the superflip, keep few
 1247         // jobs.)
 1248         let parallel_extension = parallel && !ctx.sym_filter.dedups();
 1249         let moves = loop {
 1250             let (n, moves) = (scratch.jobs.len(), job_len - scratch.jobs[0].len as usize);
 1251             if moves == 0
 1252                 || parallel_extension && (moves == 1 && n >= EXT_MIN || moves == 2 && n >= EXT_MIN2)
 1253             {
 1254                 break moves;
 1255             }
 1256             scratch.extend_jobs(solver);
 1257         };
 1258         std::mem::swap(&mut scratch.jobs, &mut inner.jobs);
 1259         std::mem::swap(&mut scratch.jobs_tmp, &mut own_bufs.out);
 1260         if moves > 0 {
 1261             inner.b = pool.take_bufs();
 1262             // Parallel extension: task c (a chunk of the parents) writes
 1263             // their children (`moves` moves longer) to
 1264             // `out[child_offsets[c]..child_offsets[c + 1]]`, in the order of
 1265             // serial `extend_jobs`.
 1266             let bufs = &mut inner.b;
 1267             bufs.extension_moves = moves;
 1268             bufs.child_offsets.clear();
 1269             let mut offset = 0u32;
 1270             let chunk = if moves == 1 { EXT_CHUNK } else { 1 };
 1271             for parents in inner.jobs.chunks(chunk) {
 1272                 bufs.child_offsets.push(offset);
 1273                 for parent in parents {
 1274                     offset += kept_children(parent, moves, &ctx.sym_filter);
 1275                 }
 1276             }
 1277             bufs.child_offsets.push(offset);
 1278             bufs.out.clear();
 1279             bufs.out.reserve(offset as usize); // (the tasks write via `out_ptr`)
 1280             bufs.out_ptr = TaskPtr(bufs.out.as_mut_ptr());
 1281             open_stage(pool, ctx, ST_EXT, bufs.child_offsets.len() - 1);
 1282             return true;
 1283         }
 1284     }
 1285     inner.extended = Instant::now();
 1286     if inner.phases > 1 {
 1287         first_pass_order(inner);
 1288         finish_build(pool, ctx);
 1289         return false;
 1290     }
 1291     if job_len > 0 && inner.jobs.len() >= SORT_MIN && parallel {
 1292         if inner.sort_by_bounds {
 1293             stats_from_bounds(&mut inner.jobs);
 1294         }
 1295         inner.b = pool.take_bufs();
 1296         inner.b.stats_from_atomics = from_atomics;
 1297         start_sort(pool, ctx);
 1298         return true;
 1299     }
 1300     if inner.sort_by_bounds {
 1301         stats_from_bounds(&mut inner.jobs);
 1302     }
 1303     // (Serial builds use this thread's buffers: those of the pool were last
 1304     // written by other threads.)
 1305     inner.feasible = if job_len > 0 {
 1306         sort_serial(&mut inner.jobs, own_bufs, target)
 1307     } else {
 1308         inner.jobs.len()
 1309     };
 1310     finish_build(pool, ctx);
 1311     false
 1312 }

 1315 /// finished, its run (ms from the cube's start), when its claims ran out
 1316 /// (and its nodes, jobs, donations and splits).

 1318 #[expect(clippy::cast_precision_loss, reason = "trace output")]
 1319 fn trace_iteration(pool: &Pool, ctx: &CubeCtx, inner: &Inner, now: Instant) {
 1320     let rarely_written = &ctx.rarely_written;
 1321     let ran_out_ms = rarely_written.claims_ran_out_ns.swap(0, Relaxed) as f64 * 1e-6;
 1322     let nodes = rarely_written.iteration_nodes.swap(0, Relaxed);
 1323     let seconds = (now - inner.published).as_secs_f64();
 1324     // Threads' finish times in this iteration (ms), sorted.
 1325     let published_ns = (inner.published - ctx.started).as_nanos();
 1326     let mut finish: Vec<f64> = ctx
 1327         .thread_done_ns
 1328         .iter()
 1329         .map(|t| t.swap(0, Relaxed))
 1330         .filter(|&t| u128::from(t) > published_ns)
 1331         .map(|t| t as f64 * 1e-6)
 1332         .collect();
 1333     finish.sort_by(f64::total_cmp);
 1334     let quantile = |percent: usize| {
 1335         finish
 1336             .get(finish.len().saturating_sub(1) * percent / 100)
 1337             .copied()
 1338             .unwrap_or(0.0)
 1339     };
 1340     tlog(
 1341         pool,
 1342         format_args!(
 1343             "trace cube {} T {} finish threads {} q10 {:.3} q50 {:.3} q90 {:.3} max {:.3}",

 1346             finish.len(),
 1347             quantile(10),
 1348             quantile(50),
 1349             quantile(90),
 1350             quantile(100)
 1351         ),
 1352     );
 1353     let counters = if pool.trace_bits & TRACE_COUNTERS == 0 {
 1354         String::new()
 1355     } else {
 1356         format!(
 1357             ", nodes {nodes} jobs {} donated {} splits {} ({:.2} Gnodes/s)",
 1358             rarely_written.iteration_jobs.swap(0, Relaxed),
 1359             rarely_written.iteration_donations.swap(0, Relaxed),
 1360             rarely_written.iteration_splits.swap(0, Relaxed),
 1361             nodes as f64 / seconds * 1e-9
 1362         )
 1363     };
 1364     tlog(
 1365         pool,
 1366         format_args!(
 1367             "trace cube {} T {} ran {:.3} .. {:.3} ms, claims out at {ran_out_ms:.3}{counters}",
 1368             ctx.index,
 1369             inner.target,
 1370             (inner.published - ctx.started).as_secs_f64() * 1e3,
 1371             (now - ctx.started).as_secs_f64() * 1e3,
 1372         ),
 1373     );
 1374 }

 1453 /// The lead of a job whose run over its root children `children` (normal
 1454 /// side) gave `child_stats`: the code (`quantize`) of its best child's
 1455 /// statistic, weighted one code step up on faces 1, 3 and 5 (`ODD_FACE`, as
 1456 /// `quantize18`), and its children whose weighted statistic is at least
 1457 /// `LEAD_SHARE` of the best one's, with the X or X' sibling of each
 1458 /// (`with_twins`). 0 (none) if no child has a statistic. On the statistics,
 1459 /// not their codes: a few µops per job end.
 1460 #[cfg(avx512)]
 1461 #[expect(clippy::cast_possible_truncation, reason = "the low 16 moves")]
 1462 fn lead_of(child_stats: &[f32; 18], children: u32, tag: u32) -> u32 {
 1463     use std::arch::x86_64::{
 1464         _mm512_loadu_ps, _mm512_mask_cmp_ps_mask, _mm512_maskz_mul_ps, _mm512_reduce_max_ps,
 1465         _mm512_set1_ps, _mm512_setr_ps, _CMP_GE_OQ,
 1466     };
 1467     const ODD: f32 = ODD_FACE;
 1468     let low = children as u16;
 1469     // SAFETY: AVX-512 is a target feature (`cfg(avx512)`); the load reads
 1470     // the first 16 statistics.
 1471     let (weighted, low_best) = unsafe {
 1472         let weights = _mm512_setr_ps(
 1473             1.0, 1.0, 1.0, ODD, ODD, ODD, 1.0, 1.0, 1.0, ODD, ODD, ODD, 1.0, 1.0, 1.0, ODD,
 1474         );
 1475         let weighted = _mm512_maskz_mul_ps(low, _mm512_loadu_ps(child_stats.as_ptr()), weights);
 1476         (weighted, _mm512_reduce_max_ps(weighted))
 1477     };
 1478     // (Moves 16 and 17 are on face 5.)
 1479     let high = |mv: usize| {
 1480         if children >> mv & 1 != 0 {
 1481             child_stats[mv] * ODD
 1482         } else {
 1483             0.0
 1484         }
 1485     };
 1486     let (stat16, stat17) = (high(16), high(17));
 1487     let best = low_best.max(stat16).max(stat17);
 1488     let code = super::quantize(best);
 1489     if code == 0 {
 1490         return 0;
 1491     }
 1492     let floor = best * LEAD_SHARE;
 1493     // SAFETY: as above.
 1494     let low_group =
 1495         unsafe { _mm512_mask_cmp_ps_mask::<_CMP_GE_OQ>(low, weighted, _mm512_set1_ps(floor)) };
 1496     let group =
 1497         u32::from(low_group) | u32::from(stat16 >= floor) << 16 | u32::from(stat17 >= floor) << 17;
 1498     lead_word(tag, code, with_twins(group) & children)
 1499 }

 1501 /// `lead_of` without AVX-512.
 1502 #[cfg(not(avx512))]
 1503 fn lead_of(child_stats: &[f32; 18], children: u32, tag: u32) -> u32 {
 1504     let weighted: [f32; 18] = std::array::from_fn(|mv| {
 1505         let weight = if mv / 3 % 2 == 1 { ODD_FACE } else { 1.0 };
 1506         if children >> mv & 1 != 0 {
 1507             child_stats[mv] * weight
 1508         } else {
 1509             0.0
 1510         }
 1511     });
 1512     let best = weighted.iter().copied().fold(0.0, f32::max);
 1513     let code = super::quantize(best);
 1514     if code == 0 {
 1515         return 0;
 1516     }
 1517     let floor = best * LEAD_SHARE;
 1518     let group = (0..18)
 1519         .filter(|&mv| children >> mv & 1 != 0 && weighted[mv] >= floor)
 1520         .fold(0, |group, mv| group | 1 << mv);
 1521     lead_word(tag, code, with_twins(group) & children)
 1522 }

 1528 /// `group` with the X and X' children of each face where it has one of
 1529 /// them (an X' child runs without its X sibling only with X's other-side
 1530 /// value, which a separate run lacks).
 1531 fn with_twins(group: u32) -> u32 {
 1532     let x_moves = 0b001_001_001_001_001_001;
 1533     group | (group & x_moves) << 2 | group >> 2 & x_moves

 1536 /// Statistics of new jobs (no iteration ran them yet) for the sort of a
 1537 /// first build that is not a first pass, from their bounds: the largest
 1538 /// bound lowest first (the largest trees), then the smallest.

 1550 /// The number of jobs `moves` (at most 2) moves longer than `job`
 1551 /// (`extend_jobs`).
 1552 fn nchildren(job: &Job, moves: usize) -> u32 {
 1553     /// By the face of the last move (6: none), and moves added.
 1554     static COUNTS: [[u32; 3]; 7] = {
 1555         let mut counts = [[1u32; 3]; 7];
 1556         let mut face = 0;
 1557         while face < 7 {
 1558             let allowed = if face == 6 {
 1559                 ALL_MOVES
 1560             } else {
 1561                 ALLOWED_AFTER[face]
 1562             };
 1563             counts[face][1] = allowed.count_ones();
 1564             counts[face][2] = 0;
 1565             let mut m = 0;
 1566             while m < 18 {
 1567                 if allowed & (1 << m) != 0 {
 1568                     counts[face][2] += ALLOWED_AFTER[m / 3].count_ones();
 1569                 }
 1570                 m += 1;
 1571             }
 1572             face += 1;
 1573         }
 1574         counts
 1575     };
 1576     let face = if job.len == 0 {
 1577         6
 1578     } else {
 1579         (job.moves[job.len as usize - 1] / 3) as usize
 1580     };
 1581     COUNTS[face][moves]
 1582 }

 1595 /// The number of extensions of `prefix[..len]` by `moves` moves that

 1599 fn count_kept(prefix: &mut [u8; 4], len: usize, moves: usize, filter: &SymFilter) -> u32 {
 1600     if moves == 0 {
 1601         return 1;
 1602     }
 1603     let mut todo = if len == 0 {
 1604         ALL_MOVES
 1605     } else {
 1606         ALLOWED_AFTER[usize::from(prefix[len - 1] / 3)]
 1607     };
 1608     let mut count = 0;
 1609     while todo != 0 {
 1610         prefix[len] = todo.trailing_zeros() as u8;
 1611         todo ^= todo.isolate_lowest_one();
 1612         if filter.keep(&prefix[..=len]) {
 1613             count += count_kept(prefix, len + 1, moves - 1, filter);
 1614         }
 1615     }
 1616     count
 1617 }

 1619 /// Sort `jobs` for iteration `target` with `bufs` (an LSD radix sort
 1620 /// of the keys of `sort_key`, then a gather); returns the number of
 1621 /// feasible jobs.
 1622 #[expect(clippy::cast_possible_truncation, reason = "11-bit digits")]
 1623 fn sort_serial(jobs: &mut Vec<Job>, bufs: &mut Bufs, target: u8) -> usize {
 1624     let n = jobs.len();
 1625     let (keys, keys_tmp) = (&mut bufs.keys, &mut bufs.keys_tmp);
 1626     keys.clear();
 1627     keys.extend((0..n).map(|i| sort_key(jobs, &[], false, target, i)));
 1628     keys_tmp.clear();
 1629     keys_tmp.resize(n, 0);
 1630     // (The indices, in the low bits, are in order.)
 1631     for (shift, bits) in [
 1632         (KEY_SHIFT, 11u32),
 1633         (KEY_SHIFT + 11, 11),
 1634         (KEY_SHIFT + 22, 11),
 1635     ] {
 1636         let mask = (1u64 << bits) - 1;
 1637         let mut counts = [0u32; 2048];
 1638         for &key in keys.iter() {
 1639             counts[((key >> shift) & mask) as usize] += 1;
 1640         }
 1641         if counts.iter().any(|&count| count as usize == n) {
 1642             continue; // all keys share this digit
 1643         }
 1644         let mut sum = 0;
 1645         for count in &mut counts {
 1646             (*count, sum) = (sum, sum + *count);
 1647         }
 1648         for &key in keys.iter() {
 1649             let digit = ((key >> shift) & mask) as usize;
 1650             keys_tmp[counts[digit] as usize] = key;
 1651             counts[digit] += 1;
 1652         }
 1653         std::mem::swap(keys, keys_tmp);
 1654     }
 1655     bufs.out.clear();
 1656     bufs.out
 1657         .extend(keys.iter().map(|&key| jobs[key_index(key)]));
 1658     std::mem::swap(jobs, &mut bufs.out);
 1659     keys.partition_point(|&key| key >> 63 == 0)
 1660 }

 1663 const ST_EXT: usize = 1;
 1664 const ST_KEYS: usize = 2;
 1665 const ST_SCATTER: usize = 3;
 1666 const ST_BUCKETS: usize = 4;
 1667 /// Parallel sort: jobs per chunk, buckets (between splitters taken from a
 1668 /// sample of `SAMPLE` keys per bucket).
 1669 const SORT_CHUNK: usize = 2048;
 1670 const NBUCKETS: usize = 64;
 1671 const SAMPLE: usize = 16;

 1673 /// Open build stage `stage` with `tasks` tasks (and list the cube: it may have
 1674 /// been dropped from the open list while the last stage's tasks ran).
 1675 // Listing the cube at every stage lets all threads run its stages: part of
 1676 // l6-share-a (-38.8% -c 1 latency at 48T).
 1677 #[expect(clippy::cast_possible_truncation, reason = "fewer than 2^24 tasks")]
 1678 fn open_stage(pool: &Pool, ctx: &Arc<CubeCtx>, stage: usize, tasks: usize) {
 1679     ctx.task_left.store(tasks as u32, Relaxed);
 1680     ctx.task
 1681         .store((stage as u64) << 56 | (tasks as u64) << 32, Release);
 1682     pool.publish(ctx);
 1683 }

 1685 /// Set up the parallel sort of `inner.jobs`, the same order as
 1686 /// `sort_jobs`: by `!job_key << 32 | index`. Keys are counted per chunk by
 1687 /// their top bits (`ST_KEYS`), scattered to their buckets (`ST_SCATTER`),
 1688 /// and runs of buckets are sorted and gathered (`ST_BUCKETS`).
 1689 fn start_sort(pool: &Pool, ctx: &Arc<CubeCtx>) {
 1690     // SAFETY: the previous stage has ended. Do not take `inner` as a
 1691     // reference argument: its function-entry borrow would remain protected
 1692     // while another worker can claim the stage published below.
 1693     let chunks = sort_setup(unsafe { &mut *ctx.inner.get() });
 1694     open_stage(pool, ctx, ST_KEYS, chunks);
 1695 }

 1697 /// Sort key of job `i` in iteration `target` (with the statistic from `stats` if
 1698 /// `from_atomics`), in increasing order: the jobs whose bounds exceed the
 1699 /// iteration (they would be skipped) last, then by decreasing `job_key`
 1700 /// (as `sort_jobs`), then by index: `infeasible << 63 | !job_key << 31 |
 1701 /// i`. The feasible jobs are in the order of `sort_jobs`, and only they
 1702 /// are claimed.
 1703 #[inline(always)]
 1704 fn sort_key(jobs: &[Job], stats: &[AtomicU32], from_atomics: bool, target: u8, i: usize) -> u64 {
 1705     let job = &jobs[i];
 1706     let stat = if from_atomics {
 1707         f32::from_bits(stats[i].load(Relaxed))
 1708     } else {
 1709         job.stat
 1710     };
 1711     let infeasible =
 1712         u32::from(job.normal_bound.max(job.inverse_bound)) + u32::from(job.len) > u32::from(target);
 1713     u64::from(infeasible) << 63
 1714         | u64::from(!(stat * job_weight(job)).to_bits()) << KEY_SHIFT
 1715         | i as u64
 1716 }
 1717 
 1718 const KEY_SHIFT: u32 = 31;
 1719 
 1720 #[inline(always)]
 1721 fn key_index(k: u64) -> usize {
 1722     (k & ((1 << KEY_SHIFT) - 1)) as usize
 1723 }

 1725 /// Splitters of the buckets and buffers of a parallel sort; returns the
 1726 /// number of `ST_KEYS` tasks.
 1727 fn sort_setup(inner: &mut Inner) -> usize {
 1728     let n = inner.jobs.len();
 1729     let chunks = n.div_ceil(SORT_CHUNK);
 1730     let sampled = |x: usize| x * n / (NBUCKETS * SAMPLE);
 1731     let from_atomics = inner.b.stats_from_atomics;
 1732     for x in 0..NBUCKETS * SAMPLE {
 1733         // (The jobs, or statistics, were just written by other threads.)
 1734         prefetch(&inner.jobs[sampled(x)], false);
 1735         if from_atomics {
 1736             prefetch(&inner.stats[sampled(x)], false);
 1737         }
 1738     }
 1739     if inner.split {
 1740         for x in 0..NBUCKETS * SAMPLE {
 1741             prefetch(&inner.leads[sampled(x)], false);
 1742         }
 1743         inner.lead_threshold = lead_threshold(inner, n);
 1744     }
 1745     let bufs = &mut inner.b;
 1746     let mut sample = std::mem::take(&mut bufs.splitters);
 1747     sample.clear();
 1748     sample.extend((0..NBUCKETS * SAMPLE).map(|x| {
 1749         sort_key(
 1750             &inner.jobs,
 1751             &inner.stats,
 1752             from_atomics,
 1753             inner.target,
 1754             sampled(x),
 1755         )
 1756     }));
 1757     sample.sort_unstable();
 1758     sample.dedup();
 1759     let len = sample.len();
 1760     for i in 1..NBUCKETS {
 1761         sample[i - 1] = sample[i * len / NBUCKETS];
 1762     }
 1763     sample.truncate(NBUCKETS - 1);
 1764     bufs.splitters = sample;
 1765     bufs.bucket_counts.clear();
 1766     bufs.bucket_counts.resize(chunks * NBUCKETS, 0);
 1767     // (Room for `n` entries in each buffer that the tasks write through its
 1768     // pointer: the buffers are sized for them already.)
 1769     for keys in [&mut bufs.keys, &mut bufs.keys_tmp] {
 1770         keys.clear();
 1771         keys.reserve(n);
 1772     }
 1773     bufs.key_buckets.clear();
 1774     bufs.key_buckets.reserve(n);
 1775     bufs.out.clear();
 1776     bufs.out.reserve(n);
 1777     bufs.keys_ptr = TaskPtr(bufs.keys.as_mut_ptr());
 1778     bufs.keys_tmp_ptr = TaskPtr(bufs.keys_tmp.as_mut_ptr());
 1779     bufs.key_buckets_ptr = TaskPtr(bufs.key_buckets.as_mut_ptr());
 1780     bufs.bucket_counts_ptr = TaskPtr(bufs.bucket_counts.as_mut_ptr());
 1781     bufs.out_ptr = TaskPtr(bufs.out.as_mut_ptr());
 1782     if inner.split {
 1783         bufs.lead_counts.clear();
 1784         bufs.lead_counts.resize(chunks * LEAD_BUCKETS, 0);
 1785         bufs.rest_children.clear();
 1786         bufs.rest_children.reserve(n);
 1787         inner.runs.clear();
 1788         inner.runs.reserve(2 * n);
 1789         bufs.rest_children_ptr = TaskPtr(bufs.rest_children.as_mut_ptr());
 1790         bufs.lead_counts_ptr = TaskPtr(bufs.lead_counts.as_mut_ptr());
 1791         bufs.runs_ptr = TaskPtr(inner.runs.as_mut_ptr());
 1792     }
 1793     chunks
 1794 }

 1796 /// The lead code above which a split iteration's `n` jobs have a first-part
 1797 /// run: about `LEAD_FRACTION` of those with a lead, from the sample of
 1798 /// `sort_setup`.
 1799 #[expect(
 1800     clippy::cast_possible_truncation,
 1801     clippy::cast_sign_loss,
 1802     reason = "a fraction of a count"
 1803 )]
 1804 fn lead_threshold(inner: &Inner, n: usize) -> u8 {
 1805     let tag = inner.read_tag;
 1806     let mut histogram = [0u32; 256];
 1807     let mut total = 0u32;
 1808     for x in 0..NBUCKETS * SAMPLE {
 1809         let i = x * n / (NBUCKETS * SAMPLE);
 1810         if let Some((code, _)) = lead_fields(inner.leads[i].load(Relaxed), tag) {
 1811             histogram[usize::from(code)] += 1;
 1812             total += 1;
 1813         }
 1814     }
 1815     let limit = (f64::from(total) * LEAD_FRACTION) as u32;
 1816     let mut count = 0;
 1817     for code in (1..=u8::MAX).rev() {
 1818         count += histogram[usize::from(code)];
 1819         if count > limit {
 1820             return code;
 1821         }
 1822     }
 1823     u8::MAX
 1824 }

 1826 /// Set up lane `sl`'s run (claim entry `entry`) in an iteration that is
 1827 /// split or records leads: its root children `children` (all of them
 1828 /// except in a split job), and whether it records its job's lead (whole
 1829 /// runs and first-part runs do).
 1830 #[inline(never)]
 1831 fn lead_run(sl: &mut SharedLane, inner: &Inner, entry: u32, children: u32) {
 1832     if children != ALL_MOVES {
 1833         // (The children are normal-side moves: a lead is recorded on the
 1834         // normal side, and the next iteration has the root there too, its
 1835         // bounds being below the remaining depth. A job's runs cover all
 1836         // its children in any case.)
 1837         sl.lane.stack[0].todo &= children;
 1838         // (A job's runs may end at once: its statistic is added.)
 1839         sl.split = true;
 1840     }
 1841     sl.lead_children = if children == ALL_MOVES || entry < inner.first_runs {
 1842         children
 1843     } else {
 1844         0
 1845     };
 1846 }

 1861 /// Whether the new iteration of `inner` (the cube's first if `first`)
 1862 /// records its jobs' leads (`record`: its jobs have their full length, and
 1863 /// it is built in stages), and whether it is split (`sorted_in_stages`: its
 1864 /// jobs are those of the last iteration, which recorded them). With h11: T = 17 records, T = 18 (the
 1865 /// last iteration for 95% of the random states that reach it) is split and
 1866 /// records, T = 19 is split, and so on.
 1867 // Split iterations (c11): -2.9% -c 1 at 48T (7 paired holds, nodes -6.8%;
 1868 // -c 0, bench share and the short sets neutral; 1T neutral).
 1869 fn plan_leads(inner: &mut Inner, first: bool, record: bool, sorted_in_stages: bool) {
 1870     if first {
 1871         inner.leads_target = 0;
 1872     }
 1873     let has_leads = !inner.leads.is_empty();
 1874     inner.split = sorted_in_stages && has_leads && inner.leads_target + 1 == inner.target;
 1875     inner.read_tag = lead_tag(inner.tag_base, inner.leads_target);
 1876     inner.record_leads = !first && record && has_leads && inner.phases == 1;
 1877     if inner.record_leads {
 1878         inner.leads_target = inner.target;
 1879         inner.write_tag = lead_tag(inner.tag_base, inner.target);
 1880     }
 1881 }

 1889 /// Task `task` of sort stage `stage`.
 1890 ///
 1891 /// # Safety
 1892 /// Tasks of a stage run concurrently only with each other, after
 1893 /// `sort_setup` and the previous stages.
 1894 #[expect(clippy::cast_possible_truncation, reason = "bucket indices < NBUCKETS")]
 1895 unsafe fn sort_task(inner: &Inner, stage: usize, task: usize) {
 1896     let jobs = &inner.jobs;
 1897     let bufs = &inner.b;
 1898     match stage {
 1899         ST_KEYS => {
 1900             let lo = task * SORT_CHUNK;
 1901             let hi = (lo + SORT_CHUNK).min(jobs.len());
 1902             let (keys, key_buckets) = (bufs.keys_ptr.0, bufs.key_buckets_ptr.0);
 1903             let mut counts = [0u32; NBUCKETS];
 1904             let mut lead_counts = [0u32; LEAD_BUCKETS];
 1905             for i in lo..hi {
 1906                 let mut key =
 1907                     sort_key(jobs, &inner.stats, bufs.stats_from_atomics, inner.target, i);
 1908                 if inner.split {
 1909                     // A first-part run for a lead above the threshold; the
 1910                     // job's rest run (none if no child is left).
 1911                     let mut rest = ALL_MOVES;
 1912                     if let Some((code, group)) =
 1913                         lead_fields(inner.leads[i].load(Relaxed), inner.read_tag)
 1914                     {
 1915                         if code > inner.lead_threshold && key >> 63 == 0 {
 1916                             let bucket = lead_bucket(code);
 1917                             lead_counts[bucket] += 1;
 1918                             if moves_after(&jobs[i]) & !group == 0 {
 1919                                 key |= 1 << 63;
 1920                             }
 1921                             // (With the bucket, for `ST_SCATTER`.)
 1922                             rest = ALL_MOVES & !group | (bucket as u32) << 24;
 1923                         }
 1924                     }
 1925                     *bufs.rest_children_ptr.0.add(i) = rest;
 1926                 }
 1927                 let bucket = bufs.splitters.partition_point(|&s| s <= key);
 1928                 *keys.add(i) = key;
 1929                 *key_buckets.add(i) = bucket as u8;
 1930                 counts[bucket] += 1;
 1931             }
 1932             let bucket_counts = bufs.bucket_counts_ptr.0.add(task * NBUCKETS);
 1933             std::ptr::copy_nonoverlapping(counts.as_ptr(), bucket_counts, NBUCKETS);
 1934             if inner.split {
 1935                 let lead_out = bufs.lead_counts_ptr.0.add(task * LEAD_BUCKETS);
 1936                 std::ptr::copy_nonoverlapping(lead_counts.as_ptr(), lead_out, LEAD_BUCKETS);
 1937             }
 1938         }
 1939         ST_SCATTER => {
 1940             let lo = task * SORT_CHUNK;
 1941             let hi = (lo + SORT_CHUNK).min(jobs.len());
 1942             let (keys, keys_tmp) = (bufs.keys_ptr.0.cast_const(), bufs.keys_tmp_ptr.0);
 1943             let key_buckets = bufs.key_buckets_ptr.0.cast_const();
 1944             // (`ST_KEYS`'s counts, turned into positions by `sort_stage_done`.)
 1945             let positions = std::slice::from_raw_parts_mut(
 1946                 bufs.bucket_counts_ptr.0.add(task * NBUCKETS),
 1947                 NBUCKETS,
 1948             );
 1949             for i in lo..hi {
 1950                 let pos = &mut positions[*key_buckets.add(i) as usize];
 1951                 *keys_tmp.add(*pos as usize) = *keys.add(i);
 1952                 *pos += 1;
 1953             }
 1954             if inner.split {
 1955                 // The first-part runs, at their positions (`sort_stage_done`).
 1956                 let lead_positions = std::slice::from_raw_parts_mut(
 1957                     bufs.lead_counts_ptr.0.add(task * LEAD_BUCKETS),
 1958                     LEAD_BUCKETS,
 1959                 );
 1960                 for i in lo..hi {
 1961                     let rest = *bufs.rest_children_ptr.0.add(i);
 1962                     if rest == ALL_MOVES {
 1963                         continue;
 1964                     }
 1965                     let group = ALL_MOVES & !rest;
 1966                     let pos = &mut lead_positions[(rest >> 24) as usize];
 1967                     *bufs.runs_ptr.0.add(*pos as usize) = i as u64 | u64::from(group) << 32;
 1968                     *pos += 1;
 1969                 }
 1970             }
 1971         }
 1972         _ => {
 1973             let lo = bufs.bucket_starts[task] as usize;
 1974             let hi = bufs.bucket_starts[task + 1] as usize;
 1975             let run = std::slice::from_raw_parts_mut(bufs.keys_tmp_ptr.0.add(lo), hi - lo);
 1976             // (Keys are distinct: the index breaks ties, as in a stable
 1977             // sort.)
 1978             run.sort_unstable();
 1979             if inner.split {
 1980                 // The rest runs: job indices (the jobs stay in place).
 1981                 let runs = bufs.runs_ptr.0.add(bufs.first_runs + lo);
 1982                 for (p, &key) in run.iter().enumerate() {
 1983                     let i = key_index(key);
 1984                     let children = *bufs.rest_children_ptr.0.add(i) & ALL_MOVES;
 1985                     *runs.add(p) = i as u64 | u64::from(children) << 32;
 1986                 }
 1987                 return;
 1988             }
 1989             let out = bufs.out_ptr.0.add(lo);
 1990             for (p, &key) in run.iter().enumerate() {
 1991                 if let Some(&ahead) = run.get(p + 16) {
 1992                     prefetch(&jobs[key_index(ahead)], false);
 1993                 }
 1994                 *out.add(p) = jobs[key_index(key)];
 1995             }
 1996         }
 1997     }
 1998 }

 2000 /// All tasks of sort stage `stage` are done: the next stage and its number
 2001 /// of tasks, or None when the jobs are sorted.
 2002 fn sort_stage_done(inner: &mut Inner, stage: usize) -> Option<(usize, usize)> {
 2003     inner.sort_stages_ended[stage - ST_KEYS] = Instant::now();
 2004     let bufs = &mut inner.b;
 2005     match stage {
 2006         ST_KEYS => {
 2007             // Counts to the first position of each (chunk, bucket); a
 2008             // bucket is a task of `ST_BUCKETS`.
 2009             let chunks = bufs.bucket_counts.len() / NBUCKETS;
 2010             bufs.bucket_starts.clear();
 2011             let mut pos = 0u32;
 2012             for bucket in 0..NBUCKETS {
 2013                 bufs.bucket_starts.push(pos);
 2014                 for chunk in 0..chunks {
 2015                     let count = &mut bufs.bucket_counts[chunk * NBUCKETS + bucket];
 2016                     (*count, pos) = (pos, pos + *count);
 2017                 }
 2018             }
 2019             bufs.bucket_starts.push(pos);
 2020             if inner.split {
 2021                 // The first-part runs by bucket, then chunk (job order).
 2022                 let mut pos = 0u32;
 2023                 for bucket in 0..LEAD_BUCKETS {
 2024                     for chunk in 0..chunks {
 2025                         let count = &mut bufs.lead_counts[chunk * LEAD_BUCKETS + bucket];
 2026                         (*count, pos) = (pos, pos + *count);
 2027                     }
 2028                 }
 2029                 bufs.first_runs = pos as usize;
 2030             }
 2031             Some((ST_SCATTER, chunks))
 2032         }
 2033         ST_SCATTER => Some((ST_BUCKETS, NBUCKETS)),
 2034         _ if inner.split => {
 2035             let n = inner.jobs.len();
 2036             inner.first_runs = u32::try_from(bufs.first_runs).unwrap_or(u32::MAX);
 2037             // SAFETY: the tasks wrote the first-part runs and a rest run per
 2038             // job (and sorted `keys_tmp`).
 2039             unsafe {
 2040                 inner.runs.set_len(bufs.first_runs + n);
 2041                 inner.feasible = bufs.first_runs
 2042                     + std::slice::from_raw_parts(bufs.keys_tmp_ptr.0.cast_const(), n)
 2043                         .partition_point(|&k| k >> 63 == 0);
 2044             }
 2045             None
 2046         }
 2047         _ => {
 2048             let n = inner.jobs.len();
 2049             // SAFETY: the tasks wrote all of `out` (and sorted `keys_tmp`).
 2050             unsafe {
 2051                 bufs.out.set_len(n);
 2052                 inner.feasible = std::slice::from_raw_parts(bufs.keys_tmp_ptr.0.cast_const(), n)
 2053                     .partition_point(|&k| k >> 63 == 0);
 2054             }
 2055             std::mem::swap(&mut inner.jobs, &mut bufs.out);
 2056             None
 2057         }
 2058     }
 2059 }

 2061 /// Do a task of `ctx`'s parallel build, if one is left (false otherwise);
 2062 /// the last task of a stage opens the next one or publishes the iteration.
 2063 // Builds in stages by all waiting threads: l5 (not measured alone); on all
 2064 // threads and without page faults since l6-share-a (-38.8% -c 1 at 48T).
 2065 fn build_task(
 2066     scratch: &mut Lane,
 2067     solver: &Solver,
 2068     pool: &Pool,
 2069     ctx: &Arc<CubeCtx>,
 2070     thread_id: usize,
 2071 ) -> bool {
 2072     let (_, count, next) = task_fields(ctx.task.load(Acquire));
 2073     if next >= count {
 2074         return false;
 2075     }
 2076     let (stage, count, task) = task_fields(ctx.task.fetch_add(1, AcqRel));
 2077     if task >= count {
 2078         return false;
 2079     }
 2080     // The stage cannot end while this task is not done; tasks write
 2081     // disjoint parts of the buffers (through the `*_ptr` copies), which are
 2082     // read only in later stages.
 2083     let inner = ctx.inner();
 2084     let start_ms = if pool.trace_bits & TRACE_TASKS != 0 {
 2085         ctx.started.elapsed().as_secs_f32() * 1e3
 2086     } else {
 2087         0.0
 2088     };
 2089     // SAFETY: see above.
 2090     unsafe {
 2091         match stage {
 2092             ST_EXT => {
 2093                 let chunk = if inner.b.extension_moves == 1 {
 2094                     EXT_CHUNK
 2095                 } else {
 2096                     1
 2097                 };
 2098                 let lo = task * chunk;
 2099                 let hi = (lo + chunk).min(inner.jobs.len());
 2100                 scratch.root = ctx.root;
 2101                 scratch.root_inverse = ctx.root_inverse;
 2102                 // (Parallel extensions run only with a filter that does not
 2103                 // dedup: the children that `kept_children` counted.)
 2104                 scratch.run_state.sym_filter = ctx.sym_filter;
 2105                 scratch.jobs.clear();
 2106                 scratch.jobs.extend_from_slice(&inner.jobs[lo..hi]);
 2107                 for _ in 0..inner.b.extension_moves {
 2108                     scratch.extend_jobs(solver);
 2109                 }
 2110                 let bufs = &inner.b;
 2111                 let first = bufs.child_offsets[task] as usize;
 2112                 let children = bufs.child_offsets[task + 1] as usize - first;
 2113                 // `kept_children` counted exactly the children that
 2114                 // `extend_jobs` keeps (the same `keep` filter on the same
 2115                 // canonical moves; a filter that dedups never extends in
 2116                 // parallel). (Should they differ, the copy still reads no
 2117                 // more jobs than there are.)
 2118                 debug_assert_eq!(scratch.jobs.len(), children);
 2119                 let out = bufs.out_ptr.0.add(first);
 2120                 let copied = scratch.jobs.len().min(children);
 2121                 std::ptr::copy_nonoverlapping(scratch.jobs.as_ptr(), out, copied);
 2122             }
 2123             _ => sort_task(inner, stage, task),
 2124         }
 2125     }
 2126     if pool.trace_bits & TRACE_TASKS != 0 && task < 256 {

 2418     /// Visit the active lanes round-robin (as `run_lanes` does) until none
 2419     /// is left (true) or `stop` is set (false). Job ends are handled by
 2420     /// `on_end` (which may swap another lane into position i), the rest
 2421     /// once per round by `maintain`. (Kept apart from the cold code, and
 2422     /// placed at a fixed offset of a 64-byte line: see `crate::hot`.)
 2423     // The solver is a parameter (noalias: the table pointer stays in a
 2424     // register): -3.0% -c 0, -2.6% -c 1, +3.8% bench share at 48T, -2.0% -c 0
 2425     // at 1T (l6-share-2).
 2426     #[inline(never)]
 2427     #[cfg_attr(
 2428         all(target_arch = "x86_64", target_os = "linux"),
 2429         unsafe(link_section = ".text.hot_share")
 2430     )]
 2431     fn rounds(&mut self, solver: &Solver, stop: &AtomicBool, round: &mut u32) -> bool {
 2432         while self.active_lanes > 0 {
 2433             let mut active = self.active_lanes;
 2434             // The lanes by pointer (`next` to `end`), as in `run_lanes`: an
 2435             // index cost a multiply (by the lane's size) at the start of
 2436             // every visit's value chain. (With m8's unsafe pass, whose loop
 2437             // alone was -1.3% in share mode at 1T, and share pad 16, on
 2438             // def2cd0: share +2.5% at 1T; -c 1 -2.1%, -c 0 -2.5%, bench share
 2439             // +2.5% at 48T.) The pointers are made again after `on_end`,
 2440             // which may swap lanes.
 2441             let mut lanes = self.lanes.as_mut_ptr();
 2442             // SAFETY: `active <= lanes.len()`.
 2443             let (mut next, mut end) = (lanes, unsafe { lanes.add(active) });
 2444             while next < end {
 2445                 // SAFETY: `next` is one of the active lanes, and nothing
 2446                 // else accesses it during the visit.
 2447                 let lane = unsafe { &mut (*next).lane };
 2448                 let value = lane.pending_value(solver);
 2449                 let outcome = match lane.phase {
 2450                     Phase::WaitOther => {
 2451                         if lane.resolve_other(value, solver) {
 2452                             // SAFETY: at most one past the active lanes.
 2453                             next = unsafe { next.add(1) };
 2454                             continue;
 2455                         }
 2456                         lane.advance(solver)
 2457                     }
 2458                     Phase::WaitMoved => lane.resolve_moved(value, solver),
 2459                 };
 2460                 match outcome {
 2461                     // SAFETY: as above.
 2462                     Step::Pending => next = unsafe { next.add(1) },
 2463                     outcome => {
 2464                         // SAFETY: `next` points into `lanes`.
 2465                         let mut i = unsafe { next.offset_from(lanes) }.cast_unsigned();
 2466                         if self.on_end(i, matches!(outcome, Step::Found)) {
 2467                             i += 1;
 2468                         }
 2469                         active = self.active_lanes;
 2470                         lanes = self.lanes.as_mut_ptr();
 2471                         // SAFETY: `i <= active <= lanes.len()`.
 2472                         (next, end) = unsafe { (lanes.add(i), lanes.add(active)) };
 2473                     }
 2474                 }
 2475             }
 2476             *round = round.wrapping_add(1);
 2477             self.counts[C_ROUNDS] += 1;
 2478             self.counts[C_VISITS] += active as u64;
 2479             if active < self.lanes.len()
 2480                 || self.pool.solved.load(Relaxed) != self.solved
 2481                 || self.pool.hungry.load(Relaxed) != 0
 2482             {
 2483                 self.maintain(*round);
 2484             }
 2485             if stop.load(Relaxed) {
 2486                 return false;
 2487             }
 2488         }
 2489         true
 2490     }

 2849     fn take(&mut self, li: usize, slot: usize) -> bool {

 2851         // the slot of a claimed or ended job, or one that is `Some`).
 2852         let loc = unsafe { joined(&mut self.locals, slot) };
 2853         if loc.next < loc.end && loc.ctx.done.load(Relaxed) {
 2854             // (dropped)
 2855             loc.pending -= loc.end - loc.next;
 2856             loc.next = loc.end;
 2857         }

 2859             let loc = unsafe { joined(&mut self.locals, slot) };
 2860             if loc.next >= loc.end {
 2861                 return false;
 2862             }

 2864             loc.next += 1;

 2953     /// Claim more jobs of the cube in `slot` (none must be unstarted).
 2954     fn claim(&mut self, slot: usize) -> bool {
 2955         // SAFETY: `slot` is joined (see `take`).
 2956         let loc = unsafe { joined(&mut self.locals, slot) };
 2957         debug_assert!(loc.next >= loc.end);
 2958         let divisor = 2 * self.pool.nthreads;
 2959         // SAFETY: `Pool::new` makes `nthreads` at least 1, and below 2^31
 2960         // (it allocates a mailbox of over 3 KB per thread). (A value-range
 2961         // hint: no division-by-zero check.)
 2962         unsafe { std::hint::assert_unchecked(divisor != 0) };
 2963         // (Chunks shrink as the iteration runs out: l5, not measured alone.)
 2964         let chunk = (loc.ctx.unclaimed() / divisor).clamp(1, CHUNK_MAX);
 2965         let Some((first, count)) = loc.ctx.claim(chunk) else {
 2966             return false;
 2967         };
 2968         loc.next = first;
 2969         loc.end = first + count;
 2970         loc.pending += count;
 2971         loc.claimed += u64::from(count);
 2972         let inner = loc.ctx.inner();
 2973         let mut jobs = (first, loc.end);
 2974         if inner.phases > 1 {
 2975             // (A first pass's entries: per phase, all the jobs again.)
 2976             let per_phase = inner.phase_entries;
 2977             // SAFETY: a first pass has at least one entry per phase
 2978             // (`finish_build`; a value-range hint).
 2979             unsafe { std::hint::assert_unchecked(per_phase != 0) };
 2980             let mut entry = first;
 2981             while entry < loc.end {
 2982                 let phase = entry / per_phase;
 2983                 let phase_end = ((phase + 1) * per_phase).min(loc.end);
 2984                 // SAFETY: claimed entries are below `phases * per_phase`,
 2985                 // and `phases <= MAX_PHASES`.
 2986                 debug_assert!((phase as usize) < MAX_PHASES);
 2987                 unsafe {
 2988                     *loc.phase_pending.get_unchecked_mut(phase as usize) += phase_end - entry;
 2989                     *loc.phase_claimed.get_unchecked_mut(phase as usize) +=
 2990                         u64::from(phase_end - entry);
 2991                 }
 2992                 entry = phase_end;
 2993             }
 2994             let base = first / per_phase * per_phase;
 2995             jobs = (first - base, (loc.end - base).min(per_phase));
 2996         }
 2997         // The claimed jobs and their statistics (written at their ends):
 2998         // the lines were written by other threads (the jobs by the build,
 2999         // the statistics by the publication), and lanes take the jobs one
 3000         // after another. (l6-share-2; screening run without it: -c 1 3.06 vs
 3001         // 2.36-2.42 ms, bench share 799 vs 845/s at 48T.)
 3002         // (By address, which a prefetch needs not be valid at: no bounds
 3003         // checks. The claimed jobs are in range anyway.)
 3004         let (first, end) = (jobs.0 as usize, jobs.1 as usize);
 3005         let (job_lines, stat_lines) = (inner.jobs.as_ptr(), inner.stats.as_ptr());
 3006         if inner.split {
 3007             // (The runs' jobs are scattered.)
 3008             // SAFETY: claimed entries are below the run count (`first..end`
 3009             // is within `runs`).
 3010             debug_assert!(end <= inner.runs.len());
 3011             for &run in unsafe { inner.runs.get_unchecked(first..end) } {
 3012                 let job = run_job(run) as usize;
 3013                 crate::prefetch::read(job_lines.wrapping_add(job));
 3014                 crate::prefetch::write(stat_lines.wrapping_add(job));
 3015             }
 3016             return true;
 3017         }
 3018         for i in (first..end).step_by(4) {
 3019             crate::prefetch::read(job_lines.wrapping_add(i));
 3020         }
 3021         crate::prefetch::read(job_lines.wrapping_add(end - 1));
 3022         for i in (first..end).step_by(16) {
 3023             crate::prefetch::write(stat_lines.wrapping_add(i));
 3024         }
 3025         crate::prefetch::write(stat_lines.wrapping_add(end - 1));
 3026         if inner.record_leads {
 3027             let lead_lines = inner.leads.as_ptr();
 3028             for i in (first..end).step_by(16) {
 3029                 crate::prefetch::write(lead_lines.wrapping_add(i));
 3030             }
 3031         }
 3032         true
 3033     }

 3035     /// If no job of `slot` is pending, count its finished jobs down from
 3036     /// the cube's outstanding jobs, and build the next iteration if they
 3037     /// were the last.

 3040         let loc = unsafe { joined(&mut self.locals, slot) };

 3042             // drop the unstarted jobs
 3043             loc.pending -= loc.end - loc.next;
 3044             loc.next = loc.end;
 3045         }
 3046         if loc.pending != 0 || loc.claimed == 0 {
 3047             return;
 3048         }
 3049         let claimed = loc.claimed;
 3050         loc.claimed = 0;
 3051         if self.pool.trace {

 3061                 .rarely_written

 3065                 .rarely_written

 3096     fn release(&mut self, slot: usize) {

 3098         let Some(loc) = self.locals.get_mut(slot).and_then(Option::take) else {

 3358     /// The lane and node to split for another lane: the active lane (other
 3359     /// than `exclude`) with the shallowest node (remaining depth at least
 3360     /// `min_r`) that has moves of two faces left. (lane, depth, moves to
 3361     /// give away.)
 3362     fn find_split(&self, min_r: u8, exclude: usize) -> Option<(usize, usize, u32)> {
 3363         let mut best = None;
 3364         let mut best_r = min_r - 1;
 3365         // (Run once per round while threads wait for work: no bounds checks.)
 3366         // SAFETY: `active_lanes <= lanes.len()`; a lane's depth is below
 3367         // MAXD (see `Lane::advance`).
 3368         let active = unsafe { self.lanes.get_unchecked(..self.active_lanes) };
 3369         for (index, sl) in active.iter().enumerate() {
 3370             if index == exclude || sl.slot == NO_SLOT {
 3371                 continue;
 3372             }

 3375             for depth in 0..=lane.depth {
 3376                 // SAFETY: see above.
 3377                 let node = unsafe { lane.stack.get_unchecked(depth) };
 3378                 if node.remaining_depth <= best_r {
 3379                     break;
 3380                 }
 3381                 if node.lower_bound > node.remaining_depth || node.todo == 0 {
 3382                     continue;
 3383                 }
 3384                 // (An active lane's pending child is `pending_move`.)
 3385                 let face = if depth < lane.depth {
 3386                     // SAFETY: `depth + 1 <= lane.depth < MAXD`.
 3387                     unsafe { lane.stack.get_unchecked(depth + 1) }.mv / 3
 3388                 } else {
 3389                     lane.pending_move / 3
 3390                 };

 3392                     best = Some((index, depth, give));

 3576     /// Drop the jobs of solved cubes (and a first pass's entries as long as
 3577     /// its shortest solution so far), then give the lanes new work.
 3578     fn cancel_done(&mut self, round: u32) {
 3579         let mut i = self.active_lanes;
 3580         let mut dropped = false;
 3581         while i > 0 {
 3582             i -= 1;
 3583             // SAFETY: `i < active_lanes <= lanes.len()`, and an active lane's
 3584             // slot is joined.
 3585             let sl = unsafe { self.lanes.get_unchecked(i) };
 3586             let ctx = &unsafe { joined_ref(&self.locals, sl.slot) }.ctx;
 3587             // (A first pass's entries as long as its shortest solution so
 3588             // far are of no use.)
 3589             if !ctx.done.load(Relaxed)
 3590                 && sl.lane.target < ctx.rarely_written.pass_limit.load(Relaxed)
 3591             {
 3592                 continue;
 3593             }
 3594             self.end_job(i, false);
 3595             self.deactivate(i);
 3596             dropped = true;
 3597         }
 3598         if !dropped {
 3599             return;
 3600         }
 3601         // Count down the jobs of the cubes left with nothing pending (as each
 3602         // dropped lane's `restart` did before they were refilled together):
 3603         // else a first pass's dropped entries stayed outstanding until the
 3604         // cube was reported, and a thread leaving the pool before then
 3605         // released its cube with jobs not counted down. (Slots are never
 3606         // removed: `slot` stays below `locals.len()`.)
 3607         for slot in 0..self.locals.len() {
 3608             // SAFETY: see above.
 3609             if unsafe { self.locals.get_unchecked(slot) }.is_some() {
 3610                 self.settle(slot);
 3611             }
 3612         }

 3614         // dropped lane looked for work on its own, all threads took the open

 3617         // often ran all 18 tasks of its extension): -5.1% -c 1 at 48T (6 paired
 3618         // holds; 15/16-move scrambles -10/-9%, depth-15 states -12%; -c 0 and
 3619         // bench share neutral).
 3620         self.release_done();
 3621         self.fill_idle(round);
 3622     }

 4290     /// Entries of a first pass dropped as useless (a solution as short was
 4291     /// found) are counted down at once, also when they were the last
 4292     /// pending jobs of the thread's share of a cube not solved yet. (They
 4293     /// were left outstanding until the cube was reported, and a thread
 4294     /// leaving the pool before released them so: the debug check in
 4295     /// `Worker::release` failed.) Two workers driven by hand, in one thread.
 4296     #[test]
 4297     fn first_pass_dropped_entries_are_counted_down() {
 4298         let corners = Cocsep::truncated(4);
 4299         let table = H48Table::zeroed(0, 2);
 4300         let solver = Solver::new(&corners, &table);
 4301         let cube = first_pass_cube(&solver);
 4302         let get = |i: usize| (i == 0).then_some(cube);
 4303         // (Two threads, one cube: not few, so builds are serial.)
 4304         let pool = Pool::new(&get, 1, 2);
 4305         let reported = Mutex::new(Vec::new());
 4306         let mut report = |i, solution, _, _| reported.lock().push((i, solution));
 4307         let mut report_too = |i, solution, _, _| reported.lock().push((i, solution));
 4308         let mut first = hand_worker(&pool, &solver, 0, 2, &mut report);
 4309         let mut second = hand_worker(&pool, &solver, 1, 2, &mut report_too);
 4310         // The first worker runs an entry of phase 0, ...
 4311         assert!(first.fresh(0));
 4312         first.active_lanes = 1;
 4313         let first_slot = first.lanes[0].slot;
 4314         let ctx = first.locals[first_slot].as_ref().unwrap().ctx.clone();
 4315         assert!(ctx.inner().phases >= 3 && first.lanes[0].phase == 0);
 4316         // ... the second one an entry of a later phase, after all the others.
 4317         let second_slot = second.local(&ctx).unwrap();
 4318         assert!(run_up_to_phase(&mut second, 0, second_slot, 1));
 4319         second.active_lanes = 1;
 4320         while take_next(&mut second, 1, second_slot) {
 4321             second.end_job(1, false);
 4322         }
 4323         assert_eq!(ctx.unclaimed(), 0);
 4324         // A solution of that phase was found: the second worker drops its
 4325         // entry, its last pending job of the cube.
 4326         let phase = second.lanes[0].phase;
 4327         {
 4328             let mut pass = ctx.pass.lock();
 4329             pass.candidate = vec![0];
 4330             pass.shortest = u32::from(phase) + 1;
 4331         }
 4332         let pass_limit = second.lanes[0].lane.target;
 4333         ctx.rarely_written.pass_limit.store(pass_limit, Relaxed);
 4334         pool.solved.fetch_add(1, Release);
 4335         second.maintain(1);
 4336         assert_eq!(second.active_lanes, 0);
 4337         assert!(!ctx.done.load(Relaxed));
 4338         assert!(second.counted_down(), "dropped entries left outstanding");
 4339         // Phase 0 ends: the solution is reported, nothing is outstanding.
 4340         first.end_job(0, false);
 4341         while take_next(&mut first, 0, first_slot) {
 4342             first.end_job(0, false);
 4343         }
 4344         first.tidy(first_slot);
 4345         assert!(ctx.done.load(Relaxed));
 4346         assert_eq!(ctx.outstanding.load(Relaxed), 0);
 4347         second.release_done();
 4348         let joined = first.locals.iter().chain(&second.locals);
 4349         assert!(joined.flatten().next().is_none());
 4350         assert_eq!(*reported.lock(), [(0, Some(vec![0]))]);
 4351     }

 4385     /// The staged sort (its tasks run in any order within a stage) gives
 4386     /// the order of `sort_jobs`.
 4387     #[test]
 4388     #[expect(clippy::cast_possible_truncation, reason = "jobs from random bits")]
 4389     fn staged_sort_matches_sort_jobs() {
 4390         let mut rng = crate::cubie::Rng(11);
 4391         for n in [8192usize, 20000, 43254] {
 4392             let jobs: Vec<Job> = (0..n)
 4393                 .map(|i| {
 4394                     let r = rng.next_u64();
 4395                     // many ties and a wide range of keys
 4396                     let stat = match r % 5 {
 4397                         0 => 0.0,
 4398                         1 => f32::from((r >> 8) as u8),
 4399                         _ => f32::from_bits((r >> 16) as u32 % 0x4000_0000 + 0x0080_0000),
 4400                     };
 4401                     Job {
 4402                         moves: [(r >> 40) as u8 % 18, 0, 0, (i % 251) as u8],
 4403                         len: 1,
 4404                         normal_bound: (i / 251) as u8,
 4405                         inverse_bound: 0,
 4406                         stat,
 4407                     }
 4408                 })
 4409                 .collect();
 4410             let mut want = jobs.clone();
 4411             super::super::sort_jobs(&mut want, &mut Vec::new());
 4412             // all feasible
 4413             let mut serial = jobs.clone();
 4414             assert_eq!(sort_serial(&mut serial, &mut Bufs::sized(), 255), n);
 4415             let id = |j: &Job| (j.moves, j.normal_bound, j.stat.to_bits());
 4416             assert!(serial.iter().map(id).eq(want.iter().map(id)), "n = {n}");

 4419             let feasible = |j: &&Job| j.normal_bound < target;
 4420             let want_t: Vec<Job> = want
 4421                 .iter()
 4422                 .filter(feasible)
 4423                 .chain(want.iter().filter(|j| !feasible(j)))
 4424                 .copied()
 4425                 .collect();
 4426             let feasible = want.iter().filter(feasible).count();
 4427             let mut serial = jobs.clone();

 4429                 sort_serial(&mut serial, &mut Bufs::sized(), target),
 4430                 feasible
 4431             );
 4432             assert!(serial.iter().map(id).eq(want_t.iter().map(id)), "n = {n}");

 4437             loop {
 4438                 run_sort_stage(&inner, stage, tasks);

 4441                     None => break,
 4442                 }
 4443             }
 4444             assert!(
 4445                 inner.jobs.iter().map(id).eq(want_t.iter().map(id)),
 4446                 "n = {n}"
