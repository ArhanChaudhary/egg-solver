Metric Groups:

Default:
  CPUs_utilized
       [Average CPU utilization]
       [(software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu else
        software@task\-clock\,name\=task\-clock@) / (duration_time * 1e9)]
  backend_cycles_idle
       [Backend stalls per cycle]
       [(stalled\-cycles\-backend / cpu\-cycles if
        has_event(stalled\-cycles\-backend) else 0)]
       [backend_cycles_idle > 0.2]
  branch_frequency
       [Branches per CPU second]
       [branches / (software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu else
        software@task\-clock\,name\=task\-clock@)]
  branch_miss_rate
       [Branch miss rate]
       [branch\-misses / branches]
       [branch_miss_rate > 0.05]
  cs_per_second
       [Context switches per CPU second]
       [software@context\-switches\,name\=context\-switches@ * 1e9 /
        (software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu else
        software@task\-clock\,name\=task\-clock@)]
  cycles_frequency
       [Cycles per CPU second]
       [cpu\-cycles / (software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu
        else software@task\-clock\,name\=task\-clock@)]
  frontend_cycles_idle
       [Frontend stalls per cycle]
       [(stalled\-cycles\-frontend / cpu\-cycles if
        has_event(stalled\-cycles\-frontend) else 0)]
       [frontend_cycles_idle > 0.1]
  insn_per_cycle
       [Instructions Per Cycle]
       [instructions / cpu\-cycles]
       [insn_per_cycle < 1]
  migrations_per_second
       [Process migrations to a new CPU per CPU second]
       [software@cpu\-migrations\,name\=cpu\-migrations@ * 1e9 /
        (software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu else
        software@task\-clock\,name\=task\-clock@)]
  page_faults_per_second
       [Page faults per CPU second]
       [software@page\-faults\,name\=page\-faults@ * 1e9 /
        (software@cpu\-clock\,name\=cpu\-clock@ if #target_cpu else
        software@task\-clock\,name\=task\-clock@)]
